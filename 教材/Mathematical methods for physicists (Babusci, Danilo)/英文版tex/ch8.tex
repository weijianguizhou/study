\chapter{Exercises and Complements IV}

\section{Sturm-Liouville Problem}

The general Sturm--Liouville eigenvalue problem is
\begin{equation}
-\frac{d}{dx}\!\left(p(x)\frac{dy}{dx}\right) + q(x)\,y = \lambda\,w(x)\,y,
\qquad x\in[a,b],
\label{eq:8.1.1}
\end{equation}
with self-adjoint boundary conditions
$\alpha_1 y(a)+\alpha_2 y'(a)=0$,
$\beta_1 y(b)+\beta_2 y'(b)=0$.

Eigenvalues $\lambda_n$ are real and ordered
$\lambda_0<\lambda_1<\lambda_2<\cdots$, with orthogonal eigenfunctions:
\begin{equation}
\int_a^b y_n(x)\,w(x)\,y_m(x)\,dx = N_n\delta_{nm}.
\label{eq:8.1.2}
\end{equation}

Completeness: any $f(x)$ satisfying the boundary conditions expands as
$f(x)=\sum_n c_n y_n(x)$ with
$c_n = \frac{1}{N_n}\int_a^b f(x)w(x)y_n(x)dx$.

\begin{exercise}
For $p=1$, $q=0$, $w=1$ on $[0,\pi]$ with Dirichlet BCs,
find $\lambda_n$, $y_n(x)$ and verify orthogonality.
\end{exercise}

\section{Green's Functions}

For $\hat L y = s(x)$, the solution is
\begin{equation}
y(x) = \int_a^b G(x,\xi)\,s(\xi)\,d\xi,
\label{eq:8.2.1}
\end{equation}
where the Green's function satisfies
$(\hat L-\lambda)G(x,\xi) = \delta(x-\xi)$ (or $\hat L G=\delta$ for $\lambda=0$).

Construction by eigenfunction expansion:
\begin{equation}
G(x,\xi) = \sum_{n=0}^{\infty}\frac{y_n(x)\,y_n(\xi)}{N_n(\lambda_n-\lambda)}.
\label{eq:8.2.2}
\end{equation}

Jump condition for second-order ODEs:
$\left.\frac{\partial G}{\partial x}\right|_{x=\xi^+}
-\left.\frac{\partial G}{\partial x}\right|_{x=\xi^-} = \frac{1}{p(\xi)}$.

\begin{exercise}
Construct $G(x,\xi)$ for $y''=s(x)$ on $[0,1]$ with $y(0)=y(1)=0$
and verify $\int_0^1 G(x,\xi)\,d\xi = x(1-x)/2$.
\end{exercise}

\section{Laguerre Polynomials, Associated Operators and PDE}

The Laguerre derivative is
$\hat\Lambda = x\frac{d^2}{dx^2}+(1-x)\frac{d}{dx}$.
It acts on Laguerre polynomials as
\begin{equation}
\hat\Lambda L_n(x) = -n\,L_n(x).
\label{eq:8.3.1}
\end{equation}

The Laguerre heat equation
$\partial_t F = \hat\Lambda F$ with $F(x,0)=x^n$
has solution
\begin{equation}
F(x,t) = L_n\!\left(\frac{x}{1-t}\right)(1-t)^n,
\qquad |t|<1.
\label{eq:8.3.2}
\end{equation}

Umbral representation:
$L_n(x)=(\hat\theta)_n\,e^{-x\hat\theta}$ with
$(\hat\theta)_n=\hat\theta(\hat\theta-1)\cdots(\hat\theta-n+1)$
and $\hat\theta e^{-x\hat\theta}=e^{-x\hat\theta}$.

\begin{exercise}
Verify \eqref{eq:8.3.1} using the umbral representation and derive the
generating function
$\sum_{n=0}^\infty L_n(x)t^n = e^{-xt/(1-t)}/(1-t)$.
\end{exercise}

\section{Appel Polynomials, Associated Operators and Partial Differential Equations}

Appel polynomials $A_n(x)$ satisfy
$\frac{d}{dx}A_n(x)=n\,A_{n-1}(x)$ and are generated by
\begin{equation}
\sum_{n=0}^{\infty} A_n(x)\,\frac{t^n}{n!}=e^{x t}\,A(t),
\qquad A(0)=1.
\label{eq:8.4.1}
\end{equation}

The associated derivative operator $\hat D_A$ defined by
$\hat D_A A_n(x)=n\,A_{n-1}(x)$ solves the Cauchy problem
$\partial_t F=\hat D_A F$, $F(x,0)=P(x)$ via
$F(x,t)=e^{t\hat D_A}P(x)$.

\begin{exercise}
For $A_n(x)=x^n$ (ordinary monomials), show
$\hat D_A=\frac{d}{dx}$ and recover Taylor's theorem from $e^{t\hat D_A}$.
\end{exercise}

\section{Riemann Function}

The Riemann method solves Cauchy problems for second-order hyperbolic
PDEs. For
\begin{equation}
\frac{\partial^2 u}{\partial x\partial y} + a(x,y)\frac{\partial u}{\partial x}
+ b(x,y)\frac{\partial u}{\partial y} + c(x,y)u = 0,
\label{eq:8.5.1}
\end{equation}
the Riemann--Green function $R(x,y;\xi,\eta)$ satisfies the adjoint equation
with characteristic boundary conditions.

The solution with Cauchy data on a non-characteristic curve is
\begin{equation}
u(P) = \frac12\left[u(A)R(P;A)+u(B)R(P;B)\right]
+ \frac12\int_{AB}\left(R\frac{\partial u}{\partial n}-u\frac{\partial R}{\partial n}\right)ds.
\label{eq:8.5.2}
\end{equation}

\begin{exercise}
For the 1D wave equation $u_{xy}=0$, show that $R=1$ and recover
d'Alembert's formula.
\end{exercise}

\section{Bessel Special Functions}

Bessel functions $J_n(x)$ satisfy
\begin{equation}
x^2J_n''+xJ_n'+(x^2-n^2)J_n=0,
\label{eq:8.6.1}
\end{equation}
with generating function
\begin{equation}
e^{\frac{x}{2}(t-1/t)} = \sum_{n=-\infty}^{\infty} J_n(x)\,t^n.
\label{eq:8.6.2}
\end{equation}

Recurrence relations:
\begin{align}
J_{n-1}+J_{n+1} &= \frac{2n}{x}J_n,\\
J_{n-1}-J_{n+1} &= 2J_n'.
\label{eq:8.6.3}
\end{align}

Orthogonality on finite intervals with weight $x$:
\begin{equation}
\int_0^a x\,J_n\!\left(\frac{\alpha_{nm}}{a}x\right)
J_n\!\left(\frac{\alpha_{nk}}{a}x\right)dx
= \frac{a^2}{2}J_{n+1}^2(\alpha_{nm})\,\delta_{mk},
\label{eq:8.6.4}
\end{equation}
where $\alpha_{nm}$ is the $m$-th zero of $J_n$.

Modified Bessel functions: $I_n(x)=i^{-n}J_n(ix)$ satisfy
$x^2I_n''+xI_n'-(x^2+n^2)I_n=0$.

Hankel functions: $H_n^{(1)}=J_n+iY_n$,
$H_n^{(2)}=J_n-iY_n$ for outgoing/incoming cylindrical waves.

\begin{exercise}
Derive the recurrence relations \eqref{eq:8.6.3} from the generating
function \eqref{eq:8.6.2}.
\end{exercise}

\begin{exercise}
Evaluate $\int_0^\infty e^{-ax}J_0(bx)\,dx=1/\sqrt{a^2+b^2}$ using the
Laplace transform of $J_0$.
\end{exercise}
