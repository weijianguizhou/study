# 熵权法

> **理论基础** → 参见 [[../../统计学分析方法/熵权法（Entropy Weight Method）|熵权法]]

熵权法利用信息熵度量指标变异程度来确定权重，指标变异越大则权重越大，客观性强。

## 三、计算步骤总结
1. **构建原始数据矩阵** $X_{m \times n}$
2. **指标正向化**：将成本型指标转换为效益型
3. **数据标准化**：计算比重矩阵 $P_{m \times n}$
4. **计算信息熵**：$E_j = -\frac{1}{\ln m} \sum_{i=1}^{m} p_{ij} \ln p_{ij}$
5. **计算信息效用值**：$D_j = 1 - E_j$
6. **计算熵权**：$w_j = D_j / \sum D_k$
7. **计算综合得分**：$S_i = \mathbf{w}^T \mathbf{p}_i$

## 四、Python 实现
### 4.1 基本熵权法

```python
import numpy as np

class EntropyWeight:
    """熵权法计算指标权重"""
    
    def __init__(self, data, cost_indices=None):
        """
        :param data: 原始数据矩阵 (m x n)
        :param cost_indices: 成本型指标索引列表
        """
        self.raw = np.array(data, dtype=float)
        self.m, self.n = self.raw.shape
        self.cost_indices = cost_indices if cost_indices else []
    
    def standardize(self):
        """数据标准化（比重法）"""
        data = self.raw.copy()
        
        # 指标正向化
        for j in self.cost_indices:
            max_val = data[:, j].max()
            min_val = data[:, j].min()
            data[:, j] = max_val - data[:, j]
        
        # 比重标准化
        col_sum = data.sum(axis=0)
        self.P = data / col_sum
        return self.P
    
    def calc_entropy(self):
        """计算信息熵"""
        self.standardize()
        
        # 避免 log(0)
        P_safe = np.where(self.P > 0, self.P, 1)
        
        k = 1 / np.log(self.m)
        self.E = -k * (P_safe * np.log(P_safe)).sum(axis=0)
        return self.E
    
    def calc_weights(self):
        """计算熵权"""
        self.calc_entropy()
        
        self.D = 1 - self.E  # 信息效用值
        self.weights = self.D / self.D.sum()
        return self.weights
    
    def calc_scores(self):
        """计算综合评价得分"""
        self.calc_weights()
        self.scores = self.P @ self.weights
        return self.scores
    
    def summary(self):
        """输出完整结果"""
        weights = self.calc_weights()
        scores = self.calc_scores()
        
        print("=" * 60)
        print("熵权法分析结果")
        print("=" * 60)
        print(f"{'指标':<10}{'信息熵':<10}{'信息效用值':<12}{'权重':<10}")
        print("-" * 60)
        for j in range(self.n):
            print(f"  X{j+1:<8}{self.E[j]:<10.4f}{self.D[j]:<12.4f}{weights[j]:<10.4f}")
        print("-" * 60)
        print(f"  权重和: {weights.sum():.4f}")
        
        ranking = np.argsort(-scores)
        print(f"\n综合评价得分排名:")
        for rank, idx in enumerate(ranking):
            print(f"  第{rank+1}名: 对象{idx+1}, 得分={scores[idx]:.4f}")
        
        return weights, scores

# 示例：城市综合评价
data = [
    [3200, 85,  92, 78, 6.5],
    [2800, 80,  88, 82, 7.0],
    [3500, 90,  85, 75, 5.8],
    [2600, 75,  90, 88, 7.5],
    [3800, 88,  80, 70, 5.2],
    [3000, 82,  95, 85, 6.8],
]

# 成本型指标：第5列（PM2.5）
ew = EntropyWeight(data, cost_indices=[4])
weights, scores = ew.summary()
```

### 4.2 熵权-TOPSIS 结合

```python
import numpy as np

class EntropyTOPSIS:
    """熵权法 + TOPSIS 综合评价"""
    
    def __init__(self, data, cost_indices=None):
        self.raw = np.array(data, dtype=float)
        self.m, self.n = self.raw.shape
        self.cost_indices = cost_indices if cost_indices else []
    
    def entropy_weights(self):
        """熵权法计算权重"""
        data = self.raw.copy()
        
        # 正向化
        for j in self.cost_indices:
            data[:, j] = data[:, j].max() - data[:, j]
        
        # 标准化
        P = data / data.sum(axis=0)
        P_safe = np.where(P > 0, P, 1)
        
        k = 1 / np.log(self.m)
        E = -k * (P_safe * np.log(P_safe)).sum(axis=0)
        D = 1 - E
        self.weights = D / D.sum()
        return self.weights
    
    def topsis(self):
        """TOPSIS 评价"""
        # 向量标准化
        norm = np.sqrt((self.raw ** 2).sum(axis=0))
        R = self.raw / norm
        
        # 加权标准化
        V = R * self.weights
        
        # 正负理想解
        A_plus = np.zeros(self.n)
        A_minus = np.zeros(self.n)
        for j in range(self.n):
            if j in self.cost_indices:
                A_plus[j] = V[:, j].min()
                A_minus[j] = V[:, j].max()
            else:
                A_plus[j] = V[:, j].max()
                A_minus[j] = V[:, j].min()
        
        # 距离
        D_plus = np.sqrt(((V - A_plus) ** 2).sum(axis=1))
        D_minus = np.sqrt(((V - A_minus) ** 2).sum(axis=1))
        
        # 贴近度
        self.C = D_minus / (D_plus + D_minus + 1e-10)
        return self.C
    
    def evaluate(self):
        """执行评价"""
        self.entropy_weights()
        scores = self.topsis()
        
        print("熵权-TOPSIS 评价结果")
        print("=" * 50)
        print(f"熵权: {np.round(self.weights, 4)}")
        print("-" * 50)
        
        ranking = np.argsort(-scores)
        for rank, idx in enumerate(ranking):
            print(f"  第{rank+1}名: 对象{idx+1}, 贴近度={scores[idx]:.4f}")
        
        return scores, self.weights

# 使用示例
data = [
    [85, 90, 78, 92, 88],
    [78, 85, 88, 80, 82],
    [92, 75, 82, 88, 90],
    [88, 82, 90, 85, 78],
    [75, 92, 85, 90, 85]
]

model = EntropyTOPSIS(data)
scores, weights = model.evaluate()
```

## 五、实际案例：区域经济发展评价
```python
import numpy as np

def regional_economy_evaluation():
    """区域经济发展水平评价"""
    
    regions = ['北京', '上海', '广州', '深圳', '杭州', '成都', '武汉', '南京']
    
    # 指标: GDP(亿元), 人均收入(元), 失业率(%), 教育支出(亿元), 环境质量指数
    # 失业率为成本型指标
    indicators = ['GDP', '人均收入', '失业率', '教育支出', '环境质量']
    cost_idx = [2]  # 失业率为成本型
    
    data = np.array([
        [36102, 69434, 1.4, 850, 78],
        [32679, 72232, 1.6, 780, 75],
        [25019, 68304, 1.8, 620, 82],
        [27670, 64878, 1.5, 680, 85],
        [18109, 64198, 2.0, 450, 88],
        [17717, 48593, 2.3, 380, 80],
        [15616, 50362, 2.1, 350, 76],
        [14817, 59868, 1.9, 420, 84]
    ])
    
    # 正向化处理（失业率取倒数）
    data_processed = data.copy()
    for j in cost_idx:
        data_processed[:, j] = 1.0 / data[:, j] * 1000  # 放大避免数值太小
    
    # 熵权法计算权重
    m, n = data_processed.shape
    P = data_processed / data_processed.sum(axis=0)
    P_safe = np.where(P > 0, P, 1)
    
    k = 1 / np.log(m)
    E = -k * (P_safe * np.log(P_safe)).sum(axis=0)
    D = 1 - E
    weights = D / D.sum()
    
    # 综合得分
    scores = P @ weights
    
    print("区域经济发展水平评价")
    print("=" * 60)
    print(f"{'地区':<8}", end='')
    for ind in indicators:
        print(f"{ind:<10}", end='')
    print(f"{'综合得分':<10}")
    print("-" * 60)
    
    ranking = np.argsort(-scores)
    for rank, idx in enumerate(ranking):
        print(f"{regions[idx]:<8}", end='')
        for val in data[idx]:
            print(f"{val:<10}", end='')
        print(f"{scores[idx]:<10.4f}")
    
    print("\n熵权法权重:")
    for ind, w in zip(indicators, weights):
        print(f"  {ind}: {w:.4f}")

regional_economy_evaluation()
```

## 七、注意事项
1. **数据正值要求**：熵权法要求数据为正值，负值需要预处理
2. **零值处理**：$p_{ij} = 0$ 时，$0 \cdot \ln 0$ 定义为 0
3. **指标数量影响**：指标过多可能导致各权重趋于均匀
4. **数据一致性**：如果某指标所有对象取值相同，则 $E_j = 1$，$w_j = 0$
5. **与主观方法结合**：纯客观赋权可能忽视指标的实际重要性，建议结合主观方法

## 八、关键要点总结
| 要点 | 内容 |
|:--|:--|
| 理论基础 | 信息论，香农熵 |
| 核心公式 | $E_j = -\frac{1}{\ln m} \sum p_{ij} \ln p_{ij}$ |
| 权重公式 | $w_j = (1 - E_j) / \sum(1 - E_k)$ |
| 数据要求 | 正值、无量纲化 |
| 变异越大 | 权重越大 |
| 变异越小 | 权重越小 |
| 完全相同 | 权重为零 |
| 优势 | 客观性强，不依赖专家判断 |
| 局限 | 忽视指标实际重要性 |

熵权法是最常用的客观赋权方法之一，在综合评价中应用广泛。理解其信息论原理，正确处理数据，可以为评价问题提供科学的权重依据。在实际应用中，建议将熵权法与 AHP 等主观方法结合使用，以获得更加全面和合理的权重。
