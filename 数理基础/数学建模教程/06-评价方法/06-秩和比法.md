# 秩和比法（RSR）

> **理论基础** → 参见 [[../../统计学分析方法/秩和比法（Rank-Sum Ratio）|秩和比法]]

秩和比法(RSR)以非参数统计为基础，利用秩次的平均值构建综合评价指标，适用于多指标排序和分档。

## 四、Python 实现
### 4.1 基本 RSR 计算

```python
import numpy as np
from scipy import stats

class RankSumRatio:
    """秩和比法综合评价"""
    
    def __init__(self, data, weights=None, cost_indices=None):
        """
        :param data: 原始数据矩阵 (m x n)
        :param weights: 指标权重 (长度为 n)
        :param cost_indices: 成本型指标索引列表
        """
        self.raw = np.array(data, dtype=float)
        self.m, self.n = self.raw.shape
        self.weights = np.ones(self.n) / self.n if weights is None else np.array(weights)
        self.cost_indices = cost_indices if cost_indices else []
    
    def standardize(self):
        """指标正向化"""
        data = self.raw.copy()
        for j in self.cost_indices:
            data[:, j] = data[:, j].max() - data[:, j]
        return data
    
    def calc_ranks(self):
        """计算秩次矩阵"""
        data = self.standardize()
        self.ranks = np.zeros_like(data)
        
        for j in range(self.n):
            # 升序排列
            sorted_indices = np.argsort(data[:, j])
            rank = np.zeros(self.m)
            
            i = 0
            while i < self.m:
                # 处理相同值
                k = i
                while k < self.m - 1 and data[sorted_indices[k], j] == data[sorted_indices[k+1], j]:
                    k += 1
                
                # 平均秩次
                avg_rank = (i + k) / 2.0 + 1
                for idx in range(i, k + 1):
                    rank[sorted_indices[idx]] = avg_rank
                i = k + 1
            
            self.ranks[:, j] = rank
        
        return self.ranks
    
    def calc_rsr(self):
        """计算秩和比"""
        self.calc_ranks()
        
        self.rsr = np.zeros(self.m)
        for i in range(self.m):
            self.rsr[i] = np.sum(self.weights * self.ranks[i]) / self.m
        
        # 归一化
        self.rsr_normalized = (self.rsr - self.rsr.min()) / (self.rsr.max() - self.rsr.min())
        
        return self.rsr
    
    def frequency_distribution(self):
        """计算频率分布"""
        # 按 RSR 排序
        sorted_idx = np.argsort(self.rsr)
        self.rsr_sorted = self.rsr[sorted_idx]
        
        # 计算累计频率
        self.cum_freq = np.arange(1, self.m + 1) / self.m
        
        # 概率单位变换 (Probit)
        self.probit = stats.norm.ppf(self.cum_freq) + 5
        
        return self.rsr_sorted, self.cum_freq, self.probit
    
    def regression(self):
        """RSR 与 Probit 的回归分析"""
        self.frequency_distribution()
        
        # 线性回归: Probit = a + b * RSR
        slope, intercept, r_value, p_value, std_err = stats.linregress(
            self.rsr_sorted, self.probit
        )
        
        self.regression_params = {
            'slope': slope,
            'intercept': intercept,
            'r_squared': r_value ** 2,
            'p_value': p_value
        }
        
        return self.regression_params
    
    def classify(self, n_classes=5):
        """分档评价"""
        self.calc_rsr()
        self.regression()
        
        # 确定分档临界值
        if n_classes == 5:
            probit_thresholds = [3.4, 4.4, 5.6, 6.6]
            labels = ['很差', '差', '中', '良', '优']
        elif n_classes == 3:
            probit_thresholds = [4.0, 6.0]
            labels = ['差', '中', '优']
        else:
            probit_thresholds = np.linspace(3, 7, n_classes + 1)[1:-1]
            labels = [f'第{i+1}档' for i in range(n_classes)]
        
        # 根据回归方程计算 RSR 临界值
        slope = self.regression_params['slope']
        intercept = self.regression_params['intercept']
        
        rsr_thresholds = [(t - intercept) / slope for t in probit_thresholds]
        
        # 分类
        self.classes = np.zeros(self.m, dtype=int)
        for i in range(self.m):
            for k, threshold in enumerate(rsr_thresholds):
                if self.rsr[i] < threshold:
                    self.classes[i] = k
                    break
            else:
                self.classes[i] = n_classes - 1
        
        self.class_labels = labels
        self.rsr_thresholds = rsr_thresholds
        self.probit_thresholds = probit_thresholds
        
        return self.classes
    
    def summary(self):
        """输出完整结果"""
        self.calc_rsr()
        self.classify()
        
        print("=" * 70)
        print("秩和比法（RSR）综合评价结果")
        print("=" * 70)
        
        # 排序输出
        ranking = np.argsort(-self.rsr)
        print(f"\n{'排名':<6}{'对象':<8}{'RSR':<10}{'概率单位':<10}{'等级':<8}")
        print("-" * 50)
        
        for rank, idx in enumerate(ranking):
            probit_val = stats.norm.cdf(self.rsr[idx] * self.regression_params['slope'] 
                                        + self.regression_params['intercept'] - 5) * 100
            label = self.class_labels[self.classes[idx]]
            print(f"{rank+1:<6}{idx+1:<8}{self.rsr[idx]:<10.4f}"
                  f"{self.probit[rank]:<10.4f}{label:<8}")
        
        print(f"\n回归方程: Probit = {self.regression_params['intercept']:.4f} "
              f"+ {self.regression_params['slope']:.4f} * RSR")
        print(f"R² = {self.regression_params['r_squared']:.4f}")
        print(f"P  = {self.regression_params['p_value']:.4f}")
        
        print(f"\n分档临界值 (RSR):")
        for k, (t, l) in enumerate(zip(self.rsr_thresholds, self.class_labels)):
            print(f"  {l}: RSR < {t:.4f}")
        
        return self.rsr, self.classes

# 示例
data = [
    [85, 90, 78, 92, 88],
    [78, 85, 88, 80, 82],
    [92, 75, 82, 88, 90],
    [88, 82, 90, 85, 78],
    [75, 92, 85, 90, 85]
]

rsr_model = RankSumRatio(data)
rsr, classes = rsr_model.summary()
```

## 五、实际案例：医院服务质量评价
```python
import numpy as np
from scipy import stats

def hospital_rsr_evaluation():
    """医院服务质量 RSR 评价"""
    
    hospitals = ['医院A', '医院B', '医院C', '医院D', '医院E',
                 '医院F', '医院G', '医院H']
    
    # 指标: 床位周转率, 入院诊断符合率, 治愈率, 病死率, 平均住院日
    # 病死率和平均住院日为成本型
    indicators = ['床位周转率', '诊断符合率', '治愈率', '病死率', '平均住院日']
    cost_idx = [3, 4]
    
    data = np.array([
        [25, 96, 88, 1.2, 8.5],
        [22, 94, 85, 1.5, 9.2],
        [28, 97, 90, 0.8, 7.8],
        [20, 92, 82, 2.0, 10.5],
        [26, 95, 87, 1.0, 8.0],
        [23, 93, 84, 1.8, 9.8],
        [30, 98, 92, 0.6, 7.2],
        [24, 94, 86, 1.3, 8.8]
    ])
    
    m, n = data.shape
    
    # 正向化
    data_pos = data.copy()
    for j in cost_idx:
        data_pos[:, j] = data[:, j].max() - data[:, j]
    
    # 计算秩次
    ranks = np.zeros_like(data_pos, dtype=float)
    for j in range(n):
        sorted_idx = np.argsort(data_pos[:, j])
        for rank_pos, idx in enumerate(sorted_idx):
            ranks[idx, j] = rank_pos + 1
    
    # RSR
    rsr = ranks.mean(axis=1)
    
    # 排序
    ranking = np.argsort(-rsr)
    
    # 频率分布
    rsr_sorted = rsr[ranking]
    cum_freq = np.arange(1, m + 1) / m
    probit = stats.norm.ppf(cum_freq) + 5
    
    # 回归
    slope, intercept, r_val, p_val, _ = stats.linregress(rsr_sorted, probit)
    
    # 分档 (5档)
    thresholds_p = [3.4, 4.4, 5.6, 6.6]
    thresholds_r = [(t - intercept) / slope for t in thresholds_p]
    labels = ['很差', '差', '中', '良', '优']
    
    # 分类
    classes = []
    for r in rsr[ranking]:
        for k, t in enumerate(thresholds_r):
            if r < t:
                classes.append(labels[k])
                break
        else:
            classes.append(labels[-1])
    
    print("医院服务质量 RSR 评价")
    print("=" * 65)
    print(f"{'排名':<6}{'医院':<8}{'RSR':<10}{'Probit':<10}{'等级':<8}")
    print("-" * 65)
    
    for k, idx in enumerate(ranking):
        print(f"{k+1:<6}{hospitals[idx]:<8}{rsr[idx]:<10.4f}"
              f"{probit[k]:<10.4f}{classes[k]:<8}")
    
    print(f"\n回归方程: Probit = {intercept:.4f} + {slope:.4f} * RSR")
    print(f"R² = {r_val**2:.4f}")

hospital_rsr_evaluation()
```

## 七、注意事项
1. **指标类型一致性**：计算前需将所有指标统一为效益型
2. **相同值处理**：相同值取平均秩次，避免偏差
3. **指标数量要求**：建议指标数 $n \geq 3$，对象数 $m \geq 5$
4. **权重确定**：等权 RSR 适用于指标重要性相近的情况
5. **分档数选择**：通常分为 3-5 档，取决于对象数量和分布特点

## 八、关键要点总结
| 要点 | 内容 |
|:--|:--|
| 核心思想 | 基于秩次的综合评价 |
| RSR 公式 | $RSR = \frac{1}{mn} \sum R_{ij}$ |
| 秩次确定 | 升序排列，相同值取平均 |
| 正向化 | 成本型指标取反 |
| 分档依据 | 概率单位（Probit）回归 |
| 统计检验 | 回归方程的显著性 |
| 优势 | 非参数方法，分布无关 |
| 局限 | 信息损失，无法反映原始数据差异 |

秩和比法是一种简单而有效的非参数综合评价方法，在医学评价、教育评价等领域应用广泛。其核心优势在于不需要对数据分布做出假设，通过秩次变换将多指标问题简化为单指标排序问题，便于实际操作和结果解释。
