# 模糊综合评价法

> **理论基础** → 参见 [[../../统计学分析方法/模糊数学基础|模糊数学]]

模糊综合评价基于模糊数学的隶属度理论，将定性评价转化为定量评价，处理模糊性和不确定性。

## 四、Python 实现
### 4.1 一级模糊综合评价

```python
import numpy as np

def fuzzy_comprehensive_evaluation(R, w, operator='weighted'):
    """
    模糊综合评价
    :param R: 模糊评价矩阵 (n x m)
    :param w: 因素权重向量 (长度为 n)
    :param operator: 合成算子类型
    :return: 综合评价向量 B
    """
    w = np.array(w, dtype=float)
    R = np.array(R, dtype=float)
    n, m = R.shape
    
    if operator == 'max_min':
        # M(∧, ∨) 取大取小算子
        B = np.zeros(m)
        for j in range(m):
            B[j] = np.max(np.minimum(w, R[:, j]))
    
    elif operator == 'product_max':
        # M(·, ∨) 乘积取大算子
        B = np.zeros(m)
        for j in range(m):
            B[j] = np.max(w * R[:, j])
    
    elif operator == 'min_sum':
        # M(∧, ⊕) 取小求和算子
        B = np.zeros(m)
        for j in range(m):
            B[j] = min(1, np.sum(np.minimum(w, R[:, j])))
    
    elif operator == 'weighted':
        # M(·, ⊕) 加权平均型（最常用）
        B = w @ R
        B = np.minimum(B, 1.0)
    
    # 归一化
    B = B / B.sum()
    return B

def determine_membership(values, grades):
    """
    根据指标值和评价等级确定隶属度
    :param values: 指标值列表
    :param grades: 评价等级的边界值列表
    :return: 隶属度矩阵 R
    """
    n = len(values)
    m = len(grades) - 1  # 评价等级数
    R = np.zeros((n, m))
    
    for i, v in enumerate(values):
        for j in range(m):
            low = grades[j]
            high = grades[j + 1]
            mid = (low + high) / 2
            
            if j == 0 and v <= low:
                R[i, j] = 1.0
            elif j == m - 1 and v >= high:
                R[i, j] = 1.0
            elif low < v <= mid:
                R[i, j] = (v - low) / (high - low) if high != low else 1.0
            elif mid < v < high:
                R[i, j] = (high - v) / (high - low) if high != low else 1.0
            else:
                R[i, j] = 0.0
    
    return R

# 示例：水质评价
print("水质模糊综合评价示例")
print("=" * 50)

# 因素集：pH, 溶解氧, COD, 氨氮

# 各指标的监测值

# 各指标在不同等级的隶属度矩阵（由专业标准确定）
R = [
    [0.6, 0.4, 0.0, 0.0, 0.0],  # pH
```

### 4.2 二级模糊综合评价

```python
import numpy as np

def two_level_fuzzy_evaluation():
    """二级模糊综合评价"""
    
    # ===== 因子层 =====
    # 子因素集 U1: 语文成绩, 数学成绩, 英语成绩
    # 子因素集 U2: 社团活动, 志愿服务, 领导力
    # 子因素集 U3: 体测成绩, 健康状况
    
    # ===== 评价集 =====
    grades = ['优秀', '良好', '中等', '较差']
    
    # ===== 一级评价 =====
    # U1 的隶属度矩阵
    R1 = np.array([
        [0.4, 0.3, 0.2, 0.1],
        [0.3, 0.4, 0.2, 0.1],
        [0.2, 0.3, 0.4, 0.1],
    ])
    w1 = np.array([0.4, 0.35, 0.25])
    B1 = w1 @ R1
    B1 = B1 / B1.sum()
    
    # U2 的隶属度矩阵
    R2 = np.array([
        [0.5, 0.3, 0.1, 0.1],
        [0.3, 0.4, 0.2, 0.1],
        [0.2, 0.5, 0.2, 0.1],
    ])
    w2 = np.array([0.3, 0.35, 0.35])
    B2 = w2 @ R2
    B2 = B2 / B2.sum()
    
    # U3 的隶属度矩阵
    R3 = np.array([
        [0.6, 0.3, 0.1, 0.0],
        [0.4, 0.4, 0.2, 0.0],
    ])
    w3 = np.array([0.5, 0.5])
    B3 = w3 @ R3
    B3 = B3 / B3.sum()
    
    # ===== 二级评价 =====
    # 构造二级评价矩阵
    R_level2 = np.vstack([B1, B2, B3])
    
    # 一级因素权重
    W_level2 = np.array([0.5, 0.3, 0.2])
    
    # 二级综合评价
    B_final = W_level2 @ R_level2
    B_final = B_final / B_final.sum()
    
    print("二级模糊综合评价结果")
    print("=" * 50)
    print(f"学习成绩评价: {np.round(B1, 4)}")
    print(f"综合素质评价: {np.round(B2, 4)}")
    print(f"身体健康评价: {np.round(B3, 4)}")
    print(f"综合评价结果: {np.round(B_final, 4)}")
    print(f"最终等级: {grades[np.argmax(B_final)]}")
    
    return B_final

two_level_fuzzy_evaluation()
```

## 六、实际案例：教学质量评价
```python
import numpy as np

def teaching_quality_evaluation():
    """教学质量模糊综合评价"""
    
    # 因素集
    categories = {
        '教学内容': {
            '因素': ['内容充实性', '知识前沿性', '难度适当性'],
            '权重': [0.4, 0.3, 0.3],
            '隶属度矩阵': [
                [0.5, 0.3, 0.15, 0.05],
                [0.3, 0.4, 0.2, 0.1],
                [0.4, 0.35, 0.2, 0.05],
            ]
        },
        '教学方法': {
            '因素': ['讲解清晰度', '互动性', '启发性'],
            '权重': [0.35, 0.35, 0.3],
            '隶属度矩阵': [
                [0.6, 0.3, 0.1, 0.0],
                [0.4, 0.35, 0.2, 0.05],
                [0.3, 0.4, 0.25, 0.05],
            ]
        },
        '教学态度': {
            '因素': ['认真程度', '耐心程度', '责任心'],
            '权重': [0.3, 0.3, 0.4],
            '隶属度矩阵': [
                [0.7, 0.2, 0.1, 0.0],
                [0.5, 0.35, 0.15, 0.0],
                [0.6, 0.3, 0.1, 0.0],
            ]
        },
        '教学效果': {
            '因素': ['知识掌握', '能力提升', '兴趣激发'],
            '权重': [0.35, 0.35, 0.3],
            '隶属度矩阵': [
                [0.4, 0.35, 0.2, 0.05],
                [0.3, 0.4, 0.25, 0.05],
                [0.5, 0.3, 0.15, 0.05],
            ]
        }
    }
    
    grades = ['优秀', '良好', '中等', '较差']
    
    # 一级评价
    B_level1 = {}
    for cat_name, cat_data in categories.items():
        R = np.array(cat_data['隶属度矩阵'])
        w = np.array(cat_data['权重'])
        B = w @ R
        B = B / B.sum()
        B_level1[cat_name] = B
        print(f"{cat_name} 一级评价: {np.round(B, 4)}")
    
    # 二级评价
    R2 = np.array([B_level1[cat] for cat in categories])
    W2 = np.array([0.3, 0.25, 0.2, 0.25])  # 类别权重
    
    B_final = W2 @ R2
    B_final = B_final / B_final.sum()
    
    print(f"\n综合评价结果: {np.round(B_final, 4)}")
    print(f"最终等级: {grades[np.argmax(B_final)]}")
    
    # 预定度排序
    print("\n各等级隶属度排序:")
    sorted_idx = np.argsort(-B_final)
    for idx in sorted_idx:
        print(f"  {grades[idx]}: {B_final[idx]:.4f}")

teaching_quality_evaluation()
```

## 八、注意事项
1. **隶属函数选择**：不同的隶属函数会导致不同的评价结果，需要结合实际情况选择
2. **评价等级划分**：等级数量和划分方式直接影响评价精度
3. **权重确定**：权重对结果影响显著，建议采用多种方法综合确定
4. **数据标准化**：不同量纲的指标需要先进行标准化处理
5. **算子选择**：加权平均型算子适用于大多数场景，取大取小算子适合突出主要因素

## 九、关键要点总结
| 要点 | 内容 |
|:--|:--|
| 理论基础 | 模糊集合论，隶属函数 |
| 核心概念 | 隶属度 $r_{ij} \in [0,1]$ |
| 评价矩阵 | $R_{n \times m}$，每行和为 1 |
| 合成算子 | 加权平均型最常用 |
| 多级评价 | 先评低层，再评高层 |
| 结果判定 | 最大隶属度原则 |
| 优势 | 处理模糊性和不确定性 |
| 局限 | 隶属函数确定主观性较强 |

模糊综合评价法特别适合处理边界模糊、难以精确量化的评价问题。在实际应用中，应结合具体问题特点，合理确定隶属函数和权重，选择合适的合成算子和结果判定方法。
