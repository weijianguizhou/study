# 灰色预测模型

> **理论基础** → 参见 [[../../统计学分析方法/灰色系统理论|灰色系统]]

灰色预测基于少量、不完全信息建立灰色模型(GM)，通过累加生成弱化随机性，适合小样本预测。

## 五、Python 实现
```python
import numpy as np
import matplotlib.pyplot as plt

class GM11:
    """GM(1,1) 灰色预测模型"""
    
    def __init__(self, data):
        """
        参数:
            data: 原始数据序列（列表或numpy数组）
        """
        self.raw_data = np.array(data, dtype=float)
        self.n = len(data)
        self.a = None  # 发展系数
        self.b = None  # 灰色作用量
        
    def _ago(self):
        """一次累加生成"""
        return np.cumsum(self.raw_data)
    
    def _mean_sequence(self, x1):
        """紧邻均值生成"""
        z = 0.5 * (x1[1:] + x1[:-1])
        return z
    
    def fit(self):
        """拟合模型，估计参数"""
        x1 = self._ago()
        z = self._mean_sequence(x1)
        
        # 构造矩阵B和向量Y
        B = np.column_stack([-z, np.ones(self.n - 1)])
        Y = self.raw_data[1:]
        
        # 最小二乘求解
        self.a, self.b = np.linalg.inv(B.T @ B) @ B.T @ Y
        return self
    
    def predict(self, steps=1):
        """
        预测未来值
        参数:
            steps: 预测步数
        返回:
            预测序列（包含拟合值）
        """
        x1 = self._ago()
        result = []
        
        for k in range(self.n - 1 + steps):
            val = (self.raw_data[0] - self.b / self.a) * np.exp(-self.a * k) + self.b / self.a
            result.append(val)
        
        # 累减还原
        predict_x0 = [result[0]]
        for i in range(1, len(result)):
            predict_x0.append(result[i] - result[i-1])
        
        return np.array(predict_x0)
    
    def accuracy(self):
        """计算模型精度"""
        predict = self.predict(steps=0)
        residuals = self.raw_data - predict
        relative_errors = np.abs(residuals) / self.raw_data * 100
        return {
            'mean_absolute_error': np.mean(np.abs(residuals)),
            'mean_relative_error': np.mean(relative_errors[1:]),
            'max_relative_error': np.max(relative_errors[1:]),
            'relative_errors': relative_errors
        }
    
    def summary(self):
        """输出模型摘要"""
        print("=" * 50)
        print("GM(1,1) 灰色预测模型")
        print("=" * 50)
        print(f"发展系数 a = {self.a:.6f}")
        print(f"灰色作用量 b = {self.b:.6f}")
        acc = self.accuracy()
        print(f"平均相对误差: {acc['mean_relative_error']:.2f}%")
        print(f"最大相对误差: {acc['max_relative_error']:.2f}%")
        print("=" * 50)

# ============ 示例1：GDP预测 ============
# 某地区GDP数据（亿元）
gdp_data = [1043.5, 1198.3, 1368.6, 1593.6, 1847.5, 2150.7, 2500.1]

# ============ 示例2：新陈代谢模型 ============
class MetabolismGM11:
    """新陈代谢GM(1,1)模型"""
    
    def __init__(self, data, window=5):
        """
        参数:
            data: 完整数据序列
            window: 滑动窗口大小
        """
        self.data = np.array(data, dtype=float)
        self.window = window
        self.predictions = []
    
    def predict(self):
        """逐步预测"""
        n = len(self.data)
        for i in range(self.window, n):
            window_data = self.data[i-self.window:i]
            gm = GM11(window_data)
            gm.fit()
            pred = gm.predict(steps=1)[-1]
            self.predictions.append(pred)
        return self.predictions
    
    def full_predict(self, steps=1):
        """预测未来值"""
        # 使用最后window个数据建立模型
        window_data = self.data[-self.window:]
        gm = GM11(window_data)
        gm.fit()
        return gm.predict(steps=steps)

# 使用示例
extended_data = [1043.5, 1198.3, 1368.6, 1593.6, 1847.5, 2150.7, 2500.1, 2863.2, 3210.5]
met_model = MetabolismGM11(extended_data, window=5)
met_model.predict()
print(f"\n新陈代谢模型预测: {met_model.full_predict(steps=2)}")
```

## 六、实际案例：某市人口预测
### 6.1 问题描述

某市近6年的人口数据如下（单位：万人），预测未来3年的人口。

| 年份 | 2015 | 2016 | 2017 | 2018 | 2019 | 2020 |
|------|------|------|------|------|------|------|
| 人口 | 523.7 | 531.2 | 538.9 | 547.1 | 555.8 | 564.3 |

### 6.2 建模步骤

```python
import numpy as np

# 原始数据
years = [2015, 2016, 2017, 2018, 2019, 2020]
population = [523.7, 531.2, 538.9, 547.1, 555.8, 564.3]

# Step 1: 建立GM(1,1)模型
model = GM11(population)
model.fit()

# Step 2: 模型检验
acc = model.accuracy()
print(f"平均相对误差: {acc['mean_relative_error']:.2f}%")

# Step 3: 预测
predict_all = model.predict(steps=3)
print("\n人口预测结果:")
for i, p in enumerate(predict_all):
    year = 2015 + i
    if i < len(population):
        print(f"{year}: 实际={population[i]}, 拟合={p:.2f}, "
              f"误差={abs(population[i]-p)/population[i]*100:.2f}%")
    else:
        print(f"{year}: 预测={p:.2f}")

# Step 4: 输出模型参数
print(f"\n模型方程: dx/dt + ({model.a:.4f})x = {model.b:.4f}")
print(f"时间响应函数: x̂(k+1) = {(population[0] - model.b/model.a):.2f} * "
      f"exp({-model.a:.4f}*k) + {model.b/model.a:.2f}")
```

## 八、关键要点总结
1. **灰色预测的核心**：通过累加生成将无规律数据转化为近似指数序列，用微分方程拟合
2. **GM(1,1)是最基本的灰色预测模型**：一阶微分方程、一个变量
3. **发展系数 $a$ 决定模型适用性**：$|a| \leq 0.3$ 时适合中长期预测
4. **模型检验三步走**：残差检验、关联度检验、后验差检验
5. **新陈代谢模型**可以有效提高预测精度
6. **灰色预测的优势**：数据需求少、计算简单、不需要概率分布假设
7. **灰色预测的局限**：适合单调序列，对波动大的数据效果差
8. **改进方向**：初始值优化、背景值优化、残差修正、与其他方法组合
