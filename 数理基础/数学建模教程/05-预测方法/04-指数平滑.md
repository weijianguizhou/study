# 指数平滑预测模型

> **理论基础** → 参见 [[../../时间序列分析/指数平滑与组合预测|指数平滑]]

指数平滑通过加权平均历史数据预测未来值，近期的观测值被赋予更高权重，实现简单高效。

## 七、Python 实现
```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.optimize import minimize

class SimpleExponentialSmoothing:
    """一次指数平滑"""
    
    def __init__(self, data, alpha=0.3):
        self.data = np.array(data, dtype=float)
        self.alpha = alpha
        self.smoothed = None
        
    def fit(self, init_value=None):
        """拟合模型"""
        n = len(self.data)
        self.smoothed = np.zeros(n)
        
        if init_value is None:
            self.smoothed[0] = self.data[0]
        else:
            self.smoothed[0] = init_value
        
        for t in range(1, n):
            self.smoothed[t] = self.alpha * self.data[t] + (1 - self.alpha) * self.smoothed[t-1]
        
        return self
    
    def predict(self, steps=1):
        """预测未来"""
        return np.full(steps, self.smoothed[-1])
    
    def optimize_alpha(self):
        """优化alpha参数"""
        def objective(alpha):
            model = SimpleExponentialSmoothing(self.data, alpha[0])
            model.fit()
            predictions = model.smoothed[:-1]
            actuals = self.data[1:]
            return np.mean((actuals - predictions) ** 2)
        
        result = minimize(objective, [0.3], bounds=[(0.01, 0.99)])
        self.alpha = result.x[0]
        self.fit()
        return self.alpha

class HoltLinearSmoothing:
    """Holt线性指数平滑"""
    
    def __init__(self, data, alpha=0.3, beta=0.1):
        self.data = np.array(data, dtype=float)
        self.alpha = alpha
        self.beta = beta
        self.level = None
        self.trend = None
        
    def fit(self):
        """拟合模型"""
        n = len(self.data)
        self.level = np.zeros(n)
        self.trend = np.zeros(n)
        
        # 初始值
        self.level[0] = self.data[0]
        self.trend[0] = self.data[1] - self.data[0] if n > 1 else 0
        
        for t in range(1, n):
            self.level[t] = self.alpha * self.data[t] + (1 - self.alpha) * (self.level[t-1] + self.trend[t-1])
            self.trend[t] = self.beta * (self.level[t] - self.level[t-1]) + (1 - self.beta) * self.trend[t-1]
        
        return self
    
    def predict(self, steps=1):
        """预测未来"""
        forecasts = np.zeros(steps)
        for m in range(1, steps + 1):
            forecasts[m-1] = self.level[-1] + self.trend[-1] * m
        return forecasts

class HoltWintersSmoothing:
    """Holt-Winters指数平滑（加法和乘法模型）"""
    
    def __init__(self, data, seasonal_period, model_type='additive',
                 alpha=0.3, beta=0.1, gamma=0.1):
        self.data = np.array(data, dtype=float)
        self.s = seasonal_period
        self.model_type = model_type  # 'additive' or 'multiplicative'
        self.alpha = alpha
        self.beta = beta
        self.gamma = gamma
        self.level = None
        self.trend = None
        self.seasonal = None
        
    def _initialize(self):
        """初始化参数"""
        n = len(self.data)
        s = self.s
        
        # 初始水平：前s个数据的均值
        self.level = np.zeros(n)
        self.trend = np.zeros(n)
        self.seasonal = np.zeros(n + s)
        
        # 初始趋势
        if self.model_type == 'additive':
            self.trend[0] = (np.mean(self.data[s:2*s]) - np.mean(self.data[:s])) / s
        else:
            self.trend[0] = (np.mean(self.data[s:2*s]) / np.mean(self.data[:s]) - 1) / s
        
        # 初始水平
        self.level[0] = np.mean(self.data[:s]) - (s + 1) / 2 * self.trend[0] \
            if self.model_type == 'additive' \
            else np.mean(self.data[:s]) / ((1 + self.trend[0]) ** ((s + 1) / 2))
        
        # 初始季节因子
        for i in range(s):
            if self.model_type == 'additive':
                self.seasonal[i] = self.data[i] - self.level[0]
            else:
                self.seasonal[i] = self.data[i] / self.level[0] if self.level[0] != 0 else 1
    
    def fit(self):
        """拟合模型"""
        n = len(self.data)
        s = self.s
        self._initialize()
        
        fitted = np.zeros(n)
        
        for t in range(1, n):
            # 水平更新
            if self.model_type == 'additive':
                self.level[t] = self.alpha * (self.data[t] - self.seasonal[t - s]) + \
                               (1 - self.alpha) * (self.level[t-1] + self.trend[t-1])
            else:
                self.level[t] = self.alpha * (self.data[t] / self.seasonal[t - s]) + \
                               (1 - self.alpha) * (self.level[t-1] + self.trend[t-1])
            
            # 趋势更新
            self.trend[t] = self.beta * (self.level[t] - self.level[t-1]) + \
                           (1 - self.beta) * self.trend[t-1]
            
            # 季节更新
            if self.model_type == 'additive':
                self.seasonal[t] = self.gamma * (self.data[t] - self.level[t]) + \
                                  (1 - self.gamma) * self.seasonal[t - s]
            else:
                self.seasonal[t] = self.gamma * (self.data[t] / self.level[t]) + \
                                  (1 - self.gamma) * self.seasonal[t - s]
            
            # 拟合值
            if self.model_type == 'additive':
                fitted[t] = self.level[t-1] + self.trend[t-1] + self.seasonal[t - s]
            else:
                fitted[t] = (self.level[t-1] + self.trend[t-1]) * self.seasonal[t - s]
        
        self.fitted = fitted
        return self
    
    def predict(self, steps=1):
        """预测未来"""
        forecasts = np.zeros(steps)
        n = len(self.data)
        s = self.s
        
        for m in range(1, steps + 1):
            if self.model_type == 'additive':
                forecasts[m-1] = self.level[-1] + self.trend[-1] * m + self.seasonal[n - s + (m - 1) % s]
            else:
                forecasts[m-1] = (self.level[-1] + self.trend[-1] * m) * self.seasonal[n - s + (m - 1) % s]
        
        return forecasts

# ============ 示例1：销售额预测（一次指数平滑） ============
np.random.seed(42)
sales = [120, 130, 115, 125, 140, 135, 150, 145, 160, 155, 170, 165]

ses = SimpleExponentialSmoothing(sales, alpha=0.3)
ses.fit()

print("=== 一次指数平滑 ===")
print(f"alpha = {ses.alpha}")
pred = ses.predict(steps=3)
print(f"未来3期预测: {pred}")

plt.figure(figsize=(10, 5))
plt.plot(range(1, len(sales)+1), sales, 'bo-', label='实际值')
plt.plot(range(1, len(sales)+1), ses.smoothed, 'r--', label='平滑值')
plt.plot(range(len(sales)+1, len(sales)+4), pred, 'g*--', label='预测值')
plt.legend()
plt.title('一次指数平滑')
plt.grid(True, alpha=0.3)
plt.show()

# ============ 示例2：带趋势数据（Holt线性平滑） ============
np.random.seed(42)
trend_data = 100 + 5 * np.arange(20) + np.random.normal(0, 3, 20)

holt = HoltLinearSmoothing(trend_data, alpha=0.5, beta=0.2)
holt.fit()

print("\n=== Holt线性指数平滑 ===")
pred_holt = holt.predict(steps=5)
print(f"未来5期预测: {pred_holt}")

plt.figure(figsize=(10, 5))
plt.plot(range(1, len(trend_data)+1), trend_data, 'bo-', label='实际值')
plt.plot(range(1, len(trend_data)+1), 
         [holt.level[i] + holt.trend[i] for i in range(len(trend_data))], 
         'r--', label='拟合值')
plt.plot(range(len(trend_data)+1, len(trend_data)+6), pred_holt, 'g*--', label='预测值')
plt.legend()

plt.grid(True, alpha=0.3)
plt.show()

# ============ 示例3：季节性数据（Holt-Winters） ============
np.random.seed(42)
```

## 八、实际案例：某商场月度销售额预测
### 8.1 问题描述

某商场近3年的月度销售额（万元）如下，预测未来6个月的销售额。

### 8.2 完整建模流程

```python
import numpy as np
import pandas as pd

# 模拟数据
np.random.seed(123)
months_index = pd.date_range('2021-01', '2023-12', freq='MS')
n = len(months_index)

# 趋势 + 季节 + 噪声
base = 500
trend = 10 * np.arange(n)
seasonal = 80 * np.sin(2 * np.pi * np.arange(n) / 12) + \
           20 * np.cos(2 * np.pi * np.arange(n) / 6)
noise = np.random.normal(0, 15, n)
sales = base + trend + seasonal + noise

df = pd.DataFrame({'date': months_index, 'sales': sales})
df.set_index('date', inplace=True)

print("=== 数据概览 ===")
print(df.describe())

# Step 1: 选择模型类型
# 通过观察时序图判断趋势和季节性
fig, axes = plt.subplots(2, 1, figsize=(12, 8))
axes[0].plot(df['sales'])
axes[0].set_title('月度销售额时序图')

# 自相关图
from statsmodels.graphics.tsaplots import plot_acf
plot_acf(df['sales'], ax=axes[1], lags=36)
plt.tight_layout()
plt.show()

# Step 2: 训练集/测试集分割
train = df['sales'][:-6]
test = df['sales'][-6:]

# Step 3: 拟合Holt-Winters模型
hw_model = HoltWintersSmoothing(train.values, seasonal_period=12,
                                model_type='additive',
                                alpha=0.3, beta=0.1, gamma=0.2)
hw_model.fit()

# Step 4: 预测
pred = hw_model.predict(steps=6)

# Step 5: 评估
mae = np.mean(np.abs(test.values - pred))
rmse = np.sqrt(np.mean((test.values - pred) ** 2))
mape = np.mean(np.abs((test.values - pred) / test.values)) * 100

print(f"\n=== 预测精度 ===")
print(f"MAE: {mae:.2f} 万元")
print(f"RMSE: {rmse:.2f} 万元")
print(f"MAPE: {mape:.2f}%")

# Step 6: 全模型预测
full_model = HoltWintersSmoothing(df['sales'].values, seasonal_period=12,
                                  model_type='additive',
                                  alpha=0.3, beta=0.1, gamma=0.2)
full_model.fit()
future_pred = full_model.predict(steps=6)

print(f"\n=== 未来6个月预测 ===")
future_dates = pd.date_range('2024-01', '2024-06', freq='MS')
for date, val in zip(future_dates, future_pred):
    print(f"{date.strftime('%Y-%m')}: {val:.2f} 万元")
```

## 九、关键要点总结
1. **一次指数平滑**适合无趋势无季节性的数据
2. **Holt线性平滑**适合有线性趋势的数据
3. **Holt-Winters**适合有趋势和季节性的数据
4. **平滑参数** $\alpha, \beta, \gamma$ 控制对新数据的响应速度
5. **加法模型**适合季节波动幅度恒定的序列
6. **乘法模型**适合季节波动随水平变化的序列
7. **参数优化**通常使用最小化MSE的方法
8. **指数平滑的优点**：计算简单、存储需求小、可在线更新
9. **指数平滑的局限**：假设数据有稳定的模式，对突变响应滞后
10. **与ARIMA的关系**：指数平滑是ARIMA的特例
