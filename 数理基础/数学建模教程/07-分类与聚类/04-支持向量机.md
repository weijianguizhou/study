# 支持向量机（SVM）

> **理论基础** → 参见 [[../../机器学习数学基础/支持向量机|SVM理论]]

支持向量机(SVM)通过最大化分类间隔寻找最优超平面，利用核技巧处理非线性分类问题。

## 8. Python 实现
### 8.1 线性 SVM

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.svm import SVC
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report, accuracy_score
from sklearn.preprocessing import StandardScaler

X, y = make_classification(n_samples=300, n_features=2, n_redundant=0,
                           n_informative=2, random_state=42, n_clusters_per_class=1)
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

scaler = StandardScaler()
X_train_scaled = scaler.fit_transform(X_train)
X_test_scaled = scaler.transform(X_test)

svm_linear = SVC(kernel='linear', C=1.0)
svm_linear.fit(X_train_scaled, y_train)

y_pred = svm_linear.predict(X_test_scaled)
print(f"线性 SVM 准确率: {accuracy_score(y_test, y_pred):.4f}")
print(f"支持向量数: {len(svm_linear.support_vectors_)}")
print(classification_report(y_test, y_pred))
```

### 8.2 决策边界可视化

```python
def plot_svm_decision_boundary(model, X, y, title, scaler=None):
    fig, ax = plt.subplots(figsize=(8, 6))

    xx, yy = np.meshgrid(
        np.linspace(X[:, 0].min() - 1, X[:, 0].max() + 1, 500),
        np.linspace(X[:, 1].min() - 1, X[:, 1].max() + 1, 500)
    )
    grid = np.c_[xx.ravel(), yy.ravel()]
    if scaler:
        grid = scaler.transform(grid)
    Z = model.decision_function(grid).reshape(xx.shape)

    ax.contourf(xx, yy, Z, levels=[-1, 0, 1], alpha=0.3, cmap='RdBu')
    ax.contour(xx, yy, Z, levels=[-1, 0, 1], linestyles=['--', '-', '--'], colors='black')

    sv = model.support_vectors_
    ax.scatter(sv[:, 0], sv[:, 1], s=200, facecolors='none', edgecolors='red', linewidths=2, label='支持向量')

    mask0 = y == 0
    mask1 = y == 1
    ax.scatter(X[mask0, 0], X[mask0, 1], c='blue', s=30, alpha=0.6, label='类别 0')
    ax.scatter(X[mask1, 0], X[mask1, 1], c='red', s=30, alpha=0.6, label='类别 1')

    ax.set_title(title)
    ax.legend()
    plt.tight_layout()
    plt.show()

plot_svm_decision_boundary(svm_linear, X_train_scaled, y_train, '线性 SVM')
```

### 8.3 核函数对比

```python
kernels = ['linear', 'poly', 'rbf', 'sigmoid']
fig, axes = plt.subplots(2, 2, figsize=(14, 12))

for ax, kernel in zip(axes.flat, kernels):
    svm = SVC(kernel=kernel, C=1.0, random_state=42)
    svm.fit(X_train_scaled, y_train)
    acc = accuracy_score(y_test, svm.predict(X_test_scaled))

    xx, yy = np.meshgrid(
        np.linspace(X_train_scaled[:, 0].min() - 1, X_train_scaled[:, 0].max() + 1, 300),
        np.linspace(X_train_scaled[:, 1].min() - 1, X_train_scaled[:, 1].max() + 1, 300)
    )
    grid = np.c_[xx.ravel(), yy.ravel()]
    Z = svm.decision_function(grid).reshape(xx.shape)

    ax.contourf(xx, yy, Z, levels=[-1, 0, 1], alpha=0.3, cmap='RdBu')
    ax.contour(xx, yy, Z, levels=[-1, 0, 1], linestyles=['--', '-', '--'], colors='black')
    ax.scatter(X_train_scaled[y_train == 0, 0], X_train_scaled[y_train == 0, 1], c='blue', s=20, alpha=0.5)
    ax.scatter(X_train_scaled[y_train == 1, 0], X_train_scaled[y_train == 1, 1], c='red', s=20, alpha=0.5)
    sv = svm.support_vectors_
    ax.scatter(sv[:, 0], sv[:, 1], s=100, facecolors='none', edgecolors='yellow', linewidths=2)
    ax.set_title(f'{kernel} 核 (准确率: {acc:.3f}, SV数: {len(sv)})')

plt.tight_layout()
plt.show()
```

### 8.4 超参数调优

```python
from sklearn.model_selection import GridSearchCV

param_grid = {
    'C': [0.01, 0.1, 1, 10, 100],
    'gamma': ['scale', 'auto', 0.01, 0.1, 1],
    'kernel': ['rbf']
}

grid_search = GridSearchCV(
    SVC(random_state=42),
    param_grid,
    cv=5,
    scoring='accuracy',
    n_jobs=-1,
    verbose=1
)
grid_search.fit(X_train_scaled, y_train)

print(f"最佳参数: {grid_search.best_params_}")
print(f"最佳交叉验证准确率: {grid_search.best_score_:.4f}")

best_svm = grid_search.best_estimator_
y_pred_best = best_svm.predict(X_test_scaled)
print(f"测试集准确率: {accuracy_score(y_test, y_pred_best):.4f}")
```

## 9. 实际案例：文本分类
```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.pipeline import Pipeline

texts = [
    "这部电影非常精彩，值得一看", "剧情很棒，演员演技好",
    "特效震撼，视觉效果极佳", "强烈推荐这部好电影",
    "浪费时间，剧情无聊", "演技差，故事平淡无奇",
    "太难看了，不推荐", "节奏拖沓，毫无亮点"
]
labels = [1, 1, 1, 1, 0, 0, 0, 0]

pipeline = Pipeline([
    ('tfidf', TfidfVectorizer(max_features=1000)),
    ('svm', SVC(kernel='linear', C=1.0))
])
pipeline.fit(texts, labels)

new_texts = ["好看推荐", "很差不值得"]
predictions = pipeline.predict(new_texts)
for text, pred in zip(new_texts, predictions):
    print(f"'{text}' → {'正面' if pred == 1 else '负面'}")
```

## 10. 关键要点总结
| 要点 | 说明 |
|------|------|
| 核心目标 | 最大化间隔 $\frac{2}{\|w\|}$ |
| 优化问题 | 凸二次规划，有全局最优解 |
| 支持向量 | 决策边界由少数支持向量决定 |
| 软间隔 | 通过参数 $C$ 控制误分类惩罚 |
| 核技巧 | 隐式映射到高维空间，避免维度灾难 |
| 常用核 | RBF 核最通用，线性核适合高维稀疏数据 |
| SMO 算法 | 标准求解方法，每次优化两个变量 |
| 合页损失 | $\max(0, 1 - yf(x))$，SVM 的损失函数 |
| 适用场景 | 小样本、高维数据、二分类 |
| 关键参数 | $C$（正则化）、$\gamma$（RBF 核宽度）、核类型 |

SVM 因其优秀的理论基础和泛化能力，在深度学习兴起之前一直是机器学习中最强的分类算法之一。即使在今天，SVM 在小样本和高维数据场景中仍然是非常有竞争力的选择。
