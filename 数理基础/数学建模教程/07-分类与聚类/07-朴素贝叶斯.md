# 朴素贝叶斯算法

> **理论基础** → 参见 [[../../机器学习数学基础/贝叶斯分类器|贝叶斯分类器]]

朴素贝叶斯基于贝叶斯定理和特征条件独立假设进行分类，计算效率高、在小样本中表现良好。

## 8. Python 实现
### 8.1 三种朴素贝叶斯对比

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.naive_bayes import GaussianNB, MultinomialNB, BernoulliNB
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score
from sklearn.preprocessing import MinMaxScaler

iris = load_iris()
X_train, X_test, y_train, y_test = train_test_split(
    iris.data, iris.target, test_size=0.3, random_state=42)

gnb = GaussianNB()
gnb.fit(X_train, y_train)
print(f"Gaussian NB: {accuracy_score(y_test, gnb.predict(X_test)):.4f}")

scaler = MinMaxScaler()
X_train_s = scaler.fit_transform(X_train)
X_test_s = scaler.transform(X_test)

mnb = MultinomialNB(alpha=1.0)
mnb.fit(X_train_s, y_train)
print(f"Multinomial NB: {accuracy_score(y_test, mnb.predict(X_test_s)):.4f}")

bnb = BernoulliNB(alpha=1.0)
X_tr_b = (X_train > X_train.mean(axis=0)).astype(int)
X_te_b = (X_test > X_train.mean(axis=0)).astype(int)
bnb.fit(X_tr_b, y_train)
print(f"Bernoulli NB: {accuracy_score(y_test, bnb.predict(X_te_b)):.4f}")
```

### 8.2 决策边界可视化

```python
from sklearn.datasets import make_classification

X_2d, y_2d = make_classification(n_samples=300, n_features=2, n_redundant=0,
                                  n_informative=2, n_clusters_per_class=1, random_state=42)
X_tr2, X_te2, y_tr2, y_te2 = train_test_split(X_2d, y_2d, test_size=0.3, random_state=42)

fig, axes = plt.subplots(1, 3, figsize=(18, 5))
xx, yy = np.meshgrid(
    np.linspace(X_2d[:, 0].min() - 1, X_2d[:, 0].max() + 1, 300),
    np.linspace(X_2d[:, 1].min() - 1, X_2d[:, 1].max() + 1, 300))
grid = np.c_[xx.ravel(), yy.ravel()]

for ax, (name, model) in zip(axes, [
    ('GaussianNB', GaussianNB()),
    ('MultinomialNB', MultinomialNB(alpha=1.0)),
    ('BernoulliNB', BernoulliNB(alpha=1.0))
]):
    if 'Multinomial' in name:
        sc = MinMaxScaler()
        Xtr_s, Xte_s, grid_s = sc.fit_transform(X_tr2), sc.transform(X_te2), sc.transform(grid)
        model.fit(Xtr_s, y_tr2)
        Z = model.predict(grid_s).reshape(xx.shape)
        acc = accuracy_score(y_te2, model.predict(Xte_s))
    elif 'Bernoulli' in name:
        Xtr_b = (X_tr2 > X_tr2.mean(axis=0)).astype(int)
        Xte_b = (X_te2 > X_tr2.mean(axis=0)).astype(int)
        grid_b = (grid > X_tr2.mean(axis=0)).astype(int)
        model.fit(Xtr_b, y_tr2)
        Z = model.predict(grid_b).reshape(xx.shape)
        acc = accuracy_score(y_te2, model.predict(Xte_b))
    else:
        model.fit(X_tr2, y_tr2)
        Z = model.predict(grid).reshape(xx.shape)
        acc = accuracy_score(y_te2, model.predict(X_te2))

    ax.contourf(xx, yy, Z, alpha=0.3, cmap='RdBu')
    ax.scatter(X_te2[y_te2 == 0, 0], X_te2[y_te2 == 0, 1], c='blue', s=20, alpha=0.6)
    ax.scatter(X_te2[y_te2 == 1, 0], X_te2[y_te2 == 1, 1], c='red', s=20, alpha=0.6)
    ax.set_title(f'{name} (准确率: {acc:.3f})')

plt.tight_layout()
plt.show()
```

### 8.3 Laplace 平滑参数影响

```python
from sklearn.model_selection import cross_val_score

alphas = [0.001, 0.01, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0]
scores = []

X_text = np.random.randint(0, 100, (500, 50))
y_text = np.random.randint(0, 3, 500)
X_tr_t, X_te_t, y_tr_t, y_te_t = train_test_split(X_text, y_text, test_size=0.3, random_state=42)

for alpha in alphas:
    mnb = MultinomialNB(alpha=alpha)
    cv_scores = cross_val_score(mnb, X_tr_t, y_tr_t, cv=5, scoring='accuracy')
    scores.append(cv_scores.mean())

fig, ax = plt.subplots(figsize=(10, 6))
ax.plot(alphas, scores, 'bo-', linewidth=2, markersize=8)
ax.set_xlabel('alpha (Laplace)')
ax.set_ylabel('CV Accuracy')
ax.set_title('Laplace Smoothing Parameter')
ax.set_xscale('log')
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()
```

### 8.5 文本分类实战

```python
from sklearn.feature_extraction.text import CountVectorizer

train_docs = [
    "this movie is great and wonderful",
    "excellent film highly recommend",
    "amazing acting and beautiful story",
    "best movie I have ever seen",
    "terrible film waste of time",
    "bad movie boring and awful",
    "worst experience horrible acting",
    "do not watch this bad film"
]
train_labels = [1, 1, 1, 1, 0, 0, 0, 0]

test_docs = [
    "great wonderful film",
    "terrible bad movie"
]

vectorizer = CountVectorizer()
X_train_doc = vectorizer.fit_transform(train_docs)
X_test_doc = vectorizer.transform(test_docs)

nb_text = MultinomialNB(alpha=1.0)
nb_text.fit(X_train_doc, train_labels)

predictions = nb_text.predict(X_test_doc)
proba = nb_text.predict_proba(X_test_doc)

for doc, pred, p in zip(test_docs, predictions, proba):
    label = "positive" if pred == 1 else "negative"
    print(f"'{doc}' -> {label} (conf: {max(p):.3f})")
```

## 9. 实际案例：垃圾邮件过滤
```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.pipeline import Pipeline

spam_texts = [
    "win free money now click here",
    "congratulations you won lottery",
    "free gift card claim now urgent",
    "make money fast investment opportunity",
    "meeting tomorrow at conference room",
    "project deadline is next friday",
    "please review the attached document",
    "lunch plans for today"
]
spam_labels = [1, 1, 1, 1, 0, 0, 0, 0]

pipeline = Pipeline([
    ('tfidf', TfidfVectorizer(max_features=100)),
    ('nb', MultinomialNB(alpha=1.0))
])
pipeline.fit(spam_texts, spam_labels)

new_texts = ["free money win now", "project meeting tomorrow"]
preds = pipeline.predict(new_texts)
for text, pred in zip(new_texts, preds):
    print(f"'{text}' -> {'spam' if pred == 1 else 'ham'}")
```

## 10. 优缺点总结
### 优点

1. **训练和预测速度极快**：只需要计算频率统计
2. **对小样本有效**：参数少，不容易过拟合
3. **天然支持多分类**
4. **对缺失数据鲁棒**
5. **可解释性强**：概率模型直观
6. **增量学习**：新数据只需更新计数

### 缺点

1. **条件独立假设过强**：特征相关时性能下降
2. **对输入数据形式敏感**：需要选择合适的变体
3. **概率估计不够准确**：尤其是 0-1 附近的概率
4. **无法学习特征间的交互**

## 11. 关键要点总结
| 要点 | 说明 |
|------|------|
| 核心公式 | $\hat{y} = \arg\max_{y_k} P(Y=y_k) \prod_j P(X_j \mid Y=y_k)$ |
| 独立假设 | 条件独立性是"朴素"的来源，简化计算 |
| 三种变体 | 高斯（连续）、多项式（计数）、伯努利（二值） |
| Laplace 平滑 | $\alpha=1$，解决零概率问题 |
| 贝叶斯解释 | 平滑等价于 Dirichlet 先验的 MAP 估计 |
| 对数空间 | 避免连乘下溢 |
| 误差界 | $R_{NB} - R^* \leq \frac{1}{2}\sum_{j\neq l}|\text{Cov}(X_j,X_l \mid Y)|$ |
| 适用场景 | 文本分类、垃圾过滤、高维稀疏数据 |
| 核心优势 | 速度快、简单、对高维数据有效 |

朴素贝叶斯虽然假设过强，但在实际中表现往往出人意料地好，是快速建立基线模型的首选。
