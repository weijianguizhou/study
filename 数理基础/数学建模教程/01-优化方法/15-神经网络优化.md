### 7.1 反向传播手动实现

```python
import numpy as np

def sigmoid(z):
    return 1 / (1 + np.exp(-z))

def sigmoid_derivative(a):
    return a * (1 - a)

class SimpleNN:
    def __init__(self, layers):
        self.weights = []
        self.biases = []
        for i in range(len(layers) - 1):
            w = np.random.randn(layers[i], layers[i+1]) * np.sqrt(2.0 / layers[i])
            b = np.zeros((1, layers[i+1]))
            self.weights.append(w)
            self.biases.append(b)
    
    def forward(self, X):
        self.activations = [X]
        self.z_values = []
        a = X
        for i in range(len(self.weights)):
            z = a @ self.weights[i] + self.biases[i]
            self.z_values.append(z)
            a = sigmoid(z)
            self.activations.append(a)
        return a
    
    def backward(self, y, lr=0.01):
        m = y.shape[0]
        L = len(self.weights)
        delta = (self.activations[-1] - y) * sigmoid_derivative(self.activations[-1])
        
        for l in reversed(range(L)):
            dW = self.activations[l].T @ delta / m
            db = np.mean(delta, axis=0, keepdims=True)
            
            if l > 0:
                delta = (delta @ self.weights[l].T) * sigmoid_derivative(self.activations[l])
            
            self.weights[l] -= lr * dW
            self.biases[l] -= lr * db
    
    def train(self, X, y, epochs=1000, lr=0.1, verbose=True):
        for epoch in range(epochs):
            output = self.forward(X)
            loss = np.mean((output - y) ** 2)
            self.backward(y, lr)
            if verbose and epoch % 200 == 0:
                print(f"Epoch {epoch}, Loss: {loss:.6f}")

# XOR 问题
X = np.array([[0,0],[0,1],[1,0],[1,1]])
y = np.array([[0],[1],[1],[0]])

nn = SimpleNN([2, 8, 1])
nn.train(X, y, epochs=2000, lr=2.0)
print(f"预测: {nn.forward(X).round()}")
```

### 7.2 使用 PyTorch 训练神经网络

```python
import torch
import torch.nn as nn
import torch.optim as optim

class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(2, 16)
        self.fc2 = nn.Linear(16, 1)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()
    
    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.sigmoid(self.fc2(x))
        return x

X = torch.tensor([[0,0],[0,1],[1,0],[1,1]], dtype=torch.float32)
y = torch.tensor([[0],[1],[1],[0]], dtype=torch.float32)

model = Net()
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.01)
scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=500, gamma=0.5)

for epoch in range(2000):
    output = model(X)
    loss = criterion(output, y)
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    scheduler.step()
    if epoch % 500 == 0:
        print(f"Epoch {epoch}, Loss: {loss.item():.6f}")

with torch.no_grad():
    predictions = model(X).round()
    print(f"预测:\n{predictions}")
```

### 7.3 Adam 优化器手动实现

```python
import numpy as np

class Adam:
    def __init__(self, params, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = [np.zeros_like(p) for p in params]
        self.v = [np.zeros_like(p) for p in params]
        self.t = 0
    
    def step(self, params, grads):
        self.t += 1
        for i in range(len(params)):
            self.m[i] = self.beta1 * self.m[i] + (1 - self.beta1) * grads[i]
            self.v[i] = self.beta2 * self.v[i] + (1 - self.beta2) * grads[i]**2
            m_hat = self.m[i] / (1 - self.beta1**self.t)
            v_hat = self.v[i] / (1 - self.beta2**self.t)
            params[i] -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

# 测试
W = np.random.randn(3, 3)
b = np.zeros(3)
params = [W, b]
optimizer = Adam(params, lr=0.001)

for step in range(100):
    grads = [2 * W, np.ones_like(b)]  # 简化的梯度
    optimizer.step(params, grads)
print(f"W 范数: {np.linalg.norm(params[0]):.6f}")
```

## 八、关键要点总结
1. **反向传播**利用链式法则高效计算所有参数的梯度，是神经网络训练的基石
2. **Adam**是目前最常用的优化器，结合了动量和自适应学习率
3. **学习率调度**对训练效果至关重要，余弦退火和预热是常用策略
4. **Xavier/He 初始化**确保前向和反向传播中信号的方差稳定
5. **Batch Normalization** 加速训练，允许使用更大学习率
6. **Dropout** 和 **L2正则化**防止过拟合
7. 梯度裁剪（Gradient Clipping）防止梯度爆炸
8. 现代训练实践：AdamW + 余弦退火 + 预热 + BatchNorm + Dropout
