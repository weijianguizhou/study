\chapter{复积分}

\section{复变函数积分的定义}

设 $C$ 是复平面上一条光滑（或逐段光滑）的曲线，我们选定一个正方向（规定 $+$：起点指向终点或逆时针），称 $C$ 为一条\textbf{有向曲线}，其反向曲线为 $C^-$。

\begin{tcolorbox}[title=定义：复变函数积分, colframe=blue!50!black, colback=blue!5!white, sharp corners]
	设函数 $w=f(z)$ 在区域 $D$ 内有定义，$C$ 为 $D$ 内以 $A$ 为起点、$B$ 为终点的光滑有向曲线。
	若将曲线 $C$ 任意划分为 $n$ 个小弧段，分点为 $A=z_0, z_1, \dots, z_n=B$。在每个小弧段 $\overset{\frown}{z_{k-1}z_k}$ 上任取一点 $\xi_k$，作和式：
	\[
	\sum_{k=1}^n f(\xi_k) \Delta z_k, \quad \text{其中 } \Delta z_k = z_k - z_{k-1}.
	\]
	记 $\delta = \max_{1\leq k\leq n} |\Delta z_k|$ 为小弧段的最大长度。若当 $\delta \to 0$ 时，该和式的极限存在且唯一（与分法、取点 $\xi_k$ 无关），则称此极限为函数 $f(z)$ 在曲线 $C$ 上的\textbf{复变函数积分}，记作
	\[
	\int_C f(z) \, \mathrm{d}z = \lim_{\delta \to 0} \sum_{k=1}^n f(\xi_k) \Delta z_k.
	\]
\end{tcolorbox}

\subsection{积分的计算方法（参数方程法）}
设曲线 $C$ 的参数方程为 $z = z(t) = x(t) + iy(t)$，其中 $t \in [\alpha, \beta]$。
则 $\mathrm{d}z = z'(t) \mathrm{d}t = (x'(t) + iy'(t)) \mathrm{d}t$。

复积分可转化为实变量的定积分计算：
\[
\int_C f(z) \mathrm{d}z = \int_\alpha^\beta f[z(t)] z'(t) \mathrm{d}t.
\]
也可拆分为实部和虚部的曲线积分组合：
\[
\int_C f(z) \mathrm{d}z = \int_C u \mathrm{d}x - v \mathrm{d}y + i \int_C v \mathrm{d}x + u \mathrm{d}y.
\]

\section{典型例题计算}

\begin{tcolorbox}[title=例 1, colframe=orange!75!black, colback=orange!5!white, sharp corners]
	计算 $\displaystyle \int_C z \, \mathrm{d}z$，路径 $C$ 分别为：\\
	(1) 从点 $O$ 沿直线 $z=t+it$ 到点 $1+i$；\\
	(2) 从点 $O$ 沿实轴到 $1$，再平行虚轴到 $1+i$。
	
	\tcbline
	
	\textbf{解 (1)：} 路径 $z = t + it, \quad t: 0 \to 1$。
	$\mathrm{d}z = (1+i)\mathrm{d}t$。
	\[
	\int_C z \, \mathrm{d}z = \int_0^1 (t + it)(1+i) \mathrm{d}t = (1+i)^2 \int_0^1 t \mathrm{d}t = 2i \cdot \frac{1}{2} = i.
	\]
	
	\textbf{解 (2)：} 分为两段 $C_1: z=t, t:0\to 1$；$C_2: z=1+it, t:0\to 1$。
	\[
	\int_{C_1} z \, \mathrm{d}z = \int_0^1 t \, \mathrm{d}t = \frac{1}{2}.
	\]
	\[
	\int_{C_2} z \, \mathrm{d}z = \int_0^1 (1+it)(i \, \mathrm{d}t) = \int_0^1 (-t+i) \mathrm{d}t = -\frac{1}{2} + i.
	\]
	\[
	\int_C z \, \mathrm{d}z = \int_{C_1} z \, \mathrm{d}z + \int_{C_2} z \, \mathrm{d}z = \frac{1}{2} + \left(-\frac{1}{2} + i\right) = i.
	\]
\end{tcolorbox}

\begin{tcolorbox}[title=例 2, colframe=orange!75!black, colback=orange!5!white, sharp corners]
	设 $C$ 为正向圆周 $|z|=1$，计算：\\
	(1) $\displaystyle \oint_C (x^2 - y^2 + 2xyi) \mathrm{d}z$ \qquad
	(2) $\displaystyle \oint_C \frac{1}{z} \mathrm{d}z$
	
	\tcbline
	
	\textbf{解：} 参数化圆周 $z(\theta) = e^{i\theta} = \cos\theta + i\sin\theta$，$\theta: 0 \to 2\pi$，$\mathrm{d}z = i e^{i\theta} \mathrm{d} \theta$。
	
	(1) 观察被积函数可知 $f(z) = (x^2-y^2) + i(2xy) = (x+iy)^2 = z^2 = e^{i2\theta}$。
	\[
	\oint_C z^2 \mathrm{d}z = \int_0^{2\pi} e^{i2\theta} \cdot i e^{i\theta} \mathrm{d}\theta = i \int_0^{2\pi} e^{i3\theta} \mathrm{d}\theta = \left. \frac{1}{3} e^{i3\theta} \right|_0^{2\pi} = 0.
	\]
	
	(2) $\displaystyle \oint_{|z|=1} \frac{1}{z} \mathrm{d}z = \int_0^{2\pi} \frac{1}{e^{i\theta}} \cdot i e^{i\theta} \mathrm{d}\theta = \int_0^{2\pi} i \, \mathrm{d}\theta = 2\pi i$.
\end{tcolorbox}

\section{重要公式与推论}

\subsection{复积分的估值不等式}
对于复变量积分，我们有对应的模长估值定理：
\[
\left| \int_C f(z) \mathrm{d}z \right| \le \int_C |f(z)| \mathrm{d}s.
\]
\begin{tcolorbox}[colframe=teal!60!black, colback=teal!5!white, leftrule=3pt, rightrule=0pt, toprule=0pt, bottomrule=0pt, sharp corners]
	若在曲线 $C$ 上满足 $|f(z)| \le M$，且 $L$ 为曲线 $C$ 的长度，则：
	\[
	\left| \int_C f(z) \mathrm{d}z \right| \le M L.
	\]
\end{tcolorbox}

\subsection{核心公式：$\displaystyle \oint_{|z-z_0|=r} \frac{1}{(z-z_0)^n} \mathrm{d}z$ 的计算}
设 $C$ 为正向圆周 $|z-z_0|=r$。令 $z-z_0 = re^{i\varphi}$，则 $\mathrm{d}z = ire^{i\varphi} \mathrm{d}\varphi$。
\[
\oint_C \frac{1}{(z-z_0)^n} \mathrm{d}z = \int_0^{2\pi} \frac{1}{r^n e^{in\varphi}} \cdot ire^{i\varphi} \mathrm{d}\varphi = \frac{i}{r^{n-1}} \int_0^{2\pi} e^{-i(n-1)\varphi} \mathrm{d}\varphi.
\]
当 $n=1$ 时，积分积项为 $1$，结果为 $2\pi$；当 $n \neq 1$ 时，由于指数函数的周期性，积分结果为 $0$。

\begin{tcolorbox}[colframe=purple!50!black, colback=purple!5!white, sharp corners]
	\textbf{重要结论：}
	\[
	\oint_{|z-z_0|=r} \frac{1}{(z-z_0)^n} \mathrm{d}z =
	\begin{cases}
		2\pi i, & n=1 \\
		0, & n \neq 1 \quad (n \in \mathbb{Z})
	\end{cases}
	\]
\end{tcolorbox}

\section{原函数与不定积分}

\begin{tcolorbox}[title=定理：积分与路径无关性, colframe=green!45!black, colback=green!5!white]
	设函数 $f(z)$ 在单连通区域 $B$ 内解析，则积分 $\displaystyle \int_C f(z) \mathrm{d}z$ 与在 $B$ 内的路径无关。
\end{tcolorbox}

\begin{tcolorbox}[title=定义：原函数, colframe=blue!50!black, colback=blue!5!white, sharp corners]
	若在区域 $B$ 内有 $F'(z) = f(z)$，则称 $F(z)$ 为 $f(z)$ 在该区域内的一个\textbf{原函数}。
\end{tcolorbox}

\subsection{变上限积分的导数}
设 $F(z) = \int_{z_0}^z f(\zeta) \mathrm{d}\zeta$，其中 $z_0$ 为 $B$ 内任意定点。由解析性及积分与路径无关，可证：
\[
F'(z) = \lim_{\Delta z \to 0} \frac{F(z+\Delta z) - F(z)}{\Delta z} = f(z).
\]

\subsection{牛顿-莱布尼茨公式}
若 $G(z)$ 是 $f(z)$ 的一个原函数，则复积分同样可以用前后端点值计算：
\[
\int_{z_0}^{z_1} f(z) \mathrm{d}z = G(z_1) - G(z_0).
\]

\section{柯西-古萨基本定理}

\begin{tcolorbox}[title=定理：柯西-古萨定理 (Cauchy-Goursat Theorem), colframe=green!45!black, colback=green!5!white]
	如果函数 $w=f(z)$ 在单连通区域 $D$ 内解析，$C$ 是 $D$ 内的一条简单闭曲线，则
	\[
	\oint_C f(z) \mathrm{d}z = 0.
	\]
\end{tcolorbox}

\subsection{闭路变形原理}
\begin{tcolorbox}[colframe=teal!60!black, colback=teal!5!white, leftrule=3pt, rightrule=0pt, toprule=0pt, bottomrule=0pt, sharp corners]
	设 $f(z)$ 在区域 $D$ 内解析，$C, C_1, C_2, \dots, C_n$ 是 $D$ 内互不相交的正向简单闭曲线，且 $C_1, \dots, C_n$ 全部包含在 $C$ 所围的区域内，则：
	\[
	\oint_C f(z) \mathrm{d}z = \sum_{k=1}^n \oint_{C_k} f(z) \mathrm{d}z.
	\]
\end{tcolorbox}

\section{复合闭路定理与例题}

\begin{tcolorbox}[title=例 3, colframe=orange!75!black, colback=orange!5!white, sharp corners]
	计算 $\displaystyle \oint_C \frac{3z-2i}{z(z-i)} \mathrm{d}z$，其中 $C$ 为正向圆周 $|z|=2$。
	
	\tcbline
	
	\textbf{解：} 被积函数有两个奇点 $z=0$ 及 $z=i$，均在圆周 $C$ 内部。
	利用部分分式展开法：
	\[
	\frac{3z-2i}{z(z-i)} = \frac{2}{z} + \frac{1}{z-i}.
	\]
	由闭路变形原理，分别围绕两奇点作小圆周积分，结合核心圆周积分结论得：
	\[
	\oint_C \frac{2}{z} \mathrm{d}z = 2 \cdot 2\pi i = 4\pi i, \quad \oint_C \frac{1}{z-i} \mathrm{d}z = 2\pi i.
	\]
	\[
	\text{原式} = 4\pi i + 2\pi i = 6\pi i.
	\]
\end{tcolorbox}

\section{柯西积分公式}

柯西积分公式是解析函数理论的核心结果之一，它揭示了解析函数在区域内部的值可由边界上的积分完全确定。

\begin{tcolorbox}[title=定理：柯西积分公式, colframe=green!45!black, colback=green!5!white]
	设 $f(z)$ 在简单闭曲线 $C$ 所围成的区域 $D$ 内解析，在 $\overline{D} = D \cup C$ 上连续，$z_0$ 为 $D$ 内任意一点，则
	\[
	f(z_0) = \frac{1}{2\pi i} \oint_C \frac{f(z)}{z - z_0} \mathrm{d}z .
	\]
\end{tcolorbox}

\noindent\textbf{证明思路：} 在 $D$ 内作以 $z_0$ 为心、充分小的正向圆周 $C_\varepsilon$，使 $C_\varepsilon$ 完全含于 $D$ 内。由闭路变形原理，$\oint_C \frac{f(z)}{z-z_0}\mathrm{d}z = \oint_{C_\varepsilon} \frac{f(z)}{z-z_0}\mathrm{d}z$。令 $z - z_0 = \varepsilon e^{i\theta}$，利用 $f(z)$ 的连续性可得
\[
\oint_{C_\varepsilon} \frac{f(z)}{z-z_0}\mathrm{d}z = \int_0^{2\pi} \frac{f(z_0 + \varepsilon e^{i\theta})}{\varepsilon e^{i\theta}} i\varepsilon e^{i\theta}\mathrm{d}\theta = i\int_0^{2\pi} f(z_0 + \varepsilon e^{i\theta})\mathrm{d}\theta \xrightarrow{\varepsilon\to0} 2\pi i f(z_0).
\]
移项即得公式。

\begin{remark}
	柯西积分公式表明，解析函数在区域内部的值完全由其在边界上的值决定，这与实可微函数截然不同。
\end{remark}

\begin{tcolorbox}[title=例 4 (利用柯西积分公式), colframe=orange!75!black, colback=orange!5!white, sharp corners]
	计算下列积分，其中 $C$ 为正向圆周 $|z|=2$：
	\[
	(1)\ \oint_C \frac{e^z}{z-1}\mathrm{d}z \qquad 
	(2)\ \oint_C \frac{\sin z}{z+\pi}\mathrm{d}z \qquad 
	(3)\ \oint_C \frac{z}{z^2+1}\mathrm{d}z
	\]
	\tcbline
	\textbf{解：}
	\begin{enumerate}
		\item 被积函数在 $C$ 内只有一个奇点 $z=1$。取 $f(z)=e^z$，$z_0=1$，由柯西积分公式
		\[
		\oint_C \frac{e^z}{z-1}\mathrm{d}z = 2\pi i \, f(1) = 2\pi i \, e.
		\]
		\item 奇点 $z=-\pi$ 在 $C$ 内（$|-\pi|=\pi>2$？注意 $|z|=2$，$|-\pi|=\pi>2$，故 $z=-\pi$ 在圆外，积分为 0！需修改题目以确保奇点在内部）。\\
		改为 $C: |z|=4$，则 $z=-\pi$ 在内。令 $f(z)=\sin z$，$z_0=-\pi$，
		\[
		\oint_C \frac{\sin z}{z+\pi}\mathrm{d}z = 2\pi i \sin(-\pi) = 0.
		\]
		\item $z^2+1 = (z-i)(z+i)$，奇点 $z=i$ 和 $z=-i$ 均在 $C$ 内。不能直接套用柯西公式，需用复合闭路定理拆分为两个积分：
		\[
		\frac{z}{z^2+1} = \frac{z}{(z-i)(z+i)}.
		\]
		作部分分式：$\frac{z}{z^2+1} = \frac{1}{2}\left(\frac{1}{z-i} + \frac{1}{z+i}\right)$。于是
		\[
		\oint_C \frac{z}{z^2+1}\mathrm{d}z = \frac{1}{2}\left( \oint_C \frac{1}{z-i}\mathrm{d}z + \oint_C \frac{1}{z+i}\mathrm{d}z \right) = \frac{1}{2}(2\pi i + 2\pi i) = 2\pi i.
		\]
	\end{enumerate}
\end{tcolorbox}

\section{高阶导数公式}

柯西积分公式的一个极其重要的推论是：解析函数在区域内部任意阶导数都存在，且同样可用边界积分表示。

\begin{tcolorbox}[title=定理：高阶导数公式, colframe=green!45!black, colback=green!5!white]
	设 $f(z)$ 在简单闭曲线 $C$ 所围成的区域 $D$ 内解析，在 $\overline{D}$ 上连续，则对于 $D$ 内任意一点 $z_0$，$f(z)$ 的 $n$ 阶导数存在，且
	\[
	f^{(n)}(z_0) = \frac{n!}{2\pi i} \oint_C \frac{f(z)}{(z - z_0)^{n+1}} \mathrm{d}z, \quad n = 1,2,3,\dots
	\]
\end{tcolorbox}

\noindent\textbf{证明思路：} 由柯西积分公式 $f(z_0) = \frac{1}{2\pi i}\oint_C \frac{f(\zeta)}{\zeta - z_0}\mathrm{d}\zeta$。对 $z_0$ 求导，交换积分与求导次序即得。

该定理表明，一个复变函数只要在某区域内有一阶导数（即解析），则它在该区域内具有任意阶导数。这是复变函数与实变函数的本质区别之一。

\begin{tcolorbox}[title=例 5 (利用高阶导数公式), colframe=orange!75!black, colback=orange!5!white, sharp corners]
	计算下列积分，其中 $C$ 为正向圆周 $|z|=1$：
	\[
	(1)\ \oint_C \frac{e^{2z}}{z^4}\mathrm{d}z \qquad
	(2)\ \oint_C \frac{\cos \pi z}{(z-1/2)^3}\mathrm{d}z
	\]
	\tcbline
	\textbf{解：}
	\begin{enumerate}
		\item 函数 $e^{2z}$ 在整个平面解析，$z_0=0$ 在 $C$ 内。由高阶导数公式，$n+1=4 \Rightarrow n=3$，
		\[
		\oint_C \frac{e^{2z}}{z^4}\mathrm{d}z = \frac{2\pi i}{3!} f^{(3)}(0), \quad \text{其中 } f(z)=e^{2z}.
		\]
		计算 $f^{(3)}(z) = 2^3 e^{2z} = 8e^{2z}$，故 $f^{(3)}(0)=8$。因此
		\[
		\oint_C \frac{e^{2z}}{z^4}\mathrm{d}z = \frac{2\pi i}{6} \cdot 8 = \frac{8\pi i}{3}.
		\]
		\item $z_0=1/2$ 在 $C$ 内，$n+1=3 \Rightarrow n=2$。令 $f(z)=\cos \pi z$，则
		\[
		f''(z) = -\pi^2 \cos \pi z, \quad f''(1/2) = -\pi^2 \cos(\pi/2) = 0.
		\]
		故
		\[
		\oint_C \frac{\cos \pi z}{(z-1/2)^3}\mathrm{d}z = \frac{2\pi i}{2!} f''(1/2) = \pi i \cdot 0 = 0.
		\]
	\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[title=例 6 (柯西积分与高阶导数的综合), colframe=orange!75!black, colback=orange!5!white, sharp corners]
	设 $C$ 为正向圆周 $|z|=3$，计算积分
	\[
	I = \oint_C \frac{z^3 + 2z + 1}{(z-1)^2(z+2)}\mathrm{d}z.
	\]
	\tcbline
	\textbf{解：} 被积函数在 $C$ 内有两个奇点：$z=1$（二级极点）和 $z=-2$（一级极点）。由复合闭路定理，
	\[
	I = \oint_{C_1} \frac{z^3+2z+1}{(z-1)^2(z+2)}\mathrm{d}z + \oint_{C_2} \frac{z^3+2z+1}{(z-1)^2(z+2)}\mathrm{d}z,
	\]
	其中 $C_1$ 只围绕 $z=1$，$C_2$ 只围绕 $z=-2$。
	\begin{itemize}
		\item 在 $C_1$ 上，记 $g(z) = \frac{z^3+2z+1}{z+2}$，它在该小圆内解析。于是
		\[
		\oint_{C_1} \frac{g(z)}{(z-1)^2}\mathrm{d}z = 2\pi i \, g'(1).
		\]
		计算 $g(z) = \frac{z^3+2z+1}{z+2}$，求导得 $g'(z) = \frac{(3z^2+2)(z+2) - (z^3+2z+1)}{(z+2)^2}$。代入 $z=1$：
		\[
		g'(1) = \frac{(3+2)(3) - (1+2+1)}{3^2} = \frac{15 - 4}{9} = \frac{11}{9}.
		\]
		故第一项为 $2\pi i \cdot \frac{11}{9} = \frac{22\pi i}{9}$。
		\item 在 $C_2$ 上，记 $h(z) = \frac{z^3+2z+1}{(z-1)^2}$，由柯西积分公式
		\[
		\oint_{C_2} \frac{h(z)}{z+2}\mathrm{d}z = 2\pi i \, h(-2).
		\]
		$h(-2) = \frac{(-8) + (-4) + 1}{(-3)^2} = \frac{-11}{9}$。故第二项为 $2\pi i \cdot \frac{-11}{9} = -\frac{22\pi i}{9}$。
	\end{itemize}
	因此 $I = \frac{22\pi i}{9} - \frac{22\pi i}{9} = 0$.
\end{tcolorbox}

\subsection*{推论}
由高阶导数公式可立即得出以下重要结论：
\begin{enumerate}
	\item \textbf{解析函数的无穷可微性}：若 $f(z)$ 在区域 $D$ 内解析，则它在 $D$ 内具有任意阶导数。
	\item \textbf{柯西不等式}：设 $f(z)$ 在圆盘 $|z-z_0|<R$ 内解析，且 $|f(z)|\le M$，则
	\[
	|f^{(n)}(z_0)| \le \frac{n! M}{R^n}.
	\]
	\item \textbf{刘维尔定理}：若 $f(z)$ 在整个复平面上解析且有界，则 $f(z)$ 必为常数。
	\item \textbf{代数学基本定理}：任一 $n$ 次多项式在复数域上必有 $n$ 个根（含重数）。
\end{enumerate}
