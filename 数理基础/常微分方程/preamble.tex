    colframe=green!50!black,      % 边框：绿色50% + 黑色50%

    fonttitle=\bfseries,
    boxrule=0.5mm,
    before skip=10pt,
    after skip=10pt,
    segmentation style={green!50!black, line width=0.5mm},  % 分割线与边框同色
}
