\chapter{布尔代数}
\label{chap:布尔代数}

数字电路的基础为逻辑代数，由英国数学家 George Boole 在 1847 年提出，逻辑代数也称布尔代数。

\section{基本逻辑运算}
\label{sec:基本逻辑运算}

在逻辑代数中，变量常用字母 $A,B,C,\cdots,Y,Z$、$a,b,c,\cdots,x,y,z$ 等表示，变量的取值只能是“0”或“1”。

逻辑代数中只有三种基本逻辑运算，即“与”、“或”、“非”。

\subsection{与逻辑运算}
\label{sec:与逻辑运算}

\begin{definition}[与逻辑运算]
	只有决定一事件的\textbf{全部条件}都具备时，这件事才成立；如果有一个或一个以上条件不具备，则这件事就不成立。这样的因果关系称为“与”逻辑关系，其函数式为 $F=A\cdot B=AB$。

	与门逻辑功能概括：\textcolor{red}{有“0”出“0”，全“1”出“1”}。
\end{definition}

\subsection{或逻辑运算}
\label{sec:或逻辑运算}

\begin{definition}[或逻辑运算]
	在决定一事件的各种条件中，只要有一个或一个以上条件具备时，这件事就成立；\textbf{只有}所有的条件\textbf{都不具备}时，这件事才\textbf{不成立}。这样的因果关系称为“或”逻辑关系，其函数式为 $F=A+B$。

	或门逻辑功能概括：\textcolor{red}{有“1”出“1”，全“0”出“0”}。
\end{definition}

\subsection{非逻辑运算}
\label{sec:非逻辑运算}

\begin{definition}[非逻辑运算]
	假定事件 $F$ 成立与否同条件 $A$ 的具备与否有关：若 $A$ 具备，则 $F$ 不成立；若 $A$ 不具备，则 $F$ 成立。$F$ 和 $A$ 之间的这种因果关系称为“非”逻辑关系，其函数式为 $F=\overline{A}$。

\section{逻辑电平及正、负逻辑}
\label{sec:逻辑电平}

门电路的输入、输出为二值信号，用“0”和“1”表示，这里的“0”、“1”一般用两个不同电平值来表示。

\begin{definition}[正逻辑与负逻辑]
	\begin{itemize}
		\item \textbf{正逻辑}：若用高电平 $V_H$ 表示逻辑“1”，用低电平 $V_L$ 表示逻辑“0”，则称为正逻辑约定，简称正逻辑；
		\item \textbf{负逻辑}：若用高电平 $V_H$ 表示逻辑“0”，用低电平 $V_L$ 表示逻辑“1”，则称为负逻辑约定，简称负逻辑。
	\end{itemize}
	一般采取正逻辑表示。
\end{definition}

\begin{note}
	对一个特定的逻辑门，采用不同的逻辑表示时，其门的名称也就不同。
\end{note}

\section{基本定律和规则}
\label{sec:基本定律和规则}

\subsection{逻辑函数的相等}
\label{sec:逻辑函数的相等}

设有两个逻辑函数
\begin{equation}
	F_1=f_1(A_1,A_2,\cdots,A_n),\qquad
	F_2=f_2(A_1,A_2,\cdots,A_n)
	\label{eq:定律:函数定义}
\end{equation}
如果对于 $A_1,A_2,\cdots,A_n$ 的任何一组取值（共 $2^n$ 组），$F_1$ 和 $F_2$ 均相等，则称 $F_1$ 和 $F_2$ 相等。

因此，如两个函数的真值表相等，则这两个函数一定相等。

\subsubsection{代入规则}
\label{sec:代入规则}

任何一个含有变量 $x$ 的等式，如果将所有出现 $x$ 的位置，都用一个逻辑函数式 $F$ 代替，则等式仍然成立。

\section{常用公式}
\label{sec:常用公式}

\subsection{消去律}
\label{sec:消去律}

\begin{theorem}[消去律]
	\begin{equation}
		AB+A\overline{B}=A
		\label{eq:常用公式:消去律}
	\end{equation}
\end{theorem}

\begin{proof}
	\begin{equation}
		AB+A\overline{B}=A(B+\overline{B})=A\cdot1=A
		\xrightarrow{\text{对偶关系}}(A+B)(A+\overline{B})=A
	\end{equation}
\end{proof}

\subsection{吸收律1}
\label{sec:吸收律1}

\begin{theorem}[吸收律1]
	\begin{equation}
		A+AB=A
		\label{eq:常用公式:吸收律1}
	\end{equation}
\end{theorem}

\begin{proof}
	\begin{equation}
		A+AB=A(1+B)=A\cdot1=A
		\xrightarrow{\text{对偶关系}}A(A+B)=A
	\end{equation}
\end{proof}

\subsection{吸收律2}
\label{sec:吸收律2}

\begin{theorem}[吸收律2]
	\begin{equation}
		A+\overline{A}B=A+B
		\label{eq:常用公式:吸收律2}
	\end{equation}
\end{theorem}

\begin{proof}
	\begin{equation}
		\begin{aligned}
			A+\overline{A}B &= (A+\overline{A})(A+B)=1\cdot(A+B)
			\xrightarrow{\text{对偶关系}}A(\overline{A}+B)=AB \\
			&= A+B
		\end{aligned}
	\end{equation}
\end{proof}

\subsection{包含律}
\label{sec:包含律}

\begin{theorem}[包含律]
	\begin{equation}
		AB+\overline{A}C+BC=AB+\overline{A}C
		\label{eq:常用公式:包含律}
	\end{equation}
\end{theorem}

\begin{proof}
	\begin{equation}
		\begin{aligned}
			&AB+\overline{A}C+BC \\
			&=AB+\overline{A}C+(A+\overline{A})BC \\
			&=AB+\overline{A}C+ABC+\overline{A}BC \\
			&=AB(1+C)+\overline{A}C(1+B)
		\end{aligned}
	\end{equation}
	由对偶关系，可得
	\begin{equation}
		(A+B)(\overline{A}+C)(B+C)=(A+B)(\overline{A}+C)
	\end{equation}
\end{proof}

\section{逻辑函数的标准形式}
\label{sec:逻辑函数的标准形式}

\subsection{函数的“与--或”式和“或--与”式}
\label{sec:与或式和或与式}

\begin{definition}[“与--或”式]
	“与--或”式指一个函数表达式中包含若干个“与”项，这些“与”项的“或”表示这个函数，如
	\begin{equation}
		F(A,B,C,D)=A+B\overline{C}+ABC\overline{D}
		\label{eq:标准形式:与或式}
	\end{equation}
\end{definition}

\begin{definition}[“或--与”式]
	“或--与”式指一个函数表达式中包含若干个“或”项，这些“或”项的“与”表示这个函数，如
	\begin{equation}
		F(A,B,C,D)=(A+C+D)(\overline{B}+D)(A+\overline{B}+\overline{D})
		\label{eq:标准形式:或与式}
	\end{equation}
\end{definition}

\subsection{最小项}
\label{sec:最小项}

\begin{definition}[最小项]
	最小项是“与”项，具有以下特点：
	\begin{enumerate}
		\item $n$ 个变量构成的每个最小项一定包含 $n$ 个因子的乘积项；
		\item 每个变量必须以原变量或反变量形式作为因子出现\textbf{一次且仅一次}。
	\end{enumerate}
	例如：两变量有 $2^2=4$ 个最小项：$A\overline{B}$、$\overline{A}B$、$\overline{A}\,\overline{B}$、$AB$；三变量有 $2^3=8$ 个最小项。
\end{definition}

\begin{definition}[最小项的编号]
	任一最小项用 $m_i$ 表示，$m$ 表示最小项，下标 $i$ 为使该最小项为 1 的变量取值所对应的等效十进制数。

	例如：使最小项 $A\overline{B}C$ 为 1 的取值 $A,B,C=1,0,1$，二进制数 101 等效十进制数为 5，所以 $A\overline{B}C=m_5$。
\end{definition}

\begin{property}[最小项的性质]
	\begin{enumerate}
		\item 变量任取一组值，仅有一个最小项为 1，其他最小项为零；
		\item $n$ 变量的全体最小项之和为 1；
		\item 不同的最小项相与，结果为 0；
		\item 相邻最小项相“或”可以合并成一项，并消去一个变量因子。
	\end{enumerate}
\end{property}

\begin{definition}[最小项相邻]
	两最小项如仅有一个变量因子不同，其他变量均相同，则称这两个最小项相邻。例如 $\overline{A}BC+ABC=BC$。任一 $n$ 变量的最小项，必定和其他 $n$ 个不同最小项相邻。
\end{definition}

\subsection{最大项}
\label{sec:最大项}

\begin{definition}[最大项]
	最大项是“或”项，具有以下特点：
	\begin{enumerate}
		\item 每个最大项一定包含 $n$ 个因子的“或”项；
		\item 每个变量必须以原变量或反变量形式出现\textbf{一次且仅一次}。
	\end{enumerate}
	例如：两变量有 $A+B$、$A+\overline{B}$、$\overline{A}+B$、$\overline{A}\,\overline{B}$ 四个最大项。
\end{definition}

\begin{definition}[最大项的编号]
	任一最大项用 $M_i$ 表示，下标 $i$ 为使该最大项为 \textbf{0} 的变量取值所对应的等效十进制数。

	例如：使最大项 $\overline{A}+B+C$ 为 0 的取值为 $A,B,C=1,0,0$，二进制数 100 等效十进制数为 4，所以 $\overline{A}+B+C=M_4$。
\end{definition}

\begin{property}[最大项的性质]
	\begin{enumerate}
		\item 变量任取一组值，仅有一个最大项为 0，其他最大项为 1；
		\item $n$ 变量的全体最大项之积为 0；
		\item 不同的最大项相或，结果为 1；
		\item 两相邻的最大项相“与”可以合并成一项，并消去一个变量因子。例如 $(A+B+\overline{C})(A+B+C)=A+B$。任一 $n$ 变量的最大项，必定和其他 $n$ 个不同最大项相邻。
	\end{enumerate}
\end{property}

\subsection{最小项和最大项的关系}
\label{sec:最小项和最大项的关系}

\begin{theorem}[最小项与最大项的关系]
	编号下标相同的最小项和最大项互为反函数，即
	\begin{equation}
		M_i=\overline{m_i}
		\label{eq:标准形式:最小项最大项关系}
	\end{equation}
\end{theorem}

\subsection{逻辑函数的最小项之和形式}
\label{sec:逻辑函数的最小项之和形式}

\begin{theorem}[最小项之和形式]
	最小项之和式为“与或”式，任一逻辑函数都可以表达为最小项之和的形式，而且是\textbf{唯一}的。例如
	\begin{equation}
		F(A,B,C)=\overline{A}B\overline{C}+A\overline{B}\,\overline{C}+AB\overline{C}=\Sigma m(2,4,6)
		\label{eq:标准形式:最小项之和}
	\end{equation}
\end{theorem}

\begin{example}
	将 $F(A,B,C)=AB+\overline{A}C$ 化为最小项之和的形式：
	\begin{equation}
		\begin{aligned}
			F(A,B,C) &= AB+\overline{A}C \\
			&= AB(C+\overline{C})+\overline{A}C(B+\overline{B}) \\
			&= ABC+AB\overline{C}+\overline{A}BC+\overline{A}\,\overline{B}C \\
			&= \Sigma m(1,3,6,7)
		\end{aligned}
		\label{eq:标准形式:最小项之和例}
	\end{equation}
\end{example}

\subsection{逻辑函数的最大项之积形式}
\label{sec:逻辑函数的最大项之积形式}

\begin{theorem}[最大项之积形式]
	最大项之积式为“或与”式，任一逻辑函数都可以表达为最大项之积的形式，而且是\textbf{唯一}的。例如
	\begin{equation}
		F(A,B,C)=(A+B+C)(A+\overline{B}+C)(\overline{A}+B+C)=\Pi M(0,2,4)
		\label{eq:标准形式:最大项之积}
	\end{equation}
\end{theorem}

\begin{example}
	将 $F(A,B,C)=(\overline{A}+C)(B+\overline{C})$ 化为最大项之积的形式：
	\begin{equation}
		\begin{aligned}
			F(A,B,C) &= (\overline{A}+C)(B+\overline{C}) \\
			&= (\overline{A}+C+B\overline{B})(B+\overline{C}+A\overline{A}) \\
			&= (\overline{A}+B+C)(\overline{A}+\overline{B}+C)(A+B+\overline{C})(\overline{A}+B+\overline{C}) \\
			&= \Pi M(1,4,5,6)
		\end{aligned}
		\label{eq:标准形式:最大项之积例}
	\end{equation}
\end{example}

\subsection{两种标准形式之间的关系}
\label{sec:两种标准形式之间的关系}

\begin{theorem}[两种标准形式之间的关系]
	若 $F=\Sigma m_i$，则
	\begin{equation}
		\overline{F}=\sum_{j\neq i}m_j=\prod_{j\neq i}M_j
		\label{eq:标准形式:两种形式关系}
	\end{equation}
\end{theorem}

\begin{example}
	$F(A,B,C)=\Sigma(1,3,4,6,7)=\Pi(0,2,5)$
\end{example}

\subsection{由逻辑函数式列真值表}

\begin{example}

	\begin{equation}

			&= AB(C+\overline{C})+BC(A+\overline{A}) \\
			&= ABC+AB\overline{C}+\overline{A}BC \\
			&= \Sigma m(3,6,7)
		\end{aligned}

	\end{equation}
	最后根据最小项的性质，在真值表中对应于 $ABC$ 取值为 011、110、111 处填“1”，其它位置填“0”，即
