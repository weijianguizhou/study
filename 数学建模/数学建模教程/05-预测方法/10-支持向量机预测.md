# 支持向量机预测模型

> **理论基础** → 参见 [[../../机器学习数学基础/支持向量机|SVM理论]]

支持向量机(SVM)通过寻找最优超平面将回归问题转化为凸优化问题，在小样本非线性预测中表现突出。

## 五、Python 实现
```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.svm import SVR, SVC
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import GridSearchCV, TimeSeriesSplit
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score

class SVRPredictor:
    """支持向量回归预测模型"""
    
    def __init__(self, kernel='rbf', C=1.0, epsilon=0.1, gamma='scale'):
        """
        参数:
            kernel: 核函数类型
            C: 惩罚参数
            epsilon: epsilon不敏感损失参数
            gamma: RBF核参数
        """
        self.kernel = kernel
        self.C = C
        self.epsilon = epsilon
        self.gamma = gamma
        self.model = None
        self.scaler_X = StandardScaler()
        self.scaler_y = StandardScaler()
        
    def fit(self, X, y, scale=True):
        """拟合模型"""
        if scale:
            X = self.scaler_X.fit_transform(X)
            y = self.scaler_y.fit_transform(y.reshape(-1, 1)).ravel()
        
        self.model = SVR(
            kernel=self.kernel,
            C=self.C,
            epsilon=self.epsilon,
            gamma=self.gamma
        )
        self.model.fit(X, y)
        return self
    
    def predict(self, X, scale=True):
        """预测"""
        if scale:
            X = self.scaler_X.transform(X)
        
        y_pred = self.model.predict(X)
        
        if scale:
            y_pred = self.scaler_y.inverse_transform(y_pred.reshape(-1, 1)).ravel()
        
        return y_pred
    
    def get_metrics(self, X, y):
        """计算评估指标"""
        y_pred = self.predict(X)
        return {
            'R²': r2_score(y, y_pred),
            'MSE': mean_squared_error(y, y_pred),
            'RMSE': np.sqrt(mean_squared_error(y, y_pred)),
            'MAE': mean_absolute_error(y, y_pred),
            'MAPE': np.mean(np.abs((y - y_pred) / y)) * 100
        }
    
    def grid_search(self, X, y, param_grid=None, cv=5):
        """网格搜索超参数"""
        if param_grid is None:
            param_grid = {
                'C': [0.1, 1, 10, 100],
                'gamma': ['scale', 'auto', 0.01, 0.1, 1],
                'epsilon': [0.01, 0.05, 0.1, 0.2]
            }
        
        X_scaled = self.scaler_X.fit_transform(X)
        y_scaled = self.scaler_y.fit_transform(y.reshape(-1, 1)).ravel()
        
        svr = SVR(kernel=self.kernel)
        grid_search = GridSearchCV(svr, param_grid, cv=cv, 
                                   scoring='neg_mean_squared_error',
                                   n_jobs=-1, verbose=0)
        grid_search.fit(X_scaled, y_scaled)
        
        self.C = grid_search.best_params_['C']
        self.gamma = grid_search.best_params_['gamma']
        self.epsilon = grid_search.best_params_['epsilon']
        
        print(f"最优参数: {grid_search.best_params_}")
        print(f"最优MSE: {-grid_search.best_score_:.6f}")
        
        # 用最优参数重新训练
        self.fit(X, y)
        
        return grid_search
    
    def visualize_fit(self, X, y):
        """可视化拟合结果"""
        y_pred = self.predict(X)
        
        fig, axes = plt.subplots(1, 3, figsize=(18, 5))
        
        # 拟合曲线
        axes[0].scatter(y, y_pred, alpha=0.5, s=20)
        min_val = min(y.min(), y_pred.min())
        max_val = max(y.max(), y_pred.max())
        axes[0].plot([min_val, max_val], [min_val, max_val], 'r--', linewidth=2)
        axes[0].set_xlabel('实际值')
        axes[0].set_ylabel('预测值')
        axes[0].set_title('实际值 vs 预测值')
        axes[0].grid(True, alpha=0.3)
        
        # 残差
        residuals = y - y_pred
        axes[1].scatter(y_pred, residuals, alpha=0.5, s=20)
        axes[1].axhline(y=0, color='r', linestyle='--')
        axes[1].set_xlabel('预测值')
        axes[1].set_ylabel('残差')
        axes[1].set_title('残差图')
        axes[1].grid(True, alpha=0.3)
        
        # 残差分布
        axes[2].hist(residuals, bins=30, edgecolor='black', alpha=0.7)
        axes[2].set_xlabel('残差')
        axes[2].set_ylabel('频数')
        axes[2].set_title('残差分布')
        axes[2].grid(True, alpha=0.3)
        
        plt.tight_layout()
        plt.show()

class SVMTimeSeriesPredictor:
    """基于SVM的时间序列预测"""
    
    def __init__(self, seq_length=10, **svr_params):
        self.seq_length = seq_length
        self.svr = SVRPredictor(**svr_params)
        
    def create_features(self, data):
        """将时间序列转化为监督学习问题"""
        X, y = [], []
        for i in range(len(data) - self.seq_length):
            X.append(data[i:i+self.seq_length])
            y.append(data[i+self.seq_length])
        return np.array(X), np.array(y)
    
    def fit(self, data, test_ratio=0.2):
        """拟合模型"""
        X, y = self.create_features(data)
        
        split = int(len(X) * (1 - test_ratio))
        X_train, X_test = X[:split], X[split:]
        y_train, y_test = y[:split], y[split:]
        
        self.svr.fit(X_train, y_train)
        
        self.train_metrics = self.svr.get_metrics(X_train, y_train)
        self.test_metrics = self.svr.get_metrics(X_test, y_test)
        
        return self
    
    def predict_next(self, recent_data):
        """预测下一个值"""
        if len(recent_data) < self.seq_length:
            raise ValueError(f"需要至少{self.seq_length}个数据点")
        
        x = np.array(recent_data[-self.seq_length:]).reshape(1, -1)
        return self.svr.predict(x)[0]
    
    def predict_multi_step(self, data, steps=10):
        """多步预测"""
        predictions = list(data)
        all_preds = []
        
        for _ in range(steps):
            pred = self.predict_next(predictions)
            predictions.append(pred)
            all_preds.append(pred)
        
        return np.array(all_preds)

for kernel in kernels:
    svr = SVRPredictor(kernel=kernel, C=10, epsilon=0.1)
    svr.fit(X_train, y_train)
    metrics = svr.get_metrics(X_test, y_test)
    results[kernel] = metrics
    print(f"\n{kernel}核: R²={metrics['R²']:.4f}, RMSE={metrics['RMSE']:.4f}")

# RBF核参数优化

svr_opt.grid_search(X_train, y_train, param_grid={
    'C': [1, 10, 100],
    'gamma': [0.1, 1, 10],
    'epsilon': [0.05, 0.1, 0.2]
}, cv=5)

metrics_opt = svr_opt.get_metrics(X_test, y_test)
print(f"\n优化后: R²={metrics_opt['R²']:.4f}, RMSE={metrics_opt['RMSE']:.4f}")

svr_opt.visualize_fit(X_test, y_test)
```

## 六、实际案例：PM2.5浓度预测
### 6.1 数据准备

```python
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler

# 模拟PM2.5数据（含天气因素）
np.random.seed(42)
n_days = 365 * 3

# 特征
temperature = 15 + 15 * np.sin(2 * np.pi * np.arange(n_days) / 365) + np.random.normal(0, 3, n_days)
humidity = 60 + 20 * np.sin(2 * np.pi * np.arange(n_days) / 365 + np.pi/4) + np.random.normal(0, 5, n_days)
wind_speed = np.abs(np.random.normal(3, 2, n_days))
pm25 = 50 + 0.5 * temperature - 0.3 * humidity - 2 * wind_speed + np.random.normal(0, 10, n_days)
pm25 = np.maximum(pm25, 0)

df = pd.DataFrame({
    'temperature': temperature,
    'humidity': humidity,
    'wind_speed': wind_speed,
    'pm25': pm25
})

# 创建滞后特征
for lag in [1, 2, 3]:
    df[f'pm25_lag{lag}'] = df['pm25'].shift(lag)

df = df.dropna()
print(f"数据形状: {df.shape}")
print(df.head())
```

### 6.2 建模与评估

```python
from sklearn.model_selection import train_test_split

feature_cols = ['temperature', 'humidity', 'wind_speed', 
                'pm25_lag1', 'pm25_lag2', 'pm25_lag3']
X = df[feature_cols].values
y = df['pm25'].values

# 时序分割（避免数据泄露）

# 评估

test_metrics = svr.get_metrics(X_test, y_test)

print("=== SVR PM2.5预测 ===")
print(f"训练集: R²={train_metrics['R²']:.4f}, RMSE={train_metrics['RMSE']:.2f}")
print(f"测试集: R²={test_metrics['R²']:.4f}, RMSE={test_metrics['RMSE']:.2f}")

# 预测对比
y_pred = svr.predict(X_test)

plt.figure(figsize=(14, 5))
plt.plot(y_test[:100], 'b-', label='实际值', alpha=0.7)

plt.xlabel('样本')
plt.ylabel('PM2.5 (μg/m³)')
plt.title('SVR PM2.5浓度预测')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()
```

## 七、SVM与其他方法的比较
| 特性 | SVR | 神经网络 | 随机森林 | ARIMA |
|------|-----|---------|---------|-------|
| 小样本 | 优秀 | 差 | 一般 | 好 |
| 大样本 | 一般 | 优秀 | 优秀 | 一般 |
| 非线性 | 好（核方法） | 优秀 | 好 | 差 |
| 高维数据 | 好 | 好 | 好 | 不适用 |
| 可解释性 | 一般 | 差 | 好 | 好 |
| 训练速度 | 慢（大样本） | 慢 | 快 | 快 |
| 超参数敏感度 | 高 | 高 | 低 | 中 |

## 八、关键要点总结
1. **SVM的核心思想**：最大间隔分类/回归
2. **核技巧**：隐式映射到高维空间，避免维度灾难
3. **$\epsilon$-不敏感损失**：SVR的关键，允许预测误差在 $\epsilon$ 范围内
4. **支持向量**：决定决策边界的少数样本点
5. **核函数选择**：RBF核是最通用的选择，线性核适合高维数据
6. **超参数 $C$**：控制模型复杂度与拟合度的平衡
7. **超参数 $\gamma$**：RBF核的带宽参数，控制决策边界的复杂度
8. **数据标准化**：对SVM性能影响很大，必须进行
9. **时间序列应用**：需要将序列数据转化为监督学习格式
10. **SVM的优势**：泛化能力强、理论基础坚实、适合中小样本
11. **SVM的局限**：训练时间复杂度 $O(n^2 \sim n^3)$、对大规模数据不友好、核函数选择困难
