# 灰色关联分析

> **理论基础** → 参见 [[../../统计学分析方法/灰色系统理论|灰色系统理论]]

灰色关联分析基于序列曲线几何形状的相似程度判断因素间的关联强度，适用于贫信息、小样本系统。

## 八、Python 实现
### 8.1 基础灰色关联度计算

```python
import numpy as np

class GreyRelationalAnalysis:
    def __init__(self, ref_seq, compare_seqs, rho=0.5, method='init'):
        """
        灰色关联分析
        ref_seq: 参考序列 (1D array)
        compare_seqs: 比较序列 (2D array, 每行一个序列)
        rho: 分辨系数
        method: 无量纲化方法 ('init'=初值化, 'mean'=均值化, 'minmax'=区间化)
        """
        self.ref = np.array(ref_seq, dtype=float)
        self.comps = np.array(compare_seqs, dtype=float)
        self.rho = rho
        self.method = method
        self.n = len(ref_seq)
        self.m = len(compare_seqs)
    
    def normalize(self, data):
        """数据无量纲化"""
        if self.method == 'init':
            return data / data[0]
        elif self.method == 'mean':
            return data / np.mean(data)
        elif self.method == 'minmax':
            min_val = np.min(data)
            max_val = np.max(data)
            if max_val == min_val:
                return np.ones_like(data)
            return (data - min_val) / (max_val - min_val)
        elif self.method == 'abs_init':
            result = data.copy().astype(float)
            result[0] = abs(data[0]) if data[0] != 0 else 1
            return data / result[0]
        return data
    
    def compute(self):
        """计算灰色关联度"""
        # 无量纲化
        ref_norm = self.normalize(self.ref)
        comps_norm = np.array([self.normalize(comp) for comp in self.comps])
        
        # 计算差值序列
        diff = np.abs(ref_norm - comps_norm)
        
        # 两级最小差和最大差
        delta_min = np.min(diff)
        delta_max = np.max(diff)
        
        # 计算关联系数矩阵
        xi = (delta_min + self.rho * delta_max) / (diff + self.rho * delta_max)
        
        # 计算关联度
        r = np.mean(xi, axis=1)
        
        # 排序
        ranking = np.argsort(-r) + 1  # 从1开始编号
        
        return {
            'coefficients': xi,  # 关联系数矩阵
            'relational_degree': r,  # 关联度
            'ranking': ranking,  # 排序
            'delta_min': delta_min,
            'delta_max': delta_max,
            'rho': self.rho
        }

# 示例：比较 5 种方案对 4 个指标的表现
np.random.seed(42)

# 参考序列（理想最优方案）
reference = np.array([100, 90, 85, 95])

# 比较序列（5 种方案的 4 个指标得分）
schemes = np.array([
    [95, 85, 80, 90],   # 方案 1
    [80, 90, 75, 85],   # 方案 2
    [90, 80, 90, 80],   # 方案 3
    [70, 85, 70, 95],   # 方案 4
    [85, 95, 85, 75],   # 方案 5
])

gra = GreyRelationalAnalysis(reference, schemes, rho=0.5, method='init')
result = gra.compute()

print("=== 灰色关联分析结果 ===")
print(f"分辨系数 ρ = {result['rho']}")
print(f"两级最小差 Δ_min = {result['delta_min']:.4f}")
print(f"两级最大差 Δ_max = {result['delta_max']:.4f}")

print(f"\n关联系数矩阵:")
for i in range(len(schemes)):
    coeffs_str = "  ".join([f"{c:.4f}" for c in result['coefficients'][i]])
    print(f"  方案{i+1}: {coeffs_str}")

print(f"\n关联度:")
for i in range(len(schemes)):
    print(f"  方案{i+1}: {result['relational_degree'][i]:.4f}")

print(f"\n排名: {result['ranking']}")
print(f"最优方案: 方案{result['ranking'][0]}")
```

### 8.2 多指标评价案例

```python
import numpy as np

def grey_relational_evaluation(indicators, weights, rho=0.5):
    """
    加权灰色关联评价
    indicators: 形状为 (n_samples, n_criteria) 的矩阵
    weights: 各指标权重
    """
    n, m = indicators.shape
    
    # 参考序列：每个指标的最优值
    # 对于效益型指标（越大越好），取最大值
    # 对于成本型指标（越小越好），取最小值
    # 这里假设所有指标都是效益型
    reference = np.max(indicators, axis=0)
    
    # 无量纲化（均值化）
    indicators_norm = indicators / np.mean(indicators, axis=0)
    reference_norm = reference / np.mean(indicators, axis=0)
    
    # 计算差值
    diff = np.abs(indicators_norm - reference_norm)
    
    delta_min = np.min(diff)
    delta_max = np.max(diff)
    
    # 关联系数
    xi = (delta_min + rho * delta_max) / (diff + rho * delta_max)
    
    # 加权关联度
    weighted_degree = xi @ weights
    
    # 排序
    ranking = np.argsort(-weighted_degree)
    
    return {
        'coefficients': xi,
        'degree': weighted_degree,
        'ranking': ranking + 1,
        'reference': reference
    }

# 案例：城市综合实力评价
cities = np.array([
    # GDP, 人口, 教育投入, 科技产出, 环境质量
    [12000, 2500, 500, 800, 85],   # 城市 A
    [15000, 3200, 650, 950, 78],   # 城市 B
    [8000, 1800, 350, 500, 92],    # 城市 C
    [20000, 4000, 800, 1200, 70],  # 城市 D
    [10000, 2000, 450, 650, 88],   # 城市 E
])

city_names = ['A', 'B', 'C', 'D', 'E']
indicator_names = ['GDP', '人口', '教育投入', '科技产出', '环境质量']
weights = np.array([0.25, 0.15, 0.20, 0.25, 0.15])  # 权重

result = grey_relational_evaluation(cities, weights)

print("=== 城市综合实力灰色关联评价 ===\n")
print(f"权重: {dict(zip(indicator_names, weights))}\n")

print("关联系数矩阵:")
header = f"{'城市':>6}" + "".join([f" {name:>8}" for name in indicator_names])
print(header)
for i in range(len(cities)):
    coeffs = " ".join([f"{c:>8.4f}" for c in result['coefficients'][i]])
    print(f"{'城市'+city_names[i]:>6} {coeffs}")

print(f"\n加权关联度:")
for i in result['ranking']:
    print(f"  第{list(result['ranking']).index(i)+1}名: 城市{city_names[i-1]}, 关联度 = {result['degree'][i-1]:.4f}")

print(f"\n最优城市: 城市{city_names[result['ranking'][0]-1]}")
```

import numpy as np

def compare_normalization_methods(data):
    """比较不同无量纲化方法对关联度的影响"""
    
    reference = data[0]  # 假设第 1 行是参考序列
    comparisons = data[1:]
    
    methods = {
        'init': '初值化',
        'mean': '均值化',
        'minmax': '区间化'
    }
    
    print("=== 不同无量纲化方法对关联度的影响 ===\n")

        for i, deg in enumerate(result['relational_degree']):
            print(f"  序列{i+1}: {deg:.4f}")

                   "**" if result['relational_degree'][idx-1] > 0.6 else "*"
    print(f"  第{rank+1}名: {feature_names[idx-1]}, "
          f"关联度 = {result['relational_degree'][idx-1]:.4f} {significance}")

for name, coef in zip(selected_features, model.coef_):

## 十、灰色关联度的扩展方法
### 10.1 灰色关联聚类

基于关联度矩阵进行聚类分析：

```python
import numpy as np

def grey_relational_clustering(data, rho=0.5, threshold=0.75):
    """基于灰色关联度的聚类"""
    n = len(data)
    
    # 计算关联度矩阵
    gra_matrix = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if i != j:
                remaining = np.array([data[j]])
                gra = GreyRelationalAnalysis(data[i], remaining, rho=rho, method='mean')
                result = gra.compute()
                gra_matrix[i, j] = result['relational_degree'][0]
            else:
                gra_matrix[i, j] = 1.0
    
    # 基于阈值的聚类
    clusters = []
    assigned = set()
    
    for i in range(n):
        if i not in assigned:
            cluster = {i}
            for j in range(i+1, n):
                if j not in assigned and gra_matrix[i, j] >= threshold:
                    cluster.add(j)
            clusters.append(cluster)
            assigned.update(cluster)
    
    return gra_matrix, clusters

# 示例
np.random.seed(42)
data = np.array([
    [10, 20, 30],
    [12, 22, 32],
    [50, 60, 70],
    [52, 58, 68],
    [90, 10, 50],
])

gra_matrix, clusters = grey_relational_clustering(data, threshold=0.6)
print("=== 灰色关联聚类 ===\n")
print("关联度矩阵:")
print(np.round(gra_matrix, 4))
print(f"\n聚类结果 (阈值=0.6):")
for i, cluster in enumerate(clusters):
    print(f"  类别 {i+1}: {sorted(cluster)}")
```

### 10.2 优势分析

灰色关联优势分析用于比较不同参考序列下各因素关联度的优劣关系。

---

## 关键要点
1. **灰色关联分析**通过比较序列几何形状的相似程度判断因素间的关联，适用于小样本、贫信息系统
2. **经典灰色关联度**计算步骤：无量纲化 → 计算差值 → 求关联系数 $\xi_i(k)$ → 求关联度 $r_i$
3. **分辨系数 $\rho$** 通常取 0.5，调节关联系数的区分度；$\rho$ 越小，分辨力越强
4. **绝对关联度**反映绝对量的变化关系，**相对关联度**反映相对变化速率的关系
5. **加权灰色关联度**结合各指标权重，适用于多指标综合评价问题
6. 无量纲化方法（初值化、均值化、区间化）的选择会影响结果，需根据数据特点选择
7. 灰色关联分析可与**回归分析、聚类分析、层次分析法**等方法结合使用
8. 数学建模中常用于：**因素筛选、综合评价、方案优选、影响因素排序**等场景
