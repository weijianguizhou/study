# 三、student.h

```cpp
#pragma once
#include <string>
#include <vector>
#include <iostream>

class Student {
    std::string name;
    int age;
    double score;
public:
    Student() = default;
    Student(const std::string &n, int a, double s)
        : name(n), age(a), score(s) {}

    std::string getName()    const { return name; }
    double      getScore()   const { return score; }

    void print() const {
        std::cout << "姓名: " << name
                  << "  年龄: " << age
                  << "  成绩: " << score << std::endl;
    }
};
```

# 四、main.cpp

```cpp
#include "student.h"
#include <algorithm>
#include <fstream>
#include <sstream>

std::vector<Student> students;

void add_student() {
    std::string name;
    int age;
    double score;
    std::cout << "姓名 年龄 成绩: ";
    std::cin >> name >> age >> score;
    students.emplace_back(name, age, score);
    std::cout << "添加成功\n";
}

void show_all() {
    if (students.empty()) {
        std::cout << "暂无学生\n";
        return;
    }
    for (size_t i = 0; i < students.size(); i++) {
        std::cout << "[" << i + 1 << "] ";
        students[i].print();
    }
}

void find_student() {
    std::string name;
    std::cout << "查找姓名: ";
    std::cin >> name;
    for (auto &s : students) {
        if (s.getName() == name) {
            s.print();
            return;
        }
    }
    std::cout << "未找到\n";
}

void delete_student() {
    std::string name;
    std::cout << "删除姓名: ";
    std::cin >> name;
    auto it = std::find_if(students.begin(), students.end(),
        [&](Student &s) { return s.getName() == name; });
    if (it != students.end()) {
        students.erase(it);
        std::cout << "已删除\n";
    } else {
        std::cout << "未找到\n";
    }
}

void sort_by_score() {
    std::sort(students.begin(), students.end(),
        [](Student &a, Student &b) {
            return a.getScore() > b.getScore();  // 降序
        });
    std::cout << "排序完成\n";
    show_all();
}

void save_to_file() {
    std::ofstream f("students.txt");
    for (auto &s : students)
        f << s.getName() << " " << s.getScore() << "\n";
    std::cout << "已保存到 students.txt\n";
}

void load_from_file() {
    std::ifstream f("students.txt");
    if (!f) {
        std::cout << "文件不存在\n";
        return;
    }
    students.clear();
    std::string name;
    double score;
    int age = 0;
    while (f >> name >> score)
        students.emplace_back(name, age, score);
    std::cout << "已加载 " << students.size() << " 条记录\n";
}

int main() {
    while (true) {
        std::cout << "\n===== 学生管理系统 =====\n"
                  << "1.添加 2.显示 3.查找 4.删除 5.排序\n"
                  << "6.保存 7.加载 0.退出\n"
                  << "选择: ";

        int choice;
        std::cin >> choice;

        switch (choice) {
            case 1: add_student();    break;
            case 2: show_all();       break;
            case 3: find_student();   break;
            case 4: delete_student(); break;
            case 5: sort_by_score();  break;
            case 6: save_to_file();   break;
            case 7: load_from_file(); break;
            case 0: return 0;
            default: std::cout << "无效选项\n";
        }
    }
}
```

# 五、CMakeLists.txt

cmake_minimum_required(VERSION 3.15)
