## 概念解释

### QTimer类

QTimer是Qt提供的定时器类，可以周期性地触发信号。

### 常用方法

| 方法 | 说明 |
|------|------|
| start(ms) | 启动定时器（毫秒） |
| stop() | 停止定时器 |
| setInterval(ms) | 设置间隔 |
| isActive() | 是否运行中 |
| singleShot() | 静态方法，单次触发 |

### 定时器信号

| 信号 | 说明 |
|------|------|
| timeout() | 定时器超时 |
| timeout(int) | 带参数的超时 |

## 代码示例

### 基本定时器

`cpp
class TimerWidget : public QWidget
{
    Q_OBJECT
    
public:
    explicit TimerWidget(QWidget *parent = nullptr) : QWidget(parent)
        , m_seconds(0)
    {
        // 创建定时器
        m_timer = new QTimer(this);
        connect(m_timer, &QTimer::timeout, this, &TimerWidget::onTimeout);
        
        // 启动定时器（每秒触发一次）
        m_timer->start(1000);
    }
    
private slots:
    void onTimeout()
    {
        m_seconds++;
        update();
    }
    
protected:
    void paintEvent(QPaintEvent *) override
    {
        QPainter painter(this);
        painter.setFont(QFont("Arial", 24));
        painter.drawText(rect(), Qt::AlignCenter, 
            QString("已运行 %1 秒").arg(m_seconds));
    }
    
private:
    QTimer *m_timer;
    int m_seconds;
};
`

### 动画定时器

`cpp
class AnimationWidget : public QWidget
{
    Q_OBJECT
    
public:
    explicit AnimationWidget(QWidget *parent = nullptr) : QWidget(parent)
        , m_angle(0)
    {
        m_timer = new QTimer(this);
        connect(m_timer, &QTimer::timeout, [this]() {
            m_angle = (m_angle + 5) % 360;
            update();
        });
        m_timer->start(50);  // 20fps
    }
    
protected:
    void paintEvent(QPaintEvent *) override
    {
        QPainter painter(this);
        painter.setRenderHint(QPainter::Antialiasing);
        painter.translate(width() / 2, height() / 2);
        painter.rotate(m_angle);
        
        painter.setBrush(Qt::blue);
        painter.drawRect(-25, -25, 50, 50);
    }
    
private:
    QTimer *m_timer;
    int m_angle;
};
`

### 单次定时器

`cpp
// 静态方法：单次触发
QTimer::singleShot(2000, this, [this]() {
    qDebug() << "2秒后执行";
});

// 延迟执行
QTimer::singleShot(0, this, &MyClass::doLater);
`

### 计时器

`cpp
class Stopwatch : public QObject
{
    Q_OBJECT
    
public:
    void start() {
        m_elapsed.start();
        m_timer.start();
    }
    
    void stop() {
        m_timer.stop();
    }
    
    qint64 elapsed() const {
        return m_elapsed.elapsed();
    }
    
signals:
    void elapsedChanged(qint64 ms);
    
private:
    QTimer m_timer;
    QElapsedTimer m_elapsed;
};
`

## 注意事项

1. 定时器精度受系统调度影响
2. 避免在定时器回调中执行耗时操作
3. 使用stop()停止不需要的定时器
4. 单次定时器适合延迟执行
