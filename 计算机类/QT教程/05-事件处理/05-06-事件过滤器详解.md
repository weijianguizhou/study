# 05-06 事件过滤器详解

## 概念解释

### 事件过滤器

事件过滤器允许你在事件到达目标对象之前拦截和处理事件。

### 实现事件过滤器

`cpp
class MyEventFilter : public QObject
{
    Q_OBJECT
    
public:
    explicit MyEventFilter(QObject *parent = nullptr) : QObject(parent) {}
    
protected:
    bool eventFilter(QObject *watched, QEvent *event) override
    {
        if (event->type() == QEvent::MouseButtonPress) {
            QMouseEvent *mouseEvent = static_cast<QMouseEvent*>(event);
            qDebug() << "鼠标按下在：" << mouseEvent->pos();
            
            // 返回true表示事件已处理，不再传播
            // 返回false表示继续传播
            return false;
        }
        
        return QObject::eventFilter(watched, event);
    }
};
`

### 安装事件过滤器

`cpp
// 创建目标控件
QPushButton *button = new QPushButton("按钮", this);

// 创建并安装事件过滤器
MyEventFilter *filter = new MyEventFilter(this);
button->installEventFilter(filter);
`

## 代码示例

### 通用事件过滤器

`cpp
class GeneralEventFilter : public QObject
{
    Q_OBJECT
    
public:
    explicit GeneralEventFilter(QObject *parent = nullptr) : QObject(parent) {}
    
signals:
    void mousePressed(QWidget *widget, QMouseEvent *event);
    void mouseReleased(QWidget *widget, QMouseEvent *event);
    void keyPressed(QWidget *widget, QKeyEvent *event);
    
protected:
    bool eventFilter(QObject *watched, QEvent *event) override
    {
        QWidget *widget = qobject_cast<QWidget*>(watched);
        if (!widget) return false;
        
        switch (event->type()) {
        case QEvent::MouseButtonPress: {
            QMouseEvent *mouseEvent = static_cast<QMouseEvent*>(event);
            emit mousePressed(widget, mouseEvent);
            break;
        }
        case QEvent::MouseButtonRelease: {
            QMouseEvent *mouseEvent = static_cast<QMouseEvent*>(event);
            emit mouseReleased(widget, mouseEvent);
            break;
        }
        case QEvent::KeyPress: {
            QKeyEvent *keyEvent = static_cast<QKeyEvent*>(event);
            emit keyPressed(widget, keyEvent);
            break;
        }
        default:
            break;
        }
        
        return false;
    }
};

// 使用
GeneralEventFilter *filter = new GeneralEventFilter(this);
connect(filter, &GeneralEventFilter::mousePressed, 
    [](QWidget *w, QMouseEvent *e) {
        qDebug() << "控件" << w << "被点击";
    });

// 为多个控件安装同一个过滤器
for (auto widget : findChildren<QWidget*>()) {
    widget->installEventFilter(filter);
}
`

### 输入验证过滤器

`cpp
class InputFilter : public QObject
{
    Q_OBJECT
    
public:
    explicit InputFilter(QObject *parent = nullptr) : QObject(parent) {}
    
protected:
    bool eventFilter(QObject *watched, QEvent *event) override
    {
        if (event->type() == QEvent::KeyPress) {
            QKeyEvent *keyEvent = static_cast<QKeyEvent*>(event);
            
            // 只允许数字
            if (!keyEvent->text().isEmpty()) {
                QChar ch = keyEvent->text().at(0);
                if (!ch.isDigit() && keyEvent->key() != Qt::Key_Backspace) {
                    return true;  // 拦截事件
                }
            }
        }
        
        return false;
    }
};

// 安装到输入框
QLineEdit *edit = new QLineEdit(this);
InputFilter *filter = new InputFilter(edit);
edit->installEventFilter(filter);
`

## 注意事项

1. 事件过滤器可以拦截任何对象的事件
2. 返回true表示事件已处理，不再传播
3. 事件过滤器可以修改事件或转发事件
4. 注意事件过滤器的性能影响
