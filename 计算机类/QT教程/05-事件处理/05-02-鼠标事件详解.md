# 05-02 鼠标事件详解

## 概念解释

### 鼠标事件类型

| 事件 | 说明 |
|------|------|
| mousePressEvent | 鼠标按下 |
| mouseReleaseEvent | 鼠标释放 |
| mouseMoveEvent | 鼠标移动 |
| mouseDoubleClickEvent | 双击 |
| wheelEvent | 滚轮事件 |

### 鼠标按钮

| 按钮 | 说明 |
|------|------|
| Qt::LeftButton | 左键 |
| Qt::RightButton | 右键 |
| Qt::MiddleButton | 中键 |
| Qt::BackButton | 后退按钮 |
| Qt::ForwardButton | 前进按钮 |

### 修饰键

| 修饰键 | 说明 |
|--------|------|
| Qt::ShiftModifier | Shift键 |
| Qt::ControlModifier | Ctrl键 |
| Qt::AltModifier | Alt键 |

## 代码示例

### 基本鼠标事件

`cpp
class MouseWidget : public QWidget
{
    Q_OBJECT
    
public:
    explicit MouseWidget(QWidget *parent = nullptr) : QWidget(parent)
    {
        setMouseTracking(true);
        setMinimumSize(400, 300);
    }
    
signals:
    void mouseClicked(const QPoint &pos);
    void mouseMoved(const QPoint &pos);
    
protected:
    void mousePressEvent(QMouseEvent *event) override
    {
        m_lastPos = event->pos();
        
        if (event->button() == Qt::LeftButton) {
            qDebug() << "左键按下：" << event->pos();
            emit mouseClicked(event->pos());
        } else if (event->button() == Qt::RightButton) {
            qDebug() << "右键按下：" << event->pos();
        }
    }
    
    void mouseMoveEvent(QMouseEvent *event) override
    {
        if (event->buttons() & Qt::LeftButton) {
            // 拖拽移动
            QPoint delta = event->pos() - m_lastPos;
            move(pos() + delta);
        }
        
        m_lastPos = event->pos();
        emit mouseMoved(event->pos());
    }
    
    void mouseDoubleClickEvent(QMouseEvent *event) override
    {
        qDebug() << "双击：" << event->pos();
    }
    
    void wheelEvent(QWheelEvent *event) override
    {
        int delta = event->angleDelta().y();
        if (delta > 0) {
            qDebug() << "向上滚动";
        } else {
            qDebug() << "向下滚动";
        }
    }
    
    void paintEvent(QPaintEvent *) override
    {
        QPainter painter(this);
        painter.setRenderHint(QPainter::Antialiasing);
        
        // 绘制鼠标位置
        painter.setBrush(Qt::blue);
        painter.drawEllipse(m_lastPos, 5, 5);
    }
    
private:
    QPoint m_lastPos;
};
`

### 鼠标拖拽实现

`cpp
class DragWidget : public QWidget
{
    Q_OBJECT
    
public:
    explicit DragWidget(QWidget *parent = nullptr) : QWidget(parent)
    {
        setMouseTracking(true);
    }
    
protected:
    void mousePressEvent(QMouseEvent *event) override
    {
        if (event->button() == Qt::LeftButton) {
            m_dragging = true;
            m_dragStartPosition = event->pos();
        }
    }
    
    void mouseMoveEvent(QMouseEvent *event) override
    {
        if (m_dragging && (event->buttons() & Qt::LeftButton)) {
            // 计算移动距离
            int distance = (event->pos() - m_dragStartPosition).manhattanLength();
            if (distance > QApplication::startDragDistance()) {
                // 开始拖拽
                QDrag *drag = new QDrag(this);
                QMimeData *mimeData = new QMimeData;
                mimeData->setText("拖拽的数据");
                drag->setMimeData(mimeData);
                
                Qt::DropAction dropAction = drag->exec(Qt::MoveAction);
            }
        }
    }
    
    void mouseReleaseEvent(QMouseEvent *event) override
    {
        m_dragging = false;
        QWidget::mouseReleaseEvent(event);
    }
    
private:
    bool m_dragging = false;
    QPoint m_dragStartPosition;
};
`

## 注意事项

1. 使用setMouseTracking启用鼠标跟踪
2. 检查event->buttons()获取当前按下的按钮
3. 使用event->pos()获取鼠标位置
4. 拖拽需要使用QDrag类
