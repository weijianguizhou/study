## 概念解释

### 自定义控件的方法

1. **组合现有控件**：将多个控件组合成新控件
2. **继承现有控件**：扩展已有控件的功能
3. **完全自定义**：从QWidget派生，完全自绘

### 自定义控件的步骤

1. 创建头文件，声明类
2. 实现构造函数和功能
3. 使用Q_OBJECT宏
4. 在设计器中使用（可选）

## 代码示例

### 组合控件示例：搜索框

`cpp
// searchwidget.h
#ifndef SEARCHWIDGET_H
#define SEARCHWIDGET_H

#include <QWidget>
#include <QLineEdit>
#include <QPushButton>
#include <QHBoxLayout>

class SearchWidget : public QWidget
{
    Q_OBJECT
    
public:
    explicit SearchWidget(QWidget *parent = nullptr);
    
    QString text() const { return m_lineEdit->text(); }
    void setText(const QString &text) { m_lineEdit->setText(text); }
    void setPlaceholderText(const QString &text) { 
        m_lineEdit->setPlaceholderText(text); 
    }
    
signals:
    void searchRequested(const QString &query);
    void clearRequested();
    
private slots:
    void onSearchClicked();
    void onClearClicked();
    
private:
    QLineEdit *m_lineEdit;
    QPushButton *m_searchBtn;
    QPushButton *m_clearBtn;
};

#endif // SEARCHWIDGET_H
`

`cpp
// searchwidget.cpp
#include "searchwidget.h"

SearchWidget::SearchWidget(QWidget *parent)
    : QWidget(parent)
{
    QHBoxLayout *layout = new QHBoxLayout(this);
    layout->setContentsMargins(0, 0, 0, 0);
    
    m_lineEdit = new QLineEdit(this);
    m_searchBtn = new QPushButton("搜索", this);
    m_clearBtn = new QPushButton("清除", this);
    
    layout->addWidget(m_lineEdit);
    layout->addWidget(m_searchBtn);
    layout->addWidget(m_clearBtn);
    
    connect(m_searchBtn, &QPushButton::clicked, 
            this, &SearchWidget::onSearchClicked);
    connect(m_clearBtn, &QPushButton::clicked, 
            this, &SearchWidget::onClearClicked);
    connect(m_lineEdit, &QLineEdit::returnPressed, 
            this, &SearchWidget::onSearchClicked);
}

void SearchWidget::onSearchClicked()
{
    QString query = m_lineEdit->text().trimmed();
    if (!query.isEmpty()) {
        emit searchRequested(query);
    }
}

void SearchWidget::onClearClicked()
{
    m_lineEdit->clear();
    emit clearRequested();
}
`

### 完全自定义控件：颜色选择器

`cpp
// colorpicker.h
#ifndef COLORPICKER_H
#define COLORPICKER_H

#include <QWidget>
#include <QColor>

class ColorPicker : public QWidget
{
    Q_OBJECT
    
public:
    explicit ColorPicker(QWidget *parent = nullptr);
    
    QColor color() const { return m_color; }
    void setColor(const QColor &color);
    
signals:
    void colorChanged(const QColor &color);
    
protected:
    void paintEvent(QPaintEvent *event) override;
    void mousePressEvent(QMouseEvent *event) override;
    
private:
    QColor m_color;
};

#endif // COLORPICKER_H
`

`cpp
// colorpicker.cpp
#include "colorpicker.h"
#include <QPainter>
#include <QMouseEvent>

ColorPicker::ColorPicker(QWidget *parent)
    : QWidget(parent)
{
    setFixedSize(100, 100);
    setCursor(Qt::PointingHandCursor);
}

void ColorPicker::setColor(const QColor &color)
{
    if (m_color != color) {
        m_color = color;
        update();
        emit colorChanged(color);
    }
}

void ColorPicker::paintEvent(QPaintEvent *)
{
    QPainter painter(this);
    painter.setRenderHint(QPainter::Antialiasing);
    
    // 绘制颜色方块
    painter.setBrush(m_color);
    painter.setPen(Qt::gray);
    painter.drawRoundedRect(5, 5, width()-10, height()-10, 8, 8);
}

void ColorPicker::mousePressEvent(QMouseEvent *event)
{
    if (event->button() == Qt::LeftButton) {
        // 简单的颜色循环
        QColor colors[] = {Qt::red, Qt::green, Qt::blue, Qt::yellow, Qt::cyan};
        int index = (m_color.rgb() >> 16) & 0xFF;
        int next = (index + 1) % 5;
        setColor(colors[next]);
    }
}
`

## 注意事项

1. 自定义控件需要使用Q_OBJECT宏
2. 重写paintEvent进行自定义绘制
3. 考虑控件的大小策略
4. 提供信号供外部使用
