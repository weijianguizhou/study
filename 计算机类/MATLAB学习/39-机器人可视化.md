# 一、2D动画

```matlab
function animate_2R(q_traj, L1, L2)
    figure; hold on; axis equal; grid on;
    xlim([-1, 1]); ylim([-1, 1]);
    
    h = plot([0,0,0], [0,0,0], 'o-', 'LineWidth', 4, 'MarkerSize', 10);
    
    for k = 1:10:size(q_traj, 2)
        q1 = q_traj(1,k); q2 = q_traj(2,k);
        p1 = L1 * [cos(q1); sin(q1)];
        p2 = p1 + L2 * [cos(q1+q2); sin(q1+q2)];
        
        h.XData = [0, p1(1), p2(1)];
        h.YData = [0, p1(2), p2(2)];
        drawnow;
        pause(0.01);
    end
end
```

# 二、3D动画（用plot3）

```matlab
function animate_3D(q_traj, DH_table)
    figure; hold on; grid on; view(3);
    axis([-1 1 -1 1 0 1.5]);
    
    for k = 1:20:size(q_traj,2)
        T = eye(4); pos = [0;0;0];
        for i = 1:size(DH_table,1)
            d = DH_table(i,:);
            T = T * DH_transform(d(1)+q_traj(i,k), d(2), d(3), d(4));
            pos = [pos, T(1:3,4)];
        end
        cla; plot3(pos(1,:), pos(2,:), pos(3,:), 'o-', 'LineWidth', 3);
        drawnow; pause(0.02);
    end
end
```
