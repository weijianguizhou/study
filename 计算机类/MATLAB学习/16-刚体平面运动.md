# 16-刚体平面运动

# 一、牛顿-欧拉方程（平面）

$$m\ddot{\mathbf{r}}_G = \sum \mathbf{F}, \quad I_G \ddot{\theta} = \sum M_G$$

```matlab
m = 1; I = 0.1;               % 质量、转动惯量
g = 9.81;
F = [0; -m*g];                % 重力
M = 0;                        % 外力矩

odefun = @(t, y) [y(3); y(4);     % dx/dt, dy/dt
                  F(1)/m;         % d²x/dt²
                  F(2)/m;         % d²y/dt²
                  y(6);           % dθ/dt
                  M/I];           % d²θ/dt²

[t, y] = ode45(odefun, [0, 10], [0; 10; 5; 0; 0; 2]);
% 初值: x=0, y=10, vx=5, vy=0, θ=0, ω=2
```

# 二、受外力矩的旋转

```matlab
% 恒定力矩下的旋转
F = [0; 0];                    % 无力（纯力矩）
M = 0.5;                       % 恒定力矩

odefun = @(t, y) [y(3:4); F/m; y(6); M/I];
[t, y] = ode45(odefun, [0, 5], [0; 0; 0; 0; 0; 0]);

fprintf('t=5: 角度=%.2f, 角速度=%.2f\n', y(end,5), y(end,6));
```

# 三、物理摆（单摆）

$$I\ddot{\theta} = -mgL\sin\theta$$

```matlab
m = 1; L = 1; I = m*L^2; g = 9.81;

odefun = @(t, y) [y(2); -(m*g*L/I) * sin(y(1))];
[t, y] = ode45(odefun, [0, 10], [pi/3; 0]);   % 初始角度60°

plot(t, y(:,1)); title('单摆');
```

# 四、双摆（混沌系统）

```matlab
m1 = 1; m2 = 1; L1 = 1; L2 = 1; g = 9.81;

odefun = @(t, y) double_pendulum(t, y, m1, m2, L1, L2, g);
[t, y] = ode45(odefun, [0, 20], [pi/2; 0; pi/3; 0]);

plot(t, y(:,[1,3])); legend('\theta_1', '\theta_2');

function dydt = double_pendulum(~, y, m1, m2, L1, L2, g)
    th1 = y(1); th2 = y(2); dth1 = y(3); dth2 = y(4);
    M = [(m1+m2)*L1^2, m2*L1*L2*cos(th1-th2);
         m2*L1*L2*cos(th1-th2), m2*L2^2];
    C = [0, m2*L1*L2*sin(th1-th2)*dth2;
         -m2*L1*L2*sin(th1-th2)*dth1, 0];
    G = [(m1+m2)*g*L1*sin(th1); m2*g*L2*sin(th2)];
    ddq = M \ (-C * [dth1; dth2] - G);
    dydt = [dth1; dth2; ddq(1); ddq(2)];
end
```
