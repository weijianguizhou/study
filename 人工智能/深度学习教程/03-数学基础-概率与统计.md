## 1. 概率论基础

### 1.1 概率公理（Kolmogorov 公理）

1. **非负性**: $P(A) \geq 0$
2. **规范性**: $P(\Omega) = 1$
3. **可列可加性**: 对于互斥事件 $A_i \cap A_j = \emptyset$，$P(\cup_i A_i) = \sum_i P(A_i)$

```python
import numpy as np
import matplotlib.pyplot as plt

# 模拟概率频率
np.random.seed(42)
n_trials = 10000

# 抛硬币实验
coin_flips = np.random.binomial(n=1, p=0.5, size=n_trials)
cumulative_heads = np.cumsum(coin_flips)
trials = np.arange(1, n_trials + 1)
freq = cumulative_heads / trials

fig, ax = plt.subplots(figsize=(10, 4))
ax.plot(trials, freq, alpha=0.7, linewidth=1)
ax.axhline(y=0.5, color='red', linestyle='--', label='理论概率 0.5')
ax.set_xlabel('试验次数')
ax.set_ylabel('正面频率')
ax.set_title('大数定律：频率收敛于概率')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('law_of_large_numbers.png', dpi=100)
plt.show()
```

### 1.2 条件概率与贝叶斯定理

$$P(A|B) = \frac{P(B|A) P(A)}{P(B)}$$

```python
import numpy as np

def bayes_theorem(p_a, p_b_given_a, p_b_given_not_a):
    """
    贝叶斯定理: P(A|B) = P(B|A)*P(A) / P(B)
    P(B) = P(B|A)*P(A) + P(B|¬A)*P(¬A)
    """
    p_not_a = 1 - p_a
    p_b = p_b_given_a * p_a + p_b_given_not_a * p_not_a
    p_a_given_b = (p_b_given_a * p_a) / p_b
    return p_a_given_b, p_b


# 医学检测例子:
# 疾病患病率: 1%
# 检测灵敏度 P(+|患病) = 0.95
# 检测特异度 P(-|健康) = 0.90 → P(+|健康) = 0.10
p_disease = 0.01
p_positive_given_disease = 0.95
p_positive_given_no_disease = 0.10

p_disease_given_positive, p_positive = bayes_theorem(
    p_disease, p_positive_given_disease, p_positive_given_no_disease
)

print(f"患病率 P(患病) = {p_disease:.3f}")
print(f"检测阳性概率 P(+) = {p_positive:.4f}")
print(f"阳性时真正患病 P(患病|+) = {p_disease_given_positive:.4f}")
print(f"结论: 即使检测阳性，也只有约 {p_disease_given_positive*100:.1f}% 的概率真正患病")
print(f"      这就是贝叶斯定理的'逆概率'力量")

# 贝叶斯分类器在深度学习中的应用
# Softmax 输出可以理解为后验概率 P(class|input)
def softmax(logits):
    exp_logits = np.exp(logits - np.max(logits))
    return exp_logits / exp_logits.sum()

logits = np.array([2.0, 1.0, 0.1])
probs = softmax(logits)
print(f"\nSoftmax 输出 (后验概率): {probs}")
print(f"预测类别: {np.argmax(probs)}")
```

```python
import numpy as np
import matplotlib.pyplot as plt

np.random.seed(42)

def simulate_lln(distribution='normal', n_max=5000):
    """模拟大数定律"""
    if distribution == 'normal':
        samples = np.random.randn(n_max)  # μ=0, σ=1
        true_mean = 0
    elif distribution == 'exponential':
        samples = np.random.exponential(2.0, n_max)  # λ=1/2 → μ=2
        true_mean = 2
    elif distribution == 'cauchy':
        samples = np.random.standard_cauchy(n_max)

    fig, ax = plt.subplots(figsize=(12, 5))

        ax.axhline(y=true_mean, color='red', linestyle='--', linewidth=2,
                   label=f'理论均值 = {true_mean}')
    else:
        ax.text(n_max/2, cumulative_mean.max()*0.8, 'Cauchy: 不收敛!',
                fontsize=14, color='red')
    ax.set_xlabel('样本数 n')
    ax.set_ylabel('样本均值')
    ax.set_title(f'大数定律: {distribution} 分布')
    ax.legend()
    ax.grid(True, alpha=0.3)
```

```python
from scipy import stats

np.random.seed(42)

def demonstrate_clt(population_dist, n_samples=10000, sample_sizes=[1, 2, 5, 30]):
```

## 4. 最大似然估计 (MLE)

### 4.1 MLE 基本原理

给定数据 $\mathcal{D} = \{x_1, ..., x_n\}$ 和模型参数 $\theta$：

$$\hat{\theta}_{\text{MLE}} = \arg\max_\theta \prod_{i=1}^n P(x_i|\theta) = \arg\max_\theta \sum_{i=1}^n \log P(x_i|\theta)$$

```python
import numpy as np
from scipy.optimize import minimize

# 从真实分布生成数据
np.random.seed(42)
true_mu, true_sigma = 3.0, 2.0
n_samples = 1000
data = np.random.normal(true_mu, true_sigma, n_samples)

# 负对数似然函数
def neg_log_likelihood(params, data):
    mu, log_sigma = params  # 用 log_sigma 保证 σ > 0
    sigma = np.exp(log_sigma)
    n = len(data)
    # NLL = -sum log(N(x_i|mu, sigma))
    nll = 0.5 * n * np.log(2 * np.pi) + n * np.log(sigma) + \
          0.5 * np.sum(((data - mu) / sigma) ** 2)
    return nll

# 优化
result = minimize(neg_log_likelihood, x0=[0.0, 0.0], args=(data,),
                  method='L-BFGS-B')
mu_mle = result.x[0]
sigma_mle = np.exp(result.x[1])

print(f"真实参数: μ={true_mu}, σ={true_sigma}")
print(f"MLE估计: μ={mu_mle:.4f}, σ={sigma_mle:.4f}")

# 解析解验证
mu_analytical = np.mean(data)
sigma_analytical = np.std(data, ddof=0)  # MLE 使用有偏估计
print(f"解析MLE: μ={mu_analytical:.4f}, σ={sigma_analytical:.4f}")
```

## 5. 信息论基础

### 5.1 熵、交叉熵与 KL 散度

**熵（Entropy）** — 随机变量的不确定性度量：

$$H(P) = -\sum_{x} P(x) \log P(x) = \mathbb{E}_P[-\log P(x)]$$

**交叉熵（Cross-Entropy）** — 用分布 $Q$ 来编码分布 $P$ 所需的平均编码长度：

$$H(P, Q) = -\sum_{x} P(x) \log Q(x)$$

**KL 散度（Kullback-Leibler Divergence）** — $P$ 和 $Q$ 之间的差异：

$$D_{KL}(P \| Q) = \sum_{x} P(x) \log\frac{P(x)}{Q(x)} = H(P, Q) - H(P)$$

```python
import numpy as np
import matplotlib.pyplot as plt

def entropy(p):
    """计算离散分布的熵"""
    p = np.asarray(p)
    p = p[p > 0]  # 避免 log(0)
    return -np.sum(p * np.log2(p))

def cross_entropy(p, q):
    """计算交叉熵 H(P, Q)"""
    p, q = np.asarray(p), np.asarray(q)
    mask = p > 0
    return -np.sum(p[mask] * np.log2(q[mask]))

def kl_divergence(p, q):
    """计算 KL 散度 D_KL(P||Q)"""
    p, q = np.asarray(p), np.asarray(q)
    mask = p > 0
    return np.sum(p[mask] * np.log2(p[mask] / q[mask]))


# 示例: 二项分布
p_vals = np.linspace(0.01, 0.99, 99)
entropies = []
for p in p_vals:
    entropies.append(entropy([p, 1-p]))

fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(p_vals, entropies, 'b-', linewidth=2)
ax.axvline(x=0.5, color='red', linestyle='--', label='最大熵 p=0.5')
ax.set_xlabel('p')
ax.set_ylabel('H (bits)')
ax.set_title('二元分布的熵')
ax.legend()
ax.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('entropy_binary.png', dpi=100)
plt.show()

# 计算具体值
P = [0.7, 0.2, 0.1]
Q = [0.3, 0.3, 0.4]

print(f"熵 H(P) = {entropy(P):.4f} bits")
print(f"交叉熵 H(P,Q) = {cross_entropy(P, Q):.4f} bits")
print(f"KL散度 D_KL(P||Q) = {kl_divergence(P, Q):.4f} bits")
print(f"验证: H(P,Q) = H(P) + D_KL(P||Q): {cross_entropy(P, Q):.4f} = {entropy(P) + kl_divergence(P, Q):.4f}")
```

### 5.2 互信息

$$I(X; Y) = H(X) - H(X|Y) = D_{KL}(P(X,Y) \| P(X)P(Y))$$

```python
import numpy as np

def mutual_information(joint_prob):
    """从联合分布计算互信息"""
    joint = np.asarray(joint_prob)
    p_x = joint.sum(axis=1)
    p_y = joint.sum(axis=0)

    mi = 0
    for i in range(joint.shape[0]):
        for j in range(joint.shape[1]):
            if joint[i, j] > 0:
                mi += joint[i, j] * np.log2(joint[i, j] / (p_x[i] * p_y[j]))
    return mi


# 完全独立
joint_independent = np.array([
    [0.25, 0.25],
    [0.25, 0.25]
])
print(f"独立分布 MI: {mutual_information(joint_independent):.4f}")

# 完全依赖
joint_dependent = np.array([
    [0.4, 0.0],
    [0.0, 0.6]
])
print(f"依赖分布 MI: {mutual_information(joint_dependent):.4f}")

# 部分相关
joint_partial = np.array([
    [0.3, 0.1],
    [0.1, 0.5]
])
print(f"部分相关 MI: {mutual_information(joint_partial):.4f}")
```

```python
logits = torch.tensor([[2.0, 0.5, -1.0],
                       [0.5, 2.0, 0.3],
                       [0.1, 0.2, 3.0]])
labels = torch.tensor([0, 1, 2])

ce_loss = F.cross_entropy(logits, labels)
print(f"交叉熵损失: {ce_loss.item():.4f}")

# 手动计算验证
probs = F.softmax(logits, dim=1)
manual_ce = 0
for i in range(len(labels)):
    manual_ce += -torch.log(probs[i, labels[i]])
manual_ce /= len(labels)
print(f"手动计算 CE: {manual_ce.item():.4f}")

# 2. KL 散度用于知识蒸馏
teacher_logits = torch.tensor([[3.0, 0.5, 0.1]])
student_logits = torch.tensor([[2.0, 0.8, 0.3]])

teacher_probs = F.softmax(teacher_logits / 2.0, dim=1)  # T=2
student_probs = F.log_softmax(student_logits / 2.0, dim=1)

kl_loss = F.kl_div(student_probs, teacher_probs, reduction='batchmean')
print(f"\nKL 散度 (知识蒸馏): {kl_loss.item():.4f}")

# 3. 熵正则化
probs_uniform = F.softmax(torch.randn(3, 5), dim=1)
entropy_val = -(probs_uniform * torch.log(probs_uniform + 1e-10)).sum(dim=1).mean()
print(f"\n预测分布的熵: {entropy_val.item():.4f}")
```

## 6. 蒙特卡洛方法

### 6.1 蒙特卡洛积分

$$I = \int_\Omega f(x) dx = \mathbb{E}_{x \sim U(\Omega)}[f(x)] \cdot |\Omega| \approx \frac{|\Omega|}{N}\sum_{i=1}^N f(x_i)$$

```python
import numpy as np

def monte_carlo_integral(f, a, b, n=100000):
    """蒙特卡洛积分 ∫_a^b f(x) dx"""
    x = np.random.uniform(a, b, n)
    fx = f(x)
    estimate = (b - a) * np.mean(fx)
    std = (b - a) * np.std(fx) / np.sqrt(n)
    return estimate, std


# 示例: ∫_0^π sin(x) dx = 2
f = lambda x: np.sin(x)
estimate, std = monte_carlo_integral(f, 0, np.pi, n=100000)
print(f"蒙特卡洛积分 ∫_0^π sin(x)dx:")
print(f"  估计值: {estimate:.6f}")
print(f"  真值:   2.000000")
print(f"  标准差: {std:.6f}")

# 重要性采样
def importance_sampling(f, p, q, sampler_q, n=10000):
    """
    E_p[f(x)] = E_q[f(x) * p(x)/q(x)]
    其中 p 是目标分布，q 是提议分布
    """
    samples = sampler_q(n)
    weights = p(samples) / q(samples)
    estimate = np.mean(f(samples) * weights)
    return estimate


# 计算标准正态的 E[x^4] = 3 (峰度)
p = lambda x: np.exp(-x**2 / 2) / np.sqrt(2 * np.pi)  # N(0,1)
q = lambda x: np.exp(-np.abs(x)) / 2                    # Laplace(0,1)
sampler_q = lambda n: np.random.laplace(0, 1, n)

estimate_is = importance_sampling(
    lambda x: x**4, p, q, sampler_q, n=100000
)
print(f"\n重要性采样 E[x^4] (N(0,1)): {estimate_is:.4f} (真值=3)")
```

### 6.2 蒙特卡洛 Dropout

```python
import torch
import torch.nn as nn

class MCDropoutModel(nn.Module):
    """使用 MC Dropout 进行不确定性估计"""

    def __init__(self, input_dim, hidden_dim, dropout_rate=0.5):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(hidden_dim, 1),
        )

    def forward(self, x):
        return self.net(x)

    def mc_predict(self, x, n_samples=50):
        """MC Dropout 预测，返回均值和方差"""
        self.train()  # 保持 dropout 开启
        preds = []
        for _ in range(n_samples):
            with torch.no_grad():
                preds.append(self.forward(x))
        preds = torch.stack(preds)  # (n_samples, batch, 1)
        mean = preds.mean(dim=0)
        variance = preds.var(dim=0)
        return mean, variance


model = MCDropoutModel(input_dim=10, hidden_dim=128)

# 生成数据
x = torch.randn(100, 10)

# MC Dropout 预测
mean, variance = model.mc_predict(x, n_samples=30)
print(f"MC Dropout 预测均值范围: [{mean.min().item():.4f}, {mean.max().item():.4f}]")
print(f"预测方差范围: [{variance.min().item():.6f}, {variance.max().item():.6f}]")
print(f"(不确定性估计 — 模型对不熟悉区域的输入会产生更大方差)")
```
