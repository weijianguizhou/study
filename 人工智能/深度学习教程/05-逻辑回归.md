# 逻辑回归 — 分类问题的基石

逻辑回归虽然名为"回归"，却是最重要的二分类模型之一。从 Sigmoid 函数的概率解释到交叉熵损失的完整推导，再到 Softmax 多分类扩展，它是理解神经网络分类器的必经之路。

---

## 1. Sigmoid 函数与概率解释

### 1.1 从线性回归到分类

线性回归输出的是连续值 $\hat{y} = \mathbf{w}^T\mathbf{x} + b$，不适合直接用于分类。Sigmoid 函数将任意实数映射到 $(0, 1)$：

$$\sigma(z) = \frac{1}{1 + e^{-z}}$$

将线性输出 $z = \mathbf{w}^T\mathbf{x} + b$ 输入 Sigmoid，得到类别概率：

$$P(y=1|\mathbf{x}; \mathbf{w}) = \sigma(\mathbf{w}^T\mathbf{x} + b)$$


## 2. 交叉熵损失的完整推导

### 2.1 最大似然估计视角

假设标签 $y \in \{0, 1\}$ 服从 Bernoulli 分布：

$$P(y|\mathbf{x}; \mathbf{w}) = \hat{y}^y (1 - \hat{y})^{1-y}$$

其中 $\hat{y} = \sigma(\mathbf{w}^T\mathbf{x} + b)$。

**似然函数**：

$$L(\mathbf{w}) = \prod_{i=1}^n \hat{y}_i^{y_i} (1 - \hat{y}_i)^{1-y_i}$$

**负对数似然（即交叉熵损失）**：

$$\mathcal{L}(\mathbf{w}) = -\frac{1}{n}\sum_{i=1}^n \left[y_i \log \hat{y}_i + (1-y_i) \log(1-\hat{y}_i)\right]$$

### 2.2 梯度推导

$$\frac{\partial \mathcal{L}}{\partial \hat{y}_i} = -\frac{1}{n}\left[\frac{y_i}{\hat{y}_i} - \frac{1-y_i}{1-\hat{y}_i}\right] = \frac{1}{n}\frac{\hat{y}_i - y_i}{\hat{y}_i(1-\hat{y}_i)}$$

由链式法则 $\frac{\partial \mathcal{L}}{\partial z_i} = \frac{\partial \mathcal{L}}{\partial \hat{y}_i} \cdot \frac{\partial \hat{y}_i}{\partial z_i}$ 和 $\sigma'(z) = \sigma(z)(1-\sigma(z))$：

$$\frac{\partial \mathcal{L}}{\partial z_i} = \frac{1}{n}(\hat{y}_i - y_i)$$

最终：

$$\nabla_{\mathbf{w}} \mathcal{L} = \frac{1}{n}\mathbf{X}^T(\hat{\mathbf{y}} - \mathbf{y})$$

**这个结果的优雅之处：与线性回归的梯度形式完全相同！**

```python
import numpy as np

def binary_cross_entropy(y_true, y_pred):
    """交叉熵损失"""
    eps = 1e-15
    y_pred = np.clip(y_pred, eps, 1 - eps)
    return -np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))

def bce_gradient_verification():
    """验证梯度公式 ∂L/∂z = (ŷ - y)/n"""
    np.random.seed(42)
    n = 10

    # 随机数据
    z = np.random.randn(n)
    y = np.random.randint(0, 2, n).astype(np.float64)
    y_pred = sigmoid(z)

    # 解析梯度
    ana_grad = (y_pred - y) / n

    # 数值梯度
    eps = 1e-7
    num_grad = np.zeros(n)
    for i in range(n):
        z_plus = z.copy()
        z_plus[i] += eps
        loss_plus = binary_cross_entropy(y, sigmoid(z_plus))

        z_minus = z.copy()
        z_minus[i] -= eps
        loss_minus = binary_cross_entropy(y, sigmoid(z_minus))

        num_grad[i] = (loss_plus - loss_minus) / (2 * eps)

    error = np.linalg.norm(ana_grad - num_grad) / max(np.linalg.norm(ana_grad), 1e-10)
    print(f"梯度验证: 相对误差 = {error:.2e}")
    print(f"解析梯度: {ana_grad}")
    print(f"数值梯度: {num_grad}")

bce_gradient_verification()
```

## 3. 从零实现逻辑回归

```python
import numpy as np

class LogisticRegression:
    """逻辑回归 — 从零全部实现"""

    def __init__(self, lr=0.01, n_iters=1000, batch_size=None,
                 penalty=None, C=1.0, fit_intercept=True):
        self.lr = lr
        self.n_iters = n_iters
        self.batch_size = batch_size
        self.penalty = penalty  # 'l1', 'l2', None
        self.C = C              # 正则化强度的倒数
        self.fit_intercept = fit_intercept
        self.loss_history = []

    def _sigmoid(self, z):
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))

    def _add_intercept(self, X):
        if self.fit_intercept:
            return np.c_[np.ones((X.shape[0], 1)), X]
        return X

    def _compute_loss(self, X, y, w):
        z = X @ w
        y_pred = self._sigmoid(z)
        eps = 1e-15
        y_pred = np.clip(y_pred, eps, 1 - eps)
        nll = -np.mean(y * np.log(y_pred) + (1 - y) * np.log(1 - y_pred))

        if self.penalty == 'l2':
            penalty = 0.5 * np.sum(w[1:] ** 2) / self.C if self.fit_intercept else 0.5 * np.sum(w ** 2) / self.C
            nll += penalty
        elif self.penalty == 'l1':
            penalty = np.sum(np.abs(w[1:])) / self.C if self.fit_intercept else np.sum(np.abs(w)) / self.C
            nll += penalty

        return nll

    def _compute_gradient(self, X, y, w):
        z = X @ w
        y_pred = self._sigmoid(z)
        n = X.shape[0]
        grad = X.T @ (y_pred - y) / n

        if self.penalty == 'l2':
            if self.fit_intercept:
                reg_grad = w.copy()
                reg_grad[0] = 0
            else:
                reg_grad = w
            grad += reg_grad / self.C
        elif self.penalty == 'l1':
            if self.fit_intercept:
                reg_grad = np.sign(w)
                reg_grad[0] = 0
            else:
                reg_grad = np.sign(w)
            grad += reg_grad / self.C

        return grad

    def fit(self, X, y):
        n_samples, n_features = X.shape
        X = self._add_intercept(X)

        if self.batch_size is None:
            self.batch_size = n_samples

        self.w = np.random.randn(X.shape[1]) * 0.01

        for epoch in range(self.n_iters):
            # 打乱数据
            indices = np.random.permutation(n_samples)
            X_shuffled = X[indices]
            y_shuffled = y[indices]

            for start in range(0, n_samples, self.batch_size):
                end = start + self.batch_size
                X_batch = X_shuffled[start:end]
                y_batch = y_shuffled[start:end]

                grad = self._compute_gradient(X_batch, y_batch, self.w)
                self.w -= self.lr * grad

            loss = self._compute_loss(X, y, self.w)
            self.loss_history.append(loss)

            if epoch % 100 == 0 and epoch > 0:
                if abs(self.loss_history[-1] - self.loss_history[-101]) < 1e-6:
                    break

        if self.fit_intercept:
            self.intercept_ = self.w[0]
            self.coef_ = self.w[1:]
        else:
            self.intercept_ = 0.0
            self.coef_ = self.w

        return self

    def predict_proba(self, X):
        X = self._add_intercept(X)
        return self._sigmoid(X @ self.w)

    def predict(self, X, threshold=0.5):
        return (self.predict_proba(X) >= threshold).astype(int)


# 二分类测试
np.random.seed(42)
n_samples = 500

# 两类高斯数据
X0 = np.random.randn(n_samples // 2, 2) + np.array([-2, -2])
X1 = np.random.randn(n_samples // 2, 2) + np.array([2, 2])
X = np.vstack([X0, X1])
y = np.hstack([np.zeros(n_samples // 2), np.ones(n_samples // 2)])

# 打乱
indices = np.random.permutation(n_samples)
X, y = X[indices], y[indices]

# 训练
model = LogisticRegression(lr=0.1, n_iters=500, batch_size=32, fit_intercept=True)
model.fit(X, y)

y_pred = model.predict(X)
accuracy = np.mean(y_pred == y)
print(f"训练准确率: {accuracy:.4f}")
print(f"学到的权重: w={model.coef_}, b={model.intercept_:.4f}")

# 可视化决策边界
import matplotlib.pyplot as plt

fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# 决策边界
xx, yy = np.meshgrid(np.linspace(-5, 5, 200), np.linspace(-5, 5, 200))
X_grid = np.c_[xx.ravel(), yy.ravel()]
Z = model.predict_proba(X_grid).reshape(xx.shape)

axes[0].contourf(xx, yy, Z, levels=20, cmap='RdBu', alpha=0.6)
axes[0].contour(xx, yy, Z, levels=[0.5], colors='k', linewidths=2)
axes[0].scatter(X[y==0, 0], X[y==0, 1], c='blue', alpha=0.5, s=20, label='y=0')
axes[0].scatter(X[y==1, 0], X[y==1, 1], c='red', alpha=0.5, s=20, label='y=1')
axes[0].set_xlabel('x1')
axes[0].set_ylabel('x2')
axes[0].set_title('逻辑回归决策边界')
axes[0].legend()

# 损失曲线
axes[1].plot(model.loss_history)
axes[1].set_xlabel('迭代次数')
axes[1].set_ylabel('损失')
axes[1].set_title('训练损失')
axes[1].grid(True, alpha=0.3)

plt.tight_layout()

plt.show()
```

## 4. Softmax 多分类

### 4.1 Softmax 函数

对于 $K$ 类分类问题，将 logits $\mathbf{z} = (z_1, ..., z_K)$ 转换为概率分布：

$$\text{softmax}(z_i) = \frac{e^{z_i}}{\sum_{j=1}^{K} e^{z_j}}$$

```python
import numpy as np
import torch
import torch.nn.functional as F

def softmax_naive(z):
    """直接实现 — 数值不稳定"""
    exp_z = np.exp(z)
    return exp_z / exp_z.sum()

def softmax_stable(z):
    """数值稳定实现 — 减去最大值"""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

# 数值稳定性对比
z_large = np.array([1000, 1000, 1000])
try:
    result_naive = softmax_naive(z_large)
    print(f"Naive: {result_naive}")
except:
    print("Naive Softmax: 溢出!")

result_stable = softmax_stable(z_large)
print(f"Stable: {result_stable}")  # [0.333, 0.333, 0.333]

# 多种 Softmax 实现
z = np.array([2.0, 1.0, 0.1])

print(f"\n输入 logits: {z}")
print(f"NumPy stable softmax: {softmax_stable(z)}")
print(f"PyTorch softmax:       {F.softmax(torch.tensor(z), dim=0).numpy()}")
print(f"Scipy softmax:         {np.exp(z) / np.exp(z).sum()}")

# Softmax 的温度参数
def softmax_with_temperature(z, T=1.0):
    """温度控制分布的平滑/尖锐程度"""
    z_scaled = z / T
    return softmax_stable(z_scaled)

z_example = np.array([3.0, 1.0, 0.2])
print(f"\n温度对 Softmax 的影响 (logits={z_example}):")
for T in [0.1, 0.5, 1.0, 2.0, 10.0]:
    probs = softmax_with_temperature(z_example, T)
    print(f"  T={T:.1f}: {probs}")
```

### 4.2 多分类交叉熵

$$\mathcal{L} = -\frac{1}{n}\sum_{i=1}^n \sum_{k=1}^K y_{ik} \log \hat{y}_{ik}$$

其中 $y_{ik}$ 是 one-hot 编码的真实标签，$\hat{y}_{ik} = \text{softmax}(\mathbf{z}_i)_k$。

```python
import torch
import torch.nn.functional as F
import numpy as np

# 3类分类问题
logits = torch.tensor([[2.0, 0.5, -0.5],
                       [0.5, 2.0, 0.3],
                       [0.1, 0.2, 3.0],
                       [1.0, 1.0, 1.0]])
labels = torch.tensor([0, 1, 2, 0])

# PyTorch CrossEntropyLoss
ce_loss = F.cross_entropy(logits, labels)
print(f"CrossEntropyLoss: {ce_loss.item():.6f}")

# 手动计算
def cross_entropy_manual(logits, labels):
    logits = logits.numpy()
    labels = labels.numpy()
    probs = softmax_stable(logits) if logits.ndim == 1 else \
            np.array([softmax_stable(l) for l in logits])

    n = len(labels)
    loss = 0
    for i in range(n):
        loss += -np.log(probs[i, labels[i]] + 1e-15)
    return loss / n

manual_ce = cross_entropy_manual(logits, labels)
print(f"手动 CE: {manual_ce:.6f}")

# 使用 NLLLoss + LogSoftmax (等价于 CrossEntropyLoss)
log_probs = F.log_softmax(logits, dim=1)
nll_loss = F.nll_loss(log_probs, labels)
print(f"NLLLoss + LogSoftmax: {nll_loss.item():.6f}")

# 标签平滑
def cross_entropy_with_label_smoothing(logits, labels, epsilon=0.1):
    n_classes = logits.shape[1]
    log_probs = F.log_softmax(logits, dim=1)

    # 创建平滑标签
    smooth_labels = torch.zeros_like(log_probs)
    smooth_labels.fill_(epsilon / (n_classes - 1))
    smooth_labels.scatter_(1, labels.unsqueeze(1), 1 - epsilon)

    return -(smooth_labels * log_probs).sum(dim=1).mean()

ce_smooth = cross_entropy_with_label_smoothing(logits, labels, epsilon=0.1)
print(f"CE with Label Smoothing (ε=0.1): {ce_smooth.item():.6f}")
```

## 5. 评价指标

### 5.1 混淆矩阵

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics import (
    confusion_matrix, accuracy_score, precision_score,
    recall_score, f1_score, roc_curve, auc,
    classification_report
)

def compute_metrics(y_true, y_pred, y_prob=None):
    """计算全面的分类评价指标"""

    # 混淆矩阵
    cm = confusion_matrix(y_true, y_pred)
    tn, fp, fn, tp = cm.ravel() if cm.shape == (2, 2) else (0, 0, 0, 0)

    metrics = {
        'Accuracy':  accuracy_score(y_true, y_pred),
        'Precision': precision_score(y_true, y_pred, zero_division=0),
        'Recall':    recall_score(y_true, y_pred, zero_division=0),
        'F1 Score':  f1_score(y_true, y_pred, zero_division=0),
        'TP': tp, 'TN': tn, 'FP': fp, 'FN': fn,
    }

    if cm.shape == (2, 2):
        metrics['Specificity'] = tn / (tn + fp) if (tn + fp) > 0 else 0
        metrics['NPV'] = tn / (tn + fn) if (tn + fn) > 0 else 0

    return metrics, cm
```

### 5.2 ROC 曲线与 AUC

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics import roc_curve, auc, roc_auc_score

# 生成不同质量的分类器预测
np.random.seed(42)
n = 2000
y_true_half = np.hstack([np.zeros(n // 2), np.ones(n // 2)])

# 不同质量的预测概率
classifiers = {
    '完美分类器': np.hstack([np.random.beta(1, 10, n//2), np.random.beta(10, 1, n//2)]),
    '好分类器': np.hstack([np.random.beta(2, 5, n//2), np.random.beta(5, 2, n//2)]),
    '一般分类器': np.hstack([np.random.beta(3, 3, n//2), np.random.beta(3, 3, n//2)]),
    '随机分类器': np.random.uniform(0, 1, n),
}

fig, ax = plt.subplots(figsize=(8, 7))

for name, y_prob in classifiers.items():
    fpr, tpr, thresholds = roc_curve(y_true_half, y_prob)
    roc_auc = auc(fpr, tpr)
    ax.plot(fpr, tpr, linewidth=2, label=f'{name} (AUC={roc_auc:.3f})')

# 随机分类器参考线
ax.plot([0, 1], [0, 1], 'k--', alpha=0.5, label='随机 (AUC=0.500)')

ax.set_xlabel('False Positive Rate')
ax.set_ylabel('True Positive Rate')
ax.set_title('ROC 曲线对比')
ax.legend(loc='lower right')
ax.grid(True, alpha=0.3)
ax.set_xlim([0, 1])
ax.set_ylim([0, 1])
ax.set_aspect('equal')

plt.tight_layout()
plt.savefig('roc_curves.png', dpi=100)
plt.show()

# 详细分析最优阈值
y_prob_good = classifiers['好分类器']
fpr, tpr, thresholds = roc_curve(y_true_half, y_prob_good)

# 计算 Youden 指数找最优阈值
youden_index = tpr - fpr
optimal_idx = np.argmax(youden_index)
optimal_threshold = thresholds[optimal_idx]

print(f"\n最优阈值 (Youden指数): {optimal_threshold:.4f}")
print(f"  此时 TPR={tpr[optimal_idx]:.4f}, FPR={fpr[optimal_idx]:.4f}")

# 使用 sklearn 计算 AUC
auc_score = roc_auc_score(y_true_half, y_prob_good)
print(f"\nAUC 分数: {auc_score:.4f}")
print(f"AUC 解释: 随机选一个正样本和一个负样本，正样本得分 > 负样本得分的概率")
```

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import classification_report, roc_auc_score

# 模拟 Titanic 数据集（在线不可用时使用）
np.random.seed(42)
n_passengers = 891

# 生成模拟数据
data = pd.DataFrame({
    'Pclass': np.random.choice([1, 2, 3], n_passengers, p=[0.24, 0.21, 0.55]),
    'Sex': np.random.choice(['male', 'female'], n_passengers),
    'Age': np.clip(np.random.normal(30, 14, n_passengers), 0.5, 80).astype(int),
    'SibSp': np.random.poisson(0.5, n_passengers),
    'Parch': np.random.poisson(0.4, n_passengers),
    'Fare': np.clip(np.random.exponential(50, n_passengers), 0, 512).astype(int),
    'Embarked': np.random.choice(['S', 'C', 'Q'], n_passengers, p=[0.72, 0.19, 0.09]),
})

# 生存标签（模拟真实规律：女性、高等级、低年龄更容易生存）
logit = (
    -1.0
    + 2.0 * (data['Sex'] == 'female').astype(float)
    - 0.8 * (data['Pclass'] == 3).astype(float)
    - 0.5 * (data['Pclass'] == 2).astype(float)
    - 0.02 * data['Age']
    + 0.5 * (data['Parch'] > 0).astype(float)
)
survived_prob = 1 / (1 + np.exp(-logit))
data['Survived'] = (np.random.random(n_passengers) < survived_prob).astype(int)

print("数据预览:")
print(data.head())
print(f"\n生存率: {data['Survived'].mean():.3f}")

# ==================== 特征工程与预处理 ====================

def preprocess_titanic(df, is_train=True, fitted_encoders=None):
    df = df.copy()

    # 缺失值填充
    df['Age'] = df['Age'].fillna(df['Age'].median())
    df['Fare'] = df['Fare'].fillna(df['Fare'].median())

    # 性别编码
    df['Sex'] = df['Sex'].map({'male': 0, 'female': 1})

    # 登船港口 one-hot
    df = pd.get_dummies(df, columns=['Embarked'], prefix='Emb')

    # 家庭大小特征
    df['FamilySize'] = df['SibSp'] + df['Parch'] + 1
    df['IsAlone'] = (df['FamilySize'] == 1).astype(int)

    # 年龄分桶
    df['AgeBucket'] = pd.cut(df['Age'], bins=[0, 12, 18, 35, 50, 100],
                              labels=[0, 1, 2, 3, 4]).astype(float)

    # Fare 分桶
    df['FareBucket'] = pd.qcut(df['Fare'].clip(0, 300), q=4, labels=False)

    # 选择特征
    feature_cols = ['Pclass', 'Sex', 'Age', 'Fare', 'SibSp', 'Parch',
                    'FamilySize', 'IsAlone', 'AgeBucket', 'FareBucket']

    # 确保所有 one-hot 列存在
    for col in df.columns:
        if col.startswith('Emb_'):
            if col not in feature_cols:
                feature_cols.append(col)

    X = df[feature_cols].values.astype(np.float64)
    y = df['Survived'].values if 'Survived' in df.columns else None

    if is_train:
        scaler = StandardScaler()
        X = scaler.fit_transform(X)
        return X, y, scaler, feature_cols
    else:
        return fitted_encoders.transform(X)

# ==================== 训练 ====================

# 预处理
X_processed, y_processed, scaler, feature_cols = preprocess_titanic(data)

# 划分数据集
X_train, X_test, y_train, y_test = train_test_split(
    X_processed, y_processed, test_size=0.2, random_state=42, stratify=y_processed
)

print(f"\n训练集大小: {len(X_train)}, 测试集大小: {len(X_test)}")

# 训练逻辑回归
model = LogisticRegression(lr=0.05, n_iters=1000, batch_size=32,

# 评估
y_pred_train = model.predict(X_train)
y_pred_test = model.predict(X_test)
y_prob_test = model.predict_proba(X_test)

train_acc = accuracy_score(y_train, y_pred_train)
test_acc = accuracy_score(y_test, y_pred_test)
test_auc = roc_auc_score(y_test, y_prob_test)

print(f"\n训练准确率: {train_acc:.4f}")
print(f"测试准确率: {test_acc:.4f}")
print(f"测试 AUC:   {test_auc:.4f}")

print("\n分类报告 (测试集):")
print(classification_report(y_test, y_pred_test, target_names=['未生存', '生存']))

# ==================== 模型分析 ====================

# 特征重要性
feature_importance = np.abs(model.coef_)
sorted_idx = np.argsort(feature_importance)[::-1]

fig, axes = plt.subplots(2, 2, figsize=(14, 10))
```
