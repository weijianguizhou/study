\chapter{螺纹连接}
\label{chap:螺纹连接}

\section{螺纹}
\label{sec:螺纹}

\subsection{螺纹的形成}

\begin{definition}[螺纹的形成]
\label{def:螺纹形成}
将直角三角形 $K$ 绕在圆柱体上，斜边在圆柱表面形成的空间曲线即\textbf{螺旋线}；沿螺旋线加工的沟槽即\textbf{螺纹}。改变原始三角形 $K$ 的形状（牙底形状），可得到矩形、梯形、锯齿形、管螺纹等不同牙型。
\end{definition}

\subsection{螺纹的分类}

\begin{definition}[按牙型分类]
\label{def:螺纹按牙型分类}
\begin{itemize}
	\item \textbf{联接螺纹}：\textcolor{red}{三角形螺纹（$60^\circ$）}、锥螺纹（$55^\circ$）、管螺纹（$55^\circ$）；
	\item \textbf{传动螺纹}：矩形螺纹、梯形螺纹、锯齿形螺纹（工作侧牙型角 $3^\circ$）。
\end{itemize}

\textbf{粗牙与细牙}（同属三角形螺纹）：
\begin{itemize}
	\item \textbf{粗牙}：作一般紧固件使用；
	\item \textbf{细牙}：同一公称直径下\textcolor{red}{螺距最小}，牙细而浅，\textcolor{red}{自锁性好}，强度较高，适于\textcolor{red}{薄壁零件、细小零件、受冲击振动及变载荷}的场合。
\end{itemize}
\end{definition}

\begin{definition}[按位置、旋向与线数分类]
\label{def:螺纹其它分类}
\begin{itemize}
	\item 按位置：\textbf{内螺纹}、\textbf{外螺纹}；
	\item 按旋向：\textbf{右旋}（常用）、\textbf{左旋}（如\textcolor{red}{自行车脚踏曲柄上的螺纹为左旋}）；
	\item 按螺旋线头数（线数）：\textbf{单线}（$n=1$，用于\textbf{联接}）；\textbf{双线、多线}（$n\ge2$，用于\textbf{传动}，升角大、效率高，一般 $n\le4$）。
\end{itemize}
\end{definition}

\begin{theorem}[导程与螺距的关系]
\label{thm:导程}
\begin{equation}
	L=nP
	\label{eq:螺纹:导程}
\end{equation}
其中，$L$ 为导程，$P$ 为螺距，$n$ 为线数。
\end{theorem}

\end{tabular}

螺旋升角：
\begin{equation}
	\psi=\arctan\frac{L}{\pi d_2}=\arctan\frac{nP}{\pi d_2}
	\label{eq:螺纹:螺旋升角}
\end{equation}
\end{definition}

\begin{note}
螺纹强度计算一律使用\textcolor{red}{小径 $d_1$}（危险截面直径），不是大径 $d$。
\end{note}

\subsection{螺纹联接件}

\begin{definition}[螺栓、螺母与垫圈]
\label{def:联接件}
\begin{itemize}
	\item \textbf{螺栓}：六角头螺栓（标准）、双头螺柱、螺钉；
	\item \textbf{螺母}：六角螺母（标准/扁/厚）、圆螺母（与止退垫圈配合使用）；
	\item \textbf{垫圈}：作用——\textcolor{red}{保护被联接件表面、增大接触面积、遮掩不平表面}。普通垫圈（平垫圈、斜垫圈）；弹簧垫圈——防止螺母松脱；止动垫圈——配合圆螺母用。
\end{itemize}
\end{definition}

\section{螺纹连接的预紧和防松}
\label{sec:预紧和防松}

\subsection{预紧}

\begin{definition}[松联接与紧联接]
\label{def:松紧联接}
\begin{itemize}
	\item \textbf{松联接}：装配时不拧紧，只有受外载荷时才受力；
	\item \textbf{紧联接}：装配时需拧紧，即承载时已预先受力（\textbf{预紧力} $Q_P$）。
\end{itemize}
\end{definition}

\begin{definition}[预紧的目的]
\label{def:预紧目的}
预紧目的：\textcolor{red}{增大联接的刚度、紧密性，提高防松能力}。

预紧力 $Q_P$：预先施加的轴向作用力（拉力）。预紧过紧，螺杆静载荷增大，降低螺杆强度；预紧过松，工作不可靠。
\end{definition}

\begin{theorem}[扳手拧紧力矩]
\label{thm:拧紧力矩}
\begin{equation}
	T=F_H\cdot L=T_1+T_2
	\label{eq:预紧:拧紧力矩}
\end{equation}
其中，
\begin{itemize}
    \item $T_1$ 为螺纹摩擦阻力矩$T_1=Q_P\tan(\psi+\varphi_v)\dfrac{d_2}{2}$
    \item $T_2$ 为螺母端环形面与被联接件间的摩擦力矩 $T_2=f_cQ_P\cdot\dfrac{1}{3}\cdot\dfrac{D_1^3-d_0^3}{D_1^2-d_0^2}$
\end{itemize}


拧紧力矩系数
\begin{equation}
	K=\frac{1}{2}\left[\frac{d_2}{d}\tan(\psi+\varphi_v)+\frac{2f_c}{3d}\cdot\frac{D_1^3-d_0^3}{D_1^2-d_0^2}\right]
	\label{eq:预紧:力矩系数}
\end{equation}
\begin{equation}
	\therefore\quad T=KQ_P d,\qquad K=0.1\sim0.3
	\label{eq:预紧:拧紧力矩简化}
\end{equation}
\end{theorem}


\begin{definition}[预紧力 $Q_P$ 的控制方法]
\label{def:预紧力控制}
\begin{enumerate}
	\item \textbf{测力矩扳手}——测出预紧力矩；
	\item \textbf{定力矩扳手}——达到固定的拧紧力矩时卡盘不转动；
	\item \textbf{测量预紧前后螺栓伸长量}——精度较高。
\end{enumerate}
\end{definition}

\subsection{防松}

\begin{definition}[防松的目的与原理]
\label{def:防松}
在振动、冲击、变载或温度变化时，螺纹副中正压力可能减小或瞬间消失，摩擦力为零，使螺纹联接松动；反复作用会使联接松驰而失效，必须防松。

防松原理：\textcolor{red}{消除（或限制）螺纹副之间的相对运动，或增大相对运动的难度}。
\end{definition}

\begin{definition}[防松的方法]
\label{def:防松方法}
\begin{enumerate}
	\item \textbf{摩擦防松}：双螺母（对顶螺母）、弹簧垫圈、自锁螺母（嵌尼龙圈、带颈收口、加金属防松装置）等；
	\item \textbf{机械防松}：开槽螺母与开口销、止动垫片、串联钢丝（注意正确穿法）等；
	\item \textbf{永久防松}：端铆、冲点、点焊、粘合。
\end{enumerate}
\end{definition}

\section{单个螺栓连接的强度计算}
\label{sec:单个螺栓强度}

\subsection{受力形式、失效形式与设计准则}

\begin{definition}[螺栓受力与失效形式]
\label{def:螺栓失效形式}
\begin{itemize}
	\item \textbf{螺栓杆受力形式}：轴向拉力、扭转力矩或横向剪力。
	\item \textbf{失效形式}：
	\begin{itemize}
		\item 受拉螺栓：螺栓杆和螺纹可能发生\textcolor{red}{塑性变形或断裂}；
		\item 受剪螺栓：螺栓杆、孔壁可能发生\textcolor{red}{剪断或压溃}。
	\end{itemize}
\end{itemize}
统计：90\% 以上的螺栓属于\textcolor{red}{疲劳破坏}，失效原因主要是应力集中促使疲劳裂纹的发生和发展。疲劳破坏位置：（1）螺母支承面第一、第二牙圈处占 \textcolor{red}{65\%}；（2）螺杆尾退刀槽处占 20\%；（3）钉头支承面处占 15\%。
\end{definition}

  \centering

  \caption{螺栓失效实例}

\begin{definition}[设计准则与计算方法]
\label{def:螺栓设计准则}
\begin{itemize}
	\item 受拉螺栓——保证螺栓的\textcolor{red}{疲劳拉伸强度或静强度}；
	\item 受剪螺栓——保证螺栓的\textcolor{red}{挤压强度和剪切强度}。
\end{itemize}
计算方法：由强度条件确定螺纹小径 $d_1$，按标准选相应的公称直径 $d$ 及螺距 $P$，其余尺寸按标准选取。
\end{definition}

\subsection{松螺栓联接（如吊钩螺栓）}

工作前不拧紧，无预紧力 $Q_P$，只有工作载荷 $F$ 起拉伸作用：

\begin{theorem}[松螺栓联接的强度条件]
\label{thm:松螺栓}
\begin{equation}
	\sigma=\frac{F}{\dfrac{\pi}{4}d_1^2}\le[\sigma]\qquad\text{——验算用}
	\label{eq:连接:松螺栓验算}
\end{equation}
\begin{equation}
	d_1\ge\sqrt{\frac{4F}{\pi[\sigma]}}\qquad\text{——设计用}
	\label{eq:连接:松螺栓设计}
\end{equation}
其中，$[\sigma]=\sigma_s/n$，$\sigma_s$ 查表 4-4，安全系数 $n$ 查表 4-5。
\end{theorem}
\remark{其实就是轴向正应力计算}
\subsection{紧螺栓联接——仅受预紧力（横向载荷的普通螺栓）}

工作前拧紧，在预紧力和拧紧力矩联合作用下处于复合应力状态：

预紧力 $Q_P$ 产生拉伸应力 $\sigma=\dfrac{Q_P}{\pi d_1^2/4}$；

螺纹摩擦力矩 $T_1$ 产生剪应力 $\tau$。

\begin{theorem}[仅受预紧力的紧螺栓强度（简化计算法）]
强度条件：
\begin{equation}
\label{eq:紧螺栓仅预紧}
    d_1\ge\sqrt{\frac{1.3\times4Q_P}{\pi[\sigma]}}
\end{equation}

\end{theorem}

\begin{keybox}[系数 1.3 的含义]
系数 1.3 将外载荷提高 30\%，以考虑\textcolor{red}{螺纹扭转剪应力}对螺栓强度的影响，把拉扭复合应力状态简化为纯拉伸处理，故又称\textbf{简化计算法}。
\end{keybox}

\subsection{受横向载荷的紧螺栓联接}
受横向载荷的螺栓联接时，为了防止被联接件滑移，需要施加预紧力。


\subsection{普通螺栓联接}

靠拧紧正压力 $Q_P$ 产生摩擦力平衡外载荷：

\begin{theorem}[横向载荷普通螺栓的防滑条件]
\label{thm:横向普通螺栓}
不产生相对滑移条件：
\begin{equation}
	i\cdot f\cdot Q_P\ge K_sR\quad\Rightarrow\quad Q_P\ge\frac{K_sR}{f\cdot i}
	\label{eq:连接:横向防滑}
\end{equation}
\begin{itemize}
    \item $f$ 为接缝面摩擦系数；
    \item $i$ 为接合界面数目；
    \item $K_s$ 为防滑（可靠性）系数，$K_s=1.1\sim1.3$
\end{itemize}
\end{theorem}

为了增加可靠性,减小直径,简化结构,提高承载能力。可采用如下减载装置：

\subsection{铰制孔用螺栓联接}

螺杆直接承受\textbf{挤压}和\textbf{剪切}平衡外载荷 $R$：

\begin{theorem}[横向载荷铰制孔螺栓的强度条件]
\label{thm:横向铰制孔}
\begin{equation}
	\tau=\frac{R}{\dfrac{\pi}{4}d_0^2}\le[\tau]\qquad\text{剪切强度条件}
	\label{eq:连接:铰制孔剪切}
\end{equation}
\begin{equation}
	\sigma_P=\frac{R}{d_0L_{\min}}\le[\sigma]_P\qquad\text{挤压强度条件}
	\label{eq:连接:铰制孔挤压}
\end{equation}
其中：
\begin{itemize}
	\item $R$ 为横向载荷；
	\item $d_0$ 为螺杆或孔直径；
	\item $L_{\min}$ 为被联接件中受挤压孔壁的最小长度；
	\item $[\tau]$ 为许用剪应力。
\end{itemize}
\end{theorem}

\subsection{受轴向载荷的紧螺栓联接（重点）}
\begin{definition}[受力变形分析]
\label{def:变形分析}
\begin{itemize}
	\item \textbf{拧紧（预紧状态）}：螺栓受拉 $Q_P$，伸长 $\lambda_b$；被联接件受压 $Q_P$，压缩 $\lambda_m$；
	\item \textbf{加载 $F$（工作状态）}：螺栓继续受拉，$Q_P\to Q=Q'_P+F$，总伸长 $\lambda'_b=\lambda_b+\Delta\lambda$；被联接件放松，$Q_P\to Q'_P$（\textbf{残余预紧力}），总压缩 $\lambda'_m=\lambda_m-\Delta\lambda$。
\end{itemize}
力—变形关系（螺栓刚度 $C_b$、被联接件刚度 $C_m$）：
\begin{equation}
	C_b=\tan\theta_b=\frac{Q_P}{\lambda_b}=\frac{\Delta F}{\Delta\lambda},\qquad C_m=\tan\theta_m=\frac{Q_P}{\lambda_m}=\frac{F-\Delta F}{\Delta\lambda}
	\label{eq:连接:刚度}
\end{equation}
\end{definition}

\begin{theorem}[螺栓相对刚度与总载荷]
\label{thm:相对刚度}
\textbf{螺栓相对刚度}：
\begin{equation}
	K_c=\frac{C_b}{C_b+C_m}
	\label{eq:连接:相对刚度}
\end{equation}
\begin{equation}
	\Delta F=\frac{C_b}{C_b+C_m}F=K_cF\qquad\text{——部分工作载荷}
	\label{eq:连接:部分工作载荷}
\end{equation}
\textbf{螺栓总载荷}：
\begin{equation}
	Q=F+Q'_P=\Delta F+Q_P=Q_P+K_cF
	\label{eq:连接:总载荷}
\end{equation}
\begin{equation}
	Q_P=Q'_P+(1-K_c)F
	\label{eq:连接:预紧力与残余预紧力}
\end{equation}
\end{theorem}

\begin{definition}[残余预紧力 $Q'_P$ 的取法]

\begin{note}[$K_c$ 的讨论]
\begin{itemize}
	\item 最不利情况 $C_b\gg C_m$：$Q\approx Q_P+F$（外载几乎全由螺栓承受）；
	\item 最理想情况 $C_b\ll C_m$：$Q\approx Q_P$（外载几乎全由被联接件承受）；
	\item \textcolor{red}{不允许 $Q'_P<0$}（出现缝隙，联接失效）。
\end{itemize}

\textbf{降低螺栓受力的措施}：采用刚度较小的螺栓（\textcolor{red}{空心螺栓、加长螺栓、细颈}）+ 加硬垫片或拧紧凸缘（增大被联接件刚度）。
\end{note}
\section{螺栓组连接的设计与受力分析}
\begin{theorem}[螺栓组的结构设计要求]
    \begin{enumerate}
        \item 布局要尽量对称分布，螺栓组中心与联接结合面形心重合（有利于分度、划线、
钻孔），以受力均匀。
        \item 受弯矩和扭矩作用的螺栓组，螺栓应尽量远离对称轴
        \item 合理间距，适当边距，以利用扳手装拆。
        \item 避免偏心载荷（eccentricIoading）作用


    \end{enumerate}
\end{theorem}
\subsection{受轴向载荷螺栓组联接}
单个螺栓的工作载荷：
\begin{equation}
  F=\frac{P}{Z}
\end{equation}
其中，
\begin{itemize}
    \item P:轴向外载
    \item Z:螺栓个数
\end{itemize}

\subsection{受横向载荷螺栓组联接}
特点：普通螺栓、铰制孔用螺栓皆可用；\textcolor{red}{防滑}关键在于——外载垂直于螺栓轴线。

普通螺栓——受拉伸作用。

铰制孔螺栓——受横向载荷剪切、挤压作用。

单个螺栓所承受的横向载荷相等：
\begin{equation}
  R=\frac{R_{\sum}}{Z}
\end{equation}
\remark{若外力不作用在螺栓组的形心上,则需将力移动到形心上再计算}

\subsection{受扭矩螺栓组联接——传力方式与受横向载荷的螺栓组相同}

\textbf{适用场合}：\textcolor{red}{凸缘式联轴器}（flange coupling）、齿轮与轴的凸缘联接等——转矩 $T$ 作用在\textbf{联接接合面内}（$T$ 的作用平面与接合面平行），螺栓组绕形心 $O$ 受扭。

\subsubsection*{a）普通螺栓联接（common bolt joints）}

每个螺栓处由预紧力 $Q_P$ 产生的摩擦力 $fQ_P$，其方向沿该螺栓的\textbf{切向}（垂直于 $O$ 与该螺栓的连线 $r_i$），以各螺栓的摩擦力矩平衡外转矩 $T$，故\textbf{传力方式与受横向载荷的螺栓组完全相同}。

\begin{theorem}[受扭矩普通螺栓组的防滑条件与预紧力]
\label{thm:扭矩普通螺栓}
\textbf{不产生相对滑动的条件}：
\begin{equation}
	fQ_Pr_1+fQ_Pr_2+\cdots+fQ_Pr_z\ge K_sT
	\label{eq:连接:扭矩防滑}
\end{equation}
\textbf{各螺栓所需的预紧力}：
\begin{equation}
	Q_P\ge\frac{T\cdot K_s}{f(r_1+r_2+\cdots+r_z)}=\frac{K_sT}{f\displaystyle\sum_{i=1}^{z}r_i}
	\label{eq:连接:扭矩预紧力}
\end{equation}
其中：
\begin{itemize}
	\item $f$ 为接合面摩擦系数；
	\item $K_s$ 为防滑（可靠性）系数，$K_s=1.1\sim1.3$；
	\item $z$ 为螺栓数目；
	\item $T$ 为外转矩；
	\item $r_i$ 为第 $i$ 个螺栓中心到螺栓组\textbf{形心} $O$ 的距离（$r_1$ 为受力最大螺栓的半径）。
\end{itemize}
\textbf{设计步骤}：由式 \eqref{eq:连接:扭矩预紧力} 求得 $Q_P$ 后，按仅受预紧力的紧螺栓强度条件（式 \eqref{eq:紧螺栓仅预紧}）
$d_1\ge\sqrt{\dfrac{1.3\times4Q_P}{\pi[\sigma]}}$ 确定螺栓直径。
\end{theorem}

\begin{figure}[h]
  \centering
  \includegraphics[width=0.8\textwidth]{Figure/受扭矩螺栓组普通螺栓.png}
  \caption{受扭矩的普通螺栓组（靠预紧力产生的摩擦力传递转矩）}
  \label{fig:受扭矩螺栓组普通螺栓}
\end{figure}

\begin{note}
式 \eqref{eq:连接:扭矩预紧力} 中，$z$ 个螺栓的摩擦力矩是\textbf{算术叠加} $\sum r_i$（不是平方和），这是普通螺栓组受扭与铰制孔螺栓组受扭在公式上最易混淆之处。
\end{note}

\subsubsection*{b）铰制孔螺栓联接组（bolt joints by interference with a reamed hole）}

螺栓杆与孔壁\textbf{无间隙}配合，转矩靠螺栓杆的\textbf{剪切}和\textbf{孔壁的挤压}直接传递。

\begin{theorem}[变形协调条件与最大工作剪力]
\label{thm:扭矩铰制孔螺栓}
\textbf{变形协调条件}：由变形协调条件可知，各个螺栓的变形量和受力大小与其中心到接合面\textbf{形心}的距离成正比：
\begin{equation}
	\frac{R_1}{r_1}=\frac{R_2}{r_2}=\cdots=\frac{R_z}{r_z}
	\label{eq:连接:扭矩变形协调}
\end{equation}
\begin{equation}
	\frac{R_i}{r_i}=\frac{R_{\max}}{r_{\max}}\quad\Rightarrow\quad R_i=\frac{R_{\max}}{r_{\max}}r_i
	\label{eq:连接:扭矩比例关系}
\end{equation}
\textbf{由被联接件力矩平衡}：
\begin{equation}
	R_1r_1+R_2r_2+\cdots+R_zr_z=T
	\label{eq:连接:扭矩力矩平衡}
\end{equation}
两式联立，得受力最大螺栓（距形心最远者）所受的工作剪力：
\begin{equation}
	\boxed{\;R_{\max}=\frac{T\cdot r_{\max}}{\displaystyle\sum_{i=1}^{z}r_i^{2}}\;}
	\label{eq:连接:扭矩最大剪力}
\end{equation}
其中，$R_{\max}$ 为受力最大螺栓的工作剪力，$r_{\max}$ 为该螺栓到形心 $O$ 的距离，$z$ 为螺栓数目。
\end{theorem}

\begin{figure}[h]
  \centering
  \includegraphics[width=0.85\textwidth]{Figure/受扭矩螺栓组铰制孔螺栓.png}
  \caption{受扭矩的铰制孔螺栓组（各螺栓受力与其到形心的距离成正比）}
  \label{fig:受扭矩螺栓组铰制孔螺栓}
\end{figure}

求得 $R_{\max}$ 后，按受横向载荷的铰制孔螺栓强度条件（式 \eqref{eq:连接:铰制孔剪切}、式 \eqref{eq:连接:铰制孔挤压}）作\textbf{剪切}和\textbf{挤压}强度校核。

\begin{keybox}[受扭矩螺栓组小结]
\begin{itemize}
	\item \textbf{普通螺栓}：预紧产生摩擦力矩，$Q_P\ge\dfrac{K_sT}{f\sum r_i}$，按紧螺栓拉伸强度（$\times1.3$）设计；
	\item \textbf{铰制孔螺栓}：直接受剪，$R_{\max}=\dfrac{Tr_{\max}}{\sum r_i^{2}}$，按剪切、挤压强度校核；
	\item 两种螺栓组均要求\textbf{转矩 $T$ 作用在接合面内}；若外力不作用在螺栓组的形心上，则需将力移动到形心上再计算。
\end{itemize}
\end{keybox}

\subsection{受倾覆（纵向）力矩螺栓组联接}
\label{subsec:受倾覆力矩}

\begin{definition}[受倾覆力矩螺栓组的特点]
\label{def:倾覆力矩特点}
\textbf{特点}：倾覆力矩 $M$ 在\textbf{铅直平面内}，底座有绕 $0$-$0$（对称轴线）\textbf{倾转}的趋势。取\textbf{底板}为受力对象，由\textbf{静平衡条件}：
\begin{equation}
	F_1L_1+F_2L_2+\cdots+F_zL_z=M
	\label{eq:连接:倾覆静平衡}
\end{equation}
\textbf{螺栓拉伸变形量越大，所受工作拉力就越大}，故各螺栓的工作拉力 $F_i$ 与其到螺栓组\textbf{对称轴线}的距离 $L_i$ 成正比：
\begin{equation}
	\frac{F_1}{L_1}=\frac{F_i}{L_i}=const=\frac{F_{\max}}{L_{\max}}
	\label{eq:连接:倾覆比例关系}
\end{equation}
\begin{equation}
	\therefore\quad F_{\max}=\frac{M\cdot L_{\max}}{\displaystyle\sum_{i=1}^{z}L_i^{2}}
	\label{eq:连接:倾覆最大拉力}
\end{equation}
其中：
\begin{itemize}
	\item $F_i$——第 $i$ 个螺栓受到底座的轴向力；
	\item $L_i$——第 $i$ 个螺栓到螺栓组\textbf{对称轴线}的距离；
	\item $F_{\max}$——受力最大螺栓的工作拉力，$L_{\max}$ 为其到对称轴线的距离。
\end{itemize}
\end{definition}

\begin{figure}[h]
  \centering
  \includegraphics[width=0.8\textwidth]{Figure/受倾覆力矩螺栓组.png}
  \caption{受倾覆力矩的螺栓组联接}
  \label{fig:受倾覆力矩螺栓组}
\end{figure}

\begin{warnbox}[不要与受扭矩螺栓组混淆]
\begin{itemize}
	\item 受\textbf{扭矩} $T$：螺栓受\textbf{横向剪力}，比例尺是各螺栓到\textbf{形心 $O$} 的距离 $r_i$；
	\item 受\textbf{倾覆力矩} $M$：螺栓受\textbf{轴向拉力}，比例尺是各螺栓到\textbf{对称轴线 $0$-$0$} 的距离 $L_i$；
	\item 两式形式同为“$\dfrac{\text{外载荷}\times\text{最远距离}}{\sum(\text{距离})^2}$”，但几何基准与受力性质完全不同。
\end{itemize}
\end{warnbox}

\begin{note}
若外力不作用在螺栓组的形心上，则需将力移动到形心上再计算。
\end{note}

\begin{remark}
\textbf{联接的补充（进阶）}：由式 \eqref{eq:连接:倾覆最大拉力} 求得最大工作拉力 $F_{\max}$ 后，一般还需按受轴向载荷的紧螺栓联接计算螺栓总拉力 $Q=Q'_P+K_cF_{\max}$（式 \eqref{eq:连接:总载荷}）并校核\textbf{接合面不出现缝隙}（$Q'_P>0$）与\textbf{不压溃}两个条件，也是\textbf{地脚螺栓}（$Q'_P>F$，见表 \ref{tab:连接:残余预紧力}）的设计依据。
\end{remark}

\begin{keybox}[受力分析的通用步骤]
\begin{enumerate}
	\item 将外载荷\textbf{向螺栓组形心平移}（必要时附加一个转矩），转化为上述基本受力状态；
	\item 判断每种基本状态下\textbf{哪个螺栓受力最大}；
	\item 按\textbf{叠加原理}求出受力最大螺栓的总工作载荷（拉力或剪力）；
	\item 按相应强度条件（受拉：仅预紧 $\to$ 式 \eqref{eq:紧螺栓仅预紧}，受轴向载荷 $\to$ 式 \eqref{eq:连接:总载荷}；受剪：式 \eqref{eq:连接:铰制孔剪切}、式 \eqref{eq:连接:铰制孔挤压}）完成设计或校核。
\end{enumerate}
\end{keybox}
