\section{题目一}

取广义坐标
\[
	\boldsymbol{x} = \begin{bmatrix} x_1 & y_1 & x_2 & y_2 \end{bmatrix}^{\mathrm{T}} ,
\]
势能及其梯度为
\[
	f(\boldsymbol{x}) = m_1 g y_1 + m_2 g y_2 ,
	\qquad
	f_{x}(\boldsymbol{x}) = \begin{bmatrix} 0 & m_1 g & 0 & m_2 g \end{bmatrix} .
\]
约束方程为
\[
	\boldsymbol{c}(\boldsymbol{x}) =
	\begin{bmatrix}
		x_1^2 + y_1^2 - l_1^2                        \\
		(x_2 - x_1)^2 + (y_2 - y_1)^2 - l_2^2        \\
		(x_2 - l_4)^2 + y_2^2 - l_3^2
	\end{bmatrix}
	= \boldsymbol{0} ,
\]
雅可比矩阵为
\[
	\boldsymbol{c}_{x}(\boldsymbol{x}) = \frac{\partial \boldsymbol{c}}{\partial \boldsymbol{x}} =
	\begin{bmatrix}
		2x_1         & 2y_1         & 0             & 0             \\[2pt]
		2(x_1 - x_2) & 2(y_1 - y_2) & 2(x_2 - x_1)  & 2(y_2 - y_1)  \\[2pt]
		0            & 0            & 2(x_2 - l_4)  & 2y_2
	\end{bmatrix} .
\]
代入
\[
	\begin{cases}
		f_{x}^{\mathrm{T}}(\boldsymbol{x}) + \boldsymbol{c}_{x}^{\mathrm{T}}(\boldsymbol{x}) \boldsymbol{\lambda} = \boldsymbol{0} , \\[0.6em]
		\boldsymbol{c}(\boldsymbol{x}) = \boldsymbol{0} ,
	\end{cases}
\]
即
\[
	\left\{
	\begin{aligned}
		\begin{bmatrix} 0 \\ m_1 g \\ 0 \\ m_2 g \end{bmatrix}
		+ \begin{bmatrix}
			2x_1 & 2(x_1 - x_2) & 0             \\
			2y_1 & 2(y_1 - y_2) & 0             \\
			0    & 2(x_2 - x_1) & 2(x_2 - l_4)  \\
			0    & 2(y_2 - y_1) & 2y_2
		\end{bmatrix}
		\begin{bmatrix} \lambda_1 \\ \lambda_2 \\ \lambda_3 \end{bmatrix}
		 & = \boldsymbol{0} , \\[1.4em]
		\begin{bmatrix}
			x_1^2 + y_1^2 - l_1^2                  \\
			(x_2 - x_1)^2 + (y_2 - y_1)^2 - l_2^2  \\
			(x_2 - l_4)^2 + y_2^2 - l_3^2
		\end{bmatrix}
		 & = \boldsymbol{0} .
	\end{aligned}
	\right.
\]

\section{题目二}
多元函数$y=f(\boldsymbol{x})$的极小值点满足:
\begin{equation}
\label{eq:extremum}
    \left\{\begin{aligned}
f_{\boldsymbol{x}}^{\mathrm{T}}(\bar{\boldsymbol{x}})+\boldsymbol{c}_{\boldsymbol{x}}^{\mathrm{T}}(\bar{\boldsymbol{x}}) \boldsymbol{\lambda} & =\mathbf{0}, \\
\boldsymbol{c}(\bar{\boldsymbol{x}}) & =\mathbf{0} .
\end{aligned}\right.
\end{equation}
要证明这个解$\bar{\boldsymbol{x}}$是极小值点，就是要证明：$f(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})>f(\bar{\boldsymbol{x}})$。

即$f(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})-f(\bar{\boldsymbol{x}})>0$（$\bar{\boldsymbol{x}}+\delta \boldsymbol{x}$ 为约束面上的点）。

记 $f$ 的黑塞矩阵为 $\boldsymbol{H}_{f}(\bar{\boldsymbol{x}})$，约束方程 $\boldsymbol{c}(\boldsymbol{x})=\mathbf{0}$ 的黑塞矩阵为 $\boldsymbol{H}_{\boldsymbol{c}}(\bar{\boldsymbol{x}})$，
\begin{equation}
	f(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})-f(\bar{\boldsymbol{x}})
	= f_{\boldsymbol{x}}^{\mathrm{T}}(\bar{\boldsymbol{x}}) \delta \boldsymbol{x}
	+ \frac{1}{2} \delta \boldsymbol{x}^{\mathrm{T}} \boldsymbol{H}_{f}(\bar{\boldsymbol{x}}) \delta \boldsymbol{x}
	+ o\left(\|\delta \boldsymbol{x}\|^{2}\right) .
\end{equation}
利用式~\eqref{eq:extremum} 的第一式 $f_{\boldsymbol{x}}(\bar{\boldsymbol{x}})=-\boldsymbol{\lambda}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{x}}(\bar{\boldsymbol{x}})$ 代入一阶项，得
\begin{equation}
	f(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})-f(\bar{\boldsymbol{x}})
	= -\boldsymbol{\lambda}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{x}}(\bar{\boldsymbol{x}}) \delta \boldsymbol{x}
	+ \frac{1}{2} \delta \boldsymbol{x}^{\mathrm{T}} \boldsymbol{H}_{f}(\bar{\boldsymbol{x}}) \delta \boldsymbol{x}
	+ o\left(\|\delta \boldsymbol{x}\|^{2}\right) .
\end{equation}
\[
	\boldsymbol{c}(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})
	= \boldsymbol{c}(\bar{\boldsymbol{x}})
	+ \boldsymbol{c}_{\boldsymbol{x}}(\bar{\boldsymbol{x}}) \delta \boldsymbol{x}
	+ \frac{1}{2} \delta \boldsymbol{x}^{\mathrm{T}} \boldsymbol{H}_{\boldsymbol{c}}(\bar{\boldsymbol{x}}) \delta \boldsymbol{x}
	+ o\left(\|\delta \boldsymbol{x}\|^{2}\right)
	= \mathbf{0} ,
\]
把自变量的微小可行变更分解为切向部分与法向修正部分：
\begin{equation}
	\delta \boldsymbol{x} = \boldsymbol{d} + \boldsymbol{\eta} ,
	\qquad
	\boldsymbol{c}_{\boldsymbol{x}}(\bar{\boldsymbol{x}}) \boldsymbol{d}=\mathbf{0} ,
\end{equation}
其中 $\boldsymbol{d}$ 为切向量，$\boldsymbol{\eta}$ 为剩下的。由 $\boldsymbol{c}_{\boldsymbol{x}} \boldsymbol{d}=\mathbf{0}$ 比较上式得
\[
	\boldsymbol{c}_{\boldsymbol{x}}(\bar{\boldsymbol{x}}) \boldsymbol{\eta}
	=-\frac{1}{2} \boldsymbol{d}^{\mathrm{T}} \boldsymbol{H}_{\boldsymbol{c}}(\bar{\boldsymbol{x}}) \boldsymbol{d}
	+o\left(\|\boldsymbol{d}\|^{2}\right) ,
\]
代回式~(3)，并注意 $\delta \boldsymbol{x}=\boldsymbol{d}+O\left(\|\boldsymbol{d}\|^{2}\right)$，得
\begin{equation}
\label{eq:subst}
	f(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})-f(\bar{\boldsymbol{x}})
	= \frac{1}{2} \boldsymbol{d}^{\mathrm{T}} \boldsymbol{H}_{f}(\bar{\boldsymbol{x}}) \boldsymbol{d}
	+ \frac{1}{2} \boldsymbol{\lambda}^{\mathrm{T}}\left(\boldsymbol{d}^{\mathrm{T}} \boldsymbol{H}_{\boldsymbol{c}}(\bar{\boldsymbol{x}}) \boldsymbol{d}\right)
	+ o\left(\|\boldsymbol{d}\|^{2}\right) .
\end{equation}
因此，只要对任意非零切向量 $\boldsymbol{d}$（即 $\boldsymbol{c}_{\boldsymbol{x}}(\bar{\boldsymbol{x}}) \boldsymbol{d}=\mathbf{0}$）都有
\[
	\boldsymbol{d}^{\mathrm{T}} \boldsymbol{H}_{f}(\bar{\boldsymbol{x}}) \boldsymbol{d}
	+ \boldsymbol{\lambda}^{\mathrm{T}}\left(\boldsymbol{d}^{\mathrm{T}} \boldsymbol{H}_{\boldsymbol{c}}(\bar{\boldsymbol{x}}) \boldsymbol{d}\right)>0 ,
\]
则当 $\|\delta \boldsymbol{x}\|$ 足够小时 $f(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})-f(\bar{\boldsymbol{x}})>0$，即 $f(\bar{\boldsymbol{x}}+\delta \boldsymbol{x})>f(\bar{\boldsymbol{x}})$，所以 $\bar{\boldsymbol{x}}$ 是 $f(\boldsymbol{x})$ 的严格局部极小值点。

\section{题目四}
针对题目一中的四连杆机构，取 $l_1=l_2=1.0\,\mathrm{m},\ l_3=1.2\,\mathrm{m},\ l_4=2.0\,\mathrm{m},\ m_1=m_2=1.0\,\mathrm{kg},\ g=-9.8\,\mathrm{m}\cdot\mathrm{s}^{-2}$，编写数值计算程序，求解重力势能取极小值时两质点的直角坐标 $(x_1,y_1)$ 和 $(x_2,y_2)$ 满足的等式方程，并判定等式方程的解是否为重力势能极小值点。

\begin{lstlisting}
#include <Eigen/Dense>
#include <iostream>
#include <cmath>
#include <algorithm>

using namespace Eigen;
using namespace std;

double l1 = 1.0;
double l2 = 1.0;
double l3 = 1.2;
double l4 = 2.0;
double m1 = 1.0;
double m2 = 1.0;
double g = -9.8;

int GuassSolver(const MatrixXd &A, const VectorXd &b, VectorXd &x, double tol = 1e-12)
{
    int n = A.rows();
    x.resize(n);
    if (n == 0)
        return 0;

    MatrixXd M = A;
    VectorXd rhs = b;

    for (int k = 0; k < n; k++)
    {

        int p = k;
        double maxAbs = abs(M(k, k));

        for (int i = k + 1; i < n; i++)
        {
            const double v = abs(M(i, k));
            if (v > maxAbs)
            {
                maxAbs = v;
                p = i;
            }
        }

        if (maxAbs <= tol)
            return 3;

        if (p != k)
        {
            M.row(k).swap(M.row(p));
            swap(rhs(k), rhs(p));
        }

        double pivotVal = M(k, k);

        for (int i = k + 1; i < n; i++)
        {
            double factor = M(i, k) / pivotVal;

            if (factor == 0.0)
                continue;

            M.row(i).segment(k, n - k) -= factor * M.row(k).segment(k, n - k);
            rhs(i) -= factor * rhs(k);
        }
    }

    for (int i = n - 1; i >= 0; --i)
    {
        double sum = rhs(i);

        for (int j = i + 1; j < n; ++j)
        {
            sum -= M(i, j) * x(j);
        }

        if (std::abs(M(i, i)) <= tol)
            return 3;

        x(i) = sum / M(i, i);
    }

    return 0;
}

VectorXd constraint(const VectorXd &q)
{
    const double x1 = q(0), y1 = q(1), x2 = q(2), y2 = q(3);

    VectorXd c(3);
    c << x1 * x1 + y1 * y1 - l1 * l1,
        (x2 - x1) * (x2 - x1) + (y2 - y1) * (y2 - y1) - l2 * l2,
        (x2 - l4) * (x2 - l4) + y2 * y2 - l3 * l3;
    return c;
}

MatrixXd constraintJacobian(const VectorXd &q)
{
    const double x1 = q(0), y1 = q(1), x2 = q(2), y2 = q(3);

    MatrixXd C(3, 4);
    C << 2.0 * x1, 2.0 * y1, 0.0, 0.0,
        2.0 * (x1 - x2), 2.0 * (y1 - y2), 2.0 * (x2 - x1), 2.0 * (y2 - y1),
        0.0, 0.0, 2.0 * (x2 - l4), 2.0 * y2;
    return C;
}

double potential(const VectorXd &q)
{
    return m1 * g * q(1) + m2 * g * q(3);
}

MatrixXd weightedHessian(const VectorXd &q, const VectorXd &lam)
{
    MatrixXd H = MatrixXd::Zero(4, 4);

    // c1 = x1² + y1² - l1²  →  H_c1 = diag(2, 2, 0, 0)
    H(0, 0) += 2.0 * lam(0);
    H(1, 1) += 2.0 * lam(0);

    // c2 = (x2-x1)² + (y2-y1)² - l2²
    H(0, 0) += 2.0 * lam(1);
    H(2, 2) += 2.0 * lam(1);
    H(1, 1) += 2.0 * lam(1);
    H(3, 3) += 2.0 * lam(1);
    H(0, 2) -= 2.0 * lam(1);
    H(2, 0) -= 2.0 * lam(1);
    H(1, 3) -= 2.0 * lam(1);
    H(3, 1) -= 2.0 * lam(1);

    // c3 = (x2-l4)² + y2² - l3²  →  H_c3 = diag(0, 0, 2, 2)
    H(2, 2) += 2.0 * lam(2);
    H(3, 3) += 2.0 * lam(2);

    return H;
}

VectorXd residual(const VectorXd &x)
{
    const VectorXd q = x.head(4);
    const VectorXd lam = x.tail(3);

    VectorXd fx(4);
    fx << 0.0, m1 * g, 0.0, m2 * g; // f_x = (0, m1 g, 0, m2 g)

    VectorXd F(7);
    F.head(4) = fx + constraintJacobian(q).transpose() * lam;
    F.tail(3) = constraint(q);
    return F;
}

MatrixXd jacobian(const VectorXd &x)
{
    const VectorXd q = x.head(4);
    const VectorXd lam = x.tail(3);

    const MatrixXd C = constraintJacobian(q);

    MatrixXd J = MatrixXd::Zero(7, 7);
    J.topLeftCorner(4, 4) = weightedHessian(q, lam);
    J.topRightCorner(4, 3) = C.transpose();
    J.bottomLeftCorner(3, 4) = C;
    return J; // 右下 3×3 保持为 0
}

bool newtonSolve(VectorXd &x, int maxIter, double tol)
{
    for (int iter = 0; iter < maxIter; ++iter)
    {
        const VectorXd F = residual(x);
        if (F.norm() < tol)
            return true;

        const MatrixXd J = jacobian(x);
        const VectorXd rhs = -F;

        VectorXd dx; // 用高斯消元法求解 J * dx = -F
        if (GuassSolver(J, rhs, dx) != 0)
            return false;

        x += dx;

        if (dx.norm() < 1e-14)
            break;
    }

    return residual(x).norm() < 1e-9;
}

double secondOrderQ(const VectorXd &x)
{
    const VectorXd q = x.head(4);
    const VectorXd lam = x.tail(3);
    const MatrixXd C = constraintJacobian(q); // 3×4

    VectorXd d(4);
    for (int i = 0; i < 4; ++i)
    {
        MatrixXd minor(3, 3);
        int col = 0;
        for (int j = 0; j < 4; ++j)
        {
            if (j == i)
                continue;
            minor.col(col) = C.col(j);
            ++col;
        }
        d(i) = ((i % 2 == 0) ? 1.0 : -1.0) * minor.determinant();
    }

    const MatrixXd H = weightedHessian(q, lam);
    return d.dot(H * d);
}

int main()
{

    cout << "l1=" << l1 << ", l2=" << l2 << ", l3=" << l3
         << ", l4=" << l4 << ", m1=" << m1 << ", m2=" << m2
         << ", g=" << g << "\n\n";

    const int nStarts = 4;
    const double q0s[nStarts][4] = {
        {0.5, 0.9, 1.4, 1.0},
        {0.5, -0.9, 1.4, -1.0},
        {0.3, 1.0, 1.1, 1.1},
        {0.3, -1.0, 1.1, -1.1},
    };

    VectorXd sols[8];
    int found = 0;

    for (int s = 0; s < nStarts; ++s)
    {
        VectorXd x(7);
        x << q0s[s][0], q0s[s][1], q0s[s][2], q0s[s][3], 1.0, 1.0, 1.0;

        if (!newtonSolve(x, 100, 1e-10))
            continue;

        bool isNew = true;
        for (int k = 0; k < found; ++k)
        {
            if ((x - sols[k]).norm() < 1e-6)
            {
                isNew = false;
                break;
            }
        }

        if (isNew && found < 8)
            sols[found++] = x;
    }

    cout << "共找到 " << found << " 个驻点解：\n";

    for (int k = 0; k < found; ++k)
    {
        const VectorXd x = sols[k];
        const VectorXd q = x.head(4);
        const VectorXd lam = x.tail(3);

        const double f = potential(q);
        const double Q = secondOrderQ(x);

        cout << "\n解 " << k + 1 << "：\n";
        cout << "  (x1, y1) = (" << q(0) << ", " << q(1) << ")\n";
        cout << "  (x2, y2) = (" << q(2) << ", " << q(3) << ")\n";
        cout << "  ($\lambda$1, $\lambda$2, $\lambda$3) = (" << lam(0) << ", " << lam(1) << ", " << lam(2) << ")\n";
        cout << "  势能 f = " << f << "\n";
        cout << "  二阶判据 Q = " << Q << "  ->  ";

       
    }

    return 0;
}
