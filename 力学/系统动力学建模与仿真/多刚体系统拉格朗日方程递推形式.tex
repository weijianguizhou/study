\section{系统运动微分方程}
系统动力学微分代数方程可表示为：

\[
\left[\begin{array}{ll}
\boldsymbol{M} & \boldsymbol{c}_{\boldsymbol{q}}^{\mathrm{T}} \\
\boldsymbol{c}_{\boldsymbol{q}} & \boldsymbol{O}
\end{array}\right]\left[\begin{array}{l}
\ddot{\boldsymbol{q}} \\
\boldsymbol{\lambda}
\end{array}\right]=\left[\begin{array}{l}
\boldsymbol{h}(\boldsymbol{q}, \dot{\boldsymbol{q}}, t) \\
\varepsilon(\boldsymbol{q}, \dot{\boldsymbol{q}}, t)
\end{array}\right]
\]

\[
\left\{\begin{array}{l}
\mathbf{0}=\boldsymbol{c}, \\
\mathbf{0}=\boldsymbol{c}_{\boldsymbol{q}} \dot{\boldsymbol{q}}+\boldsymbol{c}_{t},
\end{array}\right.
\]

式中：

\[
\left\{\begin{aligned}
\boldsymbol{h}(\boldsymbol{q}, \dot{\boldsymbol{q}}, t) & =\boldsymbol{f}_{\mathrm{app}}-\dot{\boldsymbol{M}} \dot{\boldsymbol{q}}+T_{\boldsymbol{q}}^{\mathrm{T}} \\
\varepsilon(\boldsymbol{q}, \dot{\boldsymbol{q}}, t) & =-\dot{\boldsymbol{c}}_{\boldsymbol{q}} \dot{\boldsymbol{q}}-\dot{\boldsymbol{c}}_{t}
\end{aligned}\right.
\]

$\boldsymbol{f}_{\mathrm{app}}$是系统广义力，包括重力、弹性阻尼力、摩擦力与控制力等.详细推导见[[2读书笔记——第二章 多体系统拉格朗日方程]]
\section{系统广义质量矩阵}
由上面的推导可知系统的速度列阵为：

\[
\boldsymbol{v}_{0, i \mid i}=\sum_{k \in \overrightarrow{\mathrm{p}}[i]} \boldsymbol{C}_{i, k} \boldsymbol{\Phi}_{\mathrm{J}_{k}} \dot{\boldsymbol{q}}_{\mathrm{J}_{k}}
\]

代入系统的动能表达式$T=\frac{1}{2} \sum_{i=1}^{n} \boldsymbol{v}_{0, i \mid i}^{\mathrm{T}} \boldsymbol{M}_{i} \boldsymbol{v}_{0, i \mid i}$ 整理可得

\[
\begin{aligned}
T & =\frac{1}{2} \sum_{i=1}^{n} \sum_{j \in \overrightarrow{\mathrm{p}}[i]} \dot{\boldsymbol{q}}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{\Phi}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{C}_{i, j}^{\mathrm{T}} \boldsymbol{M}_{i} \sum_{k \in \overrightarrow{\mathrm{p}}[i]} \boldsymbol{C}_{i, k} \boldsymbol{\Phi}_{\mathrm{J}_{k}} \dot{\boldsymbol{q}}_{\mathrm{J}_{k}} \\
& =\frac{1}{2} \sum_{i=1}^{n} \sum_{j \in \overrightarrow{\mathrm{p}}[i]} \sum_{k \in \overrightarrow{\mathrm{p}}[i]} \dot{\boldsymbol{q}}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{\Phi}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{C}_{i, j}^{\mathrm{T}} \boldsymbol{M}_{i} \boldsymbol{C}_{i, k} \boldsymbol{\Phi}_{\mathrm{J}_{k}} \dot{\boldsymbol{q}}_{\mathrm{J}_{k}} \\
& =: \frac{1}{2} \sum_{i=1}^{n} \sum_{j \in \overrightarrow{\mathrm{p}}[i]} \sum_{k \in \overrightarrow{\mathrm{p}}[i]} \dot{\boldsymbol{q}}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{M}_{j, k}^{i} \dot{\boldsymbol{q}}_{\mathrm{J}_{k}} \\
& =: \frac{1}{2} \dot{\boldsymbol{q}}^{\mathrm{T}} \boldsymbol{M} \dot{\boldsymbol{q}},
\end{aligned}
\]

式中$\boldsymbol{M}$为系统广义质量矩阵，且由前文的论证可知，系统广义质量矩阵为实对称矩阵。

\[
\boldsymbol{M}=\boldsymbol{M}^\mathrm{T}
\]
\section{系统广义质量矩阵对时间的一阶导数}
系统广义质量矩阵$\boldsymbol{M}_{j, k}^{i}$的时间倒数满足

\[
\begin{aligned}
\dot{\boldsymbol{M}}_{j, k}^{i} & =\frac{\mathrm{d}}{\mathrm{~d} t}\left(\boldsymbol{\Phi}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{C}_{i, j}^{\mathrm{T}} \boldsymbol{M}_{i} \boldsymbol{C}_{i, k} \boldsymbol{\Phi}_{\mathrm{J}_{k}}\right) \\
& =\left(\dot{\boldsymbol{\Phi}}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{C}_{i, j}^{\mathrm{T}}+\boldsymbol{\Phi}_{\mathrm{J}_{j}}^{\mathrm{T}} \dot{\boldsymbol{C}}_{i, j}^{\mathrm{T}}\right) \boldsymbol{M}_{i} \boldsymbol{C}_{i, k} \boldsymbol{\Phi}_{\mathrm{J}_{k}}+\boldsymbol{\Phi}_{\mathrm{J}_{j}}^{\mathrm{T}} \boldsymbol{C}_{i, j}^{\mathrm{T}} \boldsymbol{M}_{i}\left(\dot{\boldsymbol{C}}_{i, k} \boldsymbol{\Phi}_{\mathrm{J}_{k}}+\boldsymbol{C}_{i, k} \dot{\boldsymbol{\Phi}}_{\mathrm{J}_{k}}\right)
\end{aligned}
\]

式中$\dot{\boldsymbol{C}}_{i,k}$可通过递推方式获得，即

\[
\dot{\boldsymbol{C}}_{i, k}=\frac{\mathrm{d}}{\mathrm{~d} t}\left(\boldsymbol{C}_{i, \mathrm{p}(i)} \boldsymbol{C}_{\mathrm{p}(i), k}\right)=\dot{\boldsymbol{C}}_{i, \mathrm{p}(i)} \boldsymbol{C}_{\mathrm{p}(i), k}+\boldsymbol{C}_{i, \mathrm{p}(i)} \dot{\boldsymbol{C}}_{\mathrm{p}(i), k}

\[
\begin{aligned}

\section{动力学微分代数的约化}
下面考虑如何将动力学微分代数方程,即

\[
\left[\begin{array}{ll}
\boldsymbol{M} & \boldsymbol{c}_{\boldsymbol{q}}^{\mathrm{T}} \\
\boldsymbol{c}_{\boldsymbol{q}} & \boldsymbol{O}
\end{array}\right]\left[\begin{array}{l}

\]

\boldsymbol{q}_{\mathrm{I}} \\
\boldsymbol{q}_{\mathrm{D}}
\end{array}\right], \quad \dot{\boldsymbol{q}}=\left[\begin{array}{c}
\dot{\boldsymbol{q}}_{\mathrm{I}} \\
\dot{\boldsymbol{q}}_{\mathrm{D}}
\end{array}\right], \quad \ddot{\boldsymbol{q}}=\left[\begin{array}{c}
\ddot{\boldsymbol{q}}_{\mathrm{I}} \\
\ddot{\boldsymbol{q}}_{\mathrm{D}}
\end{array}\right]
\]

这并不是一般的情形; 对于更一般的情形，张老师书中暂不讨论.应用系统广义坐标的划分，可将系统加速度级约束方程重新整理为

\[
\begin{aligned}
\boldsymbol{\varepsilon}(\boldsymbol{q}, \dot{\boldsymbol{q}}, t) & =c_{\boldsymbol{q}} \ddot{\boldsymbol{q}} \\
& =:\left[\begin{array}{ll}
c_{q_{\mathrm{I}}} & c_{q_{\mathrm{D}}}
\end{array}\right]\left[\begin{array}{c}
\ddot{\boldsymbol{q}}_{\mathrm{I}} \\
\ddot{\boldsymbol{q}}_{\mathrm{D}}
\end{array}\right] \\
& =c_{q_{\mathrm{I}}} \ddot{\boldsymbol{q}}_{\mathrm{I}}+c_{q_{\mathrm{D}}} \ddot{\boldsymbol{q}}_{\mathrm{D}} .
\end{aligned}
\]

考虑到四足机器人系统中存在冗余约束， $c_{q_{\mathrm{D}}} \in \mathbb{R}^{20 \times 8}$  不是方阵，上式左右两端同时左乘  $c_{q_{\mathrm{D}}}^{\mathrm{T}}$  ，整理可得
\[
\boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}} \ddot{\boldsymbol{q}}_{\mathrm{D}}=-\boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{I}}} \ddot{\boldsymbol{q}}_{\mathrm{I}}+\boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \varepsilon .
\]
由上式可得
\[
\ddot{\boldsymbol{q}}_{\mathrm{D}}=-\left(\boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}\right)^{-1} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{I}}} \ddot{\boldsymbol{q}}_{\mathrm{I}}+\left(\boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}\right)^{-1} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \varepsilon .
\]
则有

\[
\begin{aligned}
\ddot{\boldsymbol{q}}=\left[\begin{array}{c}
\ddot{\boldsymbol{q}}_{\mathrm{I}} \\
\ddot{\boldsymbol{q}}_{\mathrm{D}}
\end{array}\right] & =\left[\begin{array}{c}
\boldsymbol{I}_{18} \\
-\left(\boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}\right)^{-1} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{I}}}
\end{array}\right] \ddot{\boldsymbol{q}}_{\mathrm{I}}+\left[\begin{array}{c}
\mathbf{0}_{18} \\
\left(\boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}\right)^{-1} \boldsymbol{c}_{\boldsymbol{q}_{\mathrm{D}}}^{\mathrm{T}} \varepsilon
\end{array}\right] \\
& =: \boldsymbol{\Phi} \ddot{\boldsymbol{q}}_{\mathrm{I}}+\boldsymbol{\xi} .
\end{aligned}
\]

将式（11．35）代入动力学拉格朗日方程可得
\[
\begin{aligned}
\boldsymbol{h}(\boldsymbol{q}, \dot{\boldsymbol{q}}, t) & =\boldsymbol{M} \ddot{\boldsymbol{q}}+\boldsymbol{c}_{\boldsymbol{q}}^{\mathrm{T}} \boldsymbol{\lambda} \\
& =\boldsymbol{M}\left(\boldsymbol{\Phi} \ddot{\boldsymbol{q}}_{\mathrm{I}}+\boldsymbol{\xi}\right)+\boldsymbol{c}_{\boldsymbol{q}}^{\mathrm{T}} \boldsymbol{\lambda} \\
& =\boldsymbol{M} \boldsymbol{\Phi} \ddot{\boldsymbol{q}}_{\mathrm{I}}+\boldsymbol{M} \boldsymbol{\xi}+\boldsymbol{c}_{\boldsymbol{q}}^{\mathrm{T}} \boldsymbol{\lambda} .
\end{aligned}
\]
上式左乘  $\boldsymbol{\Phi}^{\mathrm{T}}$  可得
\[
\boldsymbol{\Phi}^{\mathrm{T}} \boldsymbol{h}=\boldsymbol{\Phi}^{\mathrm{T}} \boldsymbol{M} \boldsymbol{\Phi} \ddot{\boldsymbol{q}}_{\mathrm{I}}+\boldsymbol{\Phi}^{\mathrm{T}} \boldsymbol{M} \boldsymbol{\xi}+\boldsymbol{\Phi}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}}^{\mathrm{T}} \boldsymbol{\lambda} .
\]
考虑到
\[
\boldsymbol{\Phi}^{\mathrm{T}} \boldsymbol{c}_{\boldsymbol{q}}^{\mathrm{T}} \boldsymbol{\lambda}=\mathbf{0},
\]
式（11．37）可化简为如下二阶常微分方程组
\[
\boldsymbol{\Phi}^{\mathrm{T}} \boldsymbol{M} \boldsymbol{\Phi} \ddot{\boldsymbol{q}}_{\mathrm{I}}=\boldsymbol{\Phi}^{\mathrm{T}}(\boldsymbol{h}-\boldsymbol{M} \boldsymbol{\xi}) .
\]
