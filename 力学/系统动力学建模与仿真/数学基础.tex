\section{代数基础}

本节介绍多体系统动力学建模中所需要的矩阵知识。由于后续章节仅用到反对称矩阵与叉乘矩阵，这里只保留这一部分内容，其余矩阵分析内容从略。

\subsection{矩阵分析基础}

\subsubsection{对称矩阵与反对称矩阵}

\begin{mydefinition}{对称矩阵与反对称矩阵}
设 $\boldsymbol{A}$ 为 $n$ 阶方阵。若 $\boldsymbol{A}^{\mathrm{T}} = \boldsymbol{A}$，即 $a_{ij} = a_{ji}$，则称 $\boldsymbol{A}$ 为对称矩阵；若 $\boldsymbol{A}^{\mathrm{T}} = -\boldsymbol{A}$，即 $a_{ij} = -a_{ji}$，则称 $\boldsymbol{A}$ 为反对称矩阵。反对称矩阵的主对角元素均为零。
\end{mydefinition}

任一 $n$ 阶方阵 $\boldsymbol{A}$ 均可唯一分解为一个对称矩阵与一个反对称矩阵之和：
\begin{equation}\label{eq:矩阵:对称分解}
\boldsymbol{A} = \frac{\boldsymbol{A}+\boldsymbol{A}^{\mathrm{T}}}{2} + \frac{\boldsymbol{A}-\boldsymbol{A}^{\mathrm{T}}}{2}.
\end{equation}

反对称矩阵与三维向量的叉乘有着密切的联系。对任意向量 $\boldsymbol{a} = \begin{bmatrix} a_1 & a_2 & a_3 \end{bmatrix}^{\mathrm{T}}\in\mathbb{R}^3$，定义其叉乘矩阵为反对称矩阵
\begin{equation}\label{eq:矩阵:叉乘矩阵}
\boldsymbol{a}^{\times} \triangleq \begin{bmatrix} 0 & -a_3 & a_2 \\ a_3 & 0 & -a_1 \\ -a_2 & a_1 & 0 \end{bmatrix},
\end{equation}
则向量叉乘可以表示为矩阵乘法：
\begin{equation}\label{eq:矩阵:叉乘矩阵性质}
\boldsymbol{a}\times\boldsymbol{b} = \boldsymbol{a}^{\times}\,\boldsymbol{b}, \qquad (\boldsymbol{a}\times\boldsymbol{b})^{\times} = \boldsymbol{b}\boldsymbol{a}^{\mathrm{T}} - \boldsymbol{a}\boldsymbol{b}^{\mathrm{T}},
\end{equation}
式中 $\boldsymbol{a}\times\boldsymbol{b}$ 表示向量叉乘。叉乘矩阵在刚体动力学中具有重要作用，例如角速度向量与姿态变化率的关系即借助叉乘矩阵表示。为表述方便，常把 $\mathbf{q}\times$ 看做一个矩阵，这个矩阵记为 $\tilde{\mathbf{q}}$，其元素为
\begin{equation}
\tilde{\mathbf{q}}=\begin{bmatrix}
0 & -q_3 & q_2 \\
q_3 & 0 & -q_1 \\
-q_2 & q_1 & 0
\end{bmatrix},
\end{equation}
即 $\tilde{\mathbf{q}}$ 与式~\eqref{eq:矩阵:叉乘矩阵} 中的 $\boldsymbol{a}^{\times}$ 表示同一个对象，两种记法在不同文献中通用。

反对称矩阵还具有如下性质：对 $n$ 阶反对称矩阵 $\boldsymbol{A}$，
\begin{equation*}
\det(\boldsymbol{A}^{\mathrm{T}}) = \det(-\boldsymbol{A}) = (-1)^{n}\det(\boldsymbol{A}),
\end{equation*}
当 $n$ 为奇数时 $\det\boldsymbol{A} = 0$；且反对称矩阵的特征值位于虚轴上，特征向量相互正交（酉）。此外，任意方阵均可唯一分解为对称部分与反对称部分之和（式~\eqref{eq:矩阵:对称分解}），该分解在刚体动力学中对应应变率张量与旋率张量的分解。

\section{并矢}

并矢（dyadic）是描述二阶线性变换的数学工具，刚体动力学中的转动算子与转动惯量张量均可表示为并矢。本节介绍并矢的定义、运算及其与矩阵的联系。

\subsection{并矢的定义}

\begin{mydefinition}{并矢}
设 $\boldsymbol{a}$、$\boldsymbol{b}$ 为两个向量，则按确定顺序并列而成的量 $\boldsymbol{a}\boldsymbol{b}$ 称为并矢（dyad）。有限个并矢之和
\begin{equation}\label{eq:并矢:定义}
\boldsymbol{\Phi} = \sum_{i=1}^{N}\boldsymbol{a}_i\boldsymbol{b}_i
\end{equation}
称为并矢式（dyadic）。
\end{mydefinition}

并矢与向量的点积定义为
\begin{equation}\label{eq:并矢:点积}
\boldsymbol{a}\boldsymbol{b}\cdot\boldsymbol{v} = \boldsymbol{a}(\boldsymbol{b}\cdot\boldsymbol{v}), \qquad \boldsymbol{v}\cdot\boldsymbol{a}\boldsymbol{b} = (\boldsymbol{v}\cdot\boldsymbol{a})\boldsymbol{b},
\end{equation}
即并矢与向量的点积仍是向量；注意一般地 $\boldsymbol{\Phi}\cdot\boldsymbol{v}\neq\boldsymbol{v}\cdot\boldsymbol{\Phi}$，点积的顺序不可交换。

并矢式 $\boldsymbol{\Phi}$ 关于 $\boldsymbol{v}$ 的映射 $\boldsymbol{v}\mapsto\boldsymbol{\Phi}\cdot\boldsymbol{v}$ 是线性的，故并矢式是二阶线性变换的表示。反之，任意将向量映射为向量的线性变换均可表示为并矢式，且至多需要三个并矢：

\begin{myTheorem}{并矢的基展开}
设 $\mathbf{e}_1,\mathbf{e}_2,\mathbf{e}_3$ 为一组右手正交单位向量，则任意并矢式 $\boldsymbol{\Phi}$ 可表示为
\begin{equation}\label{eq:并矢:展开}
\boldsymbol{\Phi} = \boldsymbol{\Phi}\cdot\mathbf{e}_1\,\mathbf{e}_1 + \boldsymbol{\Phi}\cdot\mathbf{e}_2\,\mathbf{e}_2 + \boldsymbol{\Phi}\cdot\mathbf{e}_3\,\mathbf{e}_3.
\end{equation}
\end{myTheorem}

\begin{proof}
对任意向量 $\boldsymbol{v}$，由单位向量组的完备性 $\displaystyle\sum_{i=1}^{3}\mathbf{e}_i(\mathbf{e}_i\cdot\boldsymbol{v}) = \boldsymbol{v}$ 可得
$$
\left(\sum_{i=1}^{3}\boldsymbol{\Phi}\cdot\mathbf{e}_i\,\mathbf{e}_i\right)\cdot\boldsymbol{v}
= \sum_{i=1}^{3}(\boldsymbol{\Phi}\cdot\mathbf{e}_i)(\mathbf{e}_i\cdot\boldsymbol{v})
= \boldsymbol{\Phi}\cdot\sum_{i=1}^{3}\mathbf{e}_i(\mathbf{e}_i\cdot\boldsymbol{v})
= \boldsymbol{\Phi}\cdot\boldsymbol{v}.
$$
由 $\boldsymbol{v}$ 的任意性即知式~\eqref{eq:并矢:展开} 成立。
\end{proof}

\subsection{并矢的运算}

并矢式之间可进行加法与数乘运算，按并矢的线性性质逐项进行，并满足结合律与分配律。两个并矢式的点积定义为
\begin{equation}\label{eq:并矢:并矢点积}
\boldsymbol{a}\boldsymbol{b}\cdot\boldsymbol{c}\boldsymbol{d} = (\boldsymbol{b}\cdot\boldsymbol{c})\,\boldsymbol{a}\boldsymbol{d},
\end{equation}
即先对相邻向量作点积，结果为并矢式；一般地 $\boldsymbol{\Phi}\cdot\boldsymbol{\Psi}\neq\boldsymbol{\Psi}\cdot\boldsymbol{\Phi}$。

并矢式与向量的叉积定义为
\begin{equation}\label{eq:并矢:叉积}
\boldsymbol{a}\boldsymbol{b}\times\boldsymbol{v} = \boldsymbol{a}(\boldsymbol{b}\times\boldsymbol{v}), \qquad \boldsymbol{v}\times\boldsymbol{a}\boldsymbol{b} = (\boldsymbol{v}\times\boldsymbol{a})\boldsymbol{b},
\end{equation}
并矢式与向量的叉积仍是并矢式。旋转并矢（见本节末尾）正是借助并矢式与向量的叉积构造的。

\subsection{单位并矢}

\begin{mydefinition}{单位并矢}
设 $\mathbf{e}_1,\mathbf{e}_2,\mathbf{e}_3$ 为一组右手正交单位向量，则
\begin{equation}\label{eq:并矢:单位并矢}
\boldsymbol{U} = \mathbf{e}_1\mathbf{e}_1 + \mathbf{e}_2\mathbf{e}_2 + \mathbf{e}_3\mathbf{e}_3
\end{equation}
称为单位并矢。
\end{mydefinition}

单位并矢具有如下性质：
\begin{enumerate}
\item 对任意向量 $\boldsymbol{v}$，$\boldsymbol{U}\cdot\boldsymbol{v} = \boldsymbol{v}\cdot\boldsymbol{U} = \boldsymbol{v}$；
\item 对任意并矢式 $\boldsymbol{\Phi}$，$\boldsymbol{U}\cdot\boldsymbol{\Phi} = \boldsymbol{\Phi}\cdot\boldsymbol{U} = \boldsymbol{\Phi}$；
\item $\boldsymbol{U}^{\mathrm{T}} = \boldsymbol{U}$，即单位并矢为对称并矢；
\item 对任意向量 $\boldsymbol{a}$、$\boldsymbol{b}$，$\boldsymbol{a}\cdot\boldsymbol{U}\cdot\boldsymbol{b} = \boldsymbol{a}\cdot\boldsymbol{b}$。
\end{enumerate}

\subsection{并矢的转置}

\begin{mydefinition}{并矢的转置}
并矢 $\boldsymbol{a}\boldsymbol{b}$ 的转置定义为
\begin{equation}\label{eq:并矢:转置定义}
(\boldsymbol{a}\boldsymbol{b})^{\mathrm{T}} = \boldsymbol{b}\boldsymbol{a},
\end{equation}
并矢式 $\boldsymbol{\Phi} = \sum_{i}\boldsymbol{a}_i\boldsymbol{b}_i$ 的转置定义为
\begin{equation}\label{eq:并矢:转置}
\boldsymbol{\Phi}^{\mathrm{T}} = \sum_{i}\boldsymbol{b}_i\boldsymbol{a}_i.
\end{equation}
\end{mydefinition}

转置运算满足 $(\boldsymbol{\Phi}^{\mathrm{T}})^{\mathrm{T}} = \boldsymbol{\Phi}$、$(\boldsymbol{\Phi}+\boldsymbol{\Psi})^{\mathrm{T}} = \boldsymbol{\Phi}^{\mathrm{T}}+\boldsymbol{\Psi}^{\mathrm{T}}$ 以及 $(\boldsymbol{\Phi}\cdot\boldsymbol{\Psi})^{\mathrm{T}} = \boldsymbol{\Psi}^{\mathrm{T}}\cdot\boldsymbol{\Phi}^{\mathrm{T}}$。若 $\boldsymbol{\Phi}^{\mathrm{T}} = \boldsymbol{\Phi}$，则称 $\boldsymbol{\Phi}$ 为对称并矢；若 $\boldsymbol{\Phi}^{\mathrm{T}} = -\boldsymbol{\Phi}$，则称 $\boldsymbol{\Phi}$ 为反对称并矢。任一并矢式均可唯一分解为对称部分与反对称部分之和：
\begin{equation}\label{eq:并矢:对称分解}
\boldsymbol{\Phi} = \frac{\boldsymbol{\Phi}+\boldsymbol{\Phi}^{\mathrm{T}}}{2} + \frac{\boldsymbol{\Phi}-\boldsymbol{\Phi}^{\mathrm{T}}}{2}.
\end{equation}

\subsection{并矢的矩阵表示}

设 $\mathbf{e}_1,\mathbf{e}_2,\mathbf{e}_3$ 为一组右手正交单位向量，则并矢式 $\boldsymbol{\Phi}$ 可展开为
\begin{equation}\label{eq:并矢:分量展开}
\boldsymbol{\Phi} = \sum_{i=1}^{3}\sum_{j=1}^{3}\Phi_{ij}\,\mathbf{e}_i\mathbf{e}_j,
\end{equation}
式中分量 $\Phi_{ij} = \mathbf{e}_i\cdot\boldsymbol{\Phi}\cdot\mathbf{e}_j$。于是 $\boldsymbol{\Phi}$ 与该基下的矩阵
\begin{equation}\label{eq:并矢:矩阵表示}
[\boldsymbol{\Phi}] = \begin{bmatrix} \Phi_{11} & \Phi_{12} & \Phi_{13} \\ \Phi_{21} & \Phi_{22} & \Phi_{23} \\ \Phi_{31} & \Phi_{32} & \Phi_{33} \end{bmatrix}
\end{equation}
一一对应，并矢与向量的点积对应矩阵与向量的乘法：
\begin{equation}\label{eq:并矢:矩阵点积}
[\boldsymbol{\Phi}\cdot\boldsymbol{v}] = [\boldsymbol{\Phi}][\boldsymbol{v}], \qquad [\boldsymbol{v}\cdot\boldsymbol{\Phi}] = [\boldsymbol{v}]^{\mathrm{T}}[\boldsymbol{\Phi}],
\end{equation}
并矢的转置对应矩阵的转置，即 $[\boldsymbol{\Phi}^{\mathrm{T}}] = [\boldsymbol{\Phi}]^{\mathrm{T}}$。

若 $\mathbf{a}_1,\mathbf{a}_2,\mathbf{a}_3$ 与 $\mathbf{b}_1,\mathbf{b}_2,\mathbf{b}_3$ 分别为两个坐标系中的右手正交单位向量，则描述两坐标系相对姿态的并矢 $\boldsymbol{\sigma}$ 可表示为
\begin{equation}\label{eq:并矢:方向余弦并矢}
\boldsymbol{\sigma} = \sum_{i=1}^{3}\sum_{j=1}^{3} a_{ij}\,\mathbf{a}_i\mathbf{b}_j, \qquad a_{ij} = \mathbf{a}_i\cdot\boldsymbol{\sigma}\cdot\mathbf{b}_j,
\end{equation}
式中 $a_{ij}$ 即为方向余弦矩阵的元素（见第 3 章运动学基础间接定向法一节）。这体现了并矢是坐标无关的几何量，而矩阵只是并矢在某组基下的具体表示。

\subsection{旋转并矢}

在简单转动（见第 3 章运动学基础）中，向量 $\boldsymbol{a}$ 绕单位向量 $\boldsymbol{\lambda}$ 旋转角 $\theta$ 得到向量 $\boldsymbol{b}$，二者满足 Rodrigues 公式
\begin{equation}\label{eq:并矢:旋转公式}
\boldsymbol{b} = \boldsymbol{a}\cos\theta - (\boldsymbol{a}\times\boldsymbol{\lambda})\sin\theta + \boldsymbol{\lambda}(\boldsymbol{\lambda}\cdot\boldsymbol{a})(1-\cos\theta).
\end{equation}
上式右端是 $\boldsymbol{a}$ 的线性函数，故可表示为某个并矢与 $\boldsymbol{a}$ 的点积：

\begin{myTheorem}{旋转并矢}
\begin{equation}\label{eq:并矢:旋转并矢}
\boldsymbol{A} = \boldsymbol{U}\cos\theta - (\boldsymbol{U}\times\boldsymbol{\lambda})\sin\theta + \boldsymbol{\lambda}\boldsymbol{\lambda}(1-\cos\theta)
\end{equation}
称为绕 $\boldsymbol{\lambda}$ 轴旋转角 $\theta$ 的旋转并矢，满足 $\boldsymbol{b} = \boldsymbol{a}\cdot\boldsymbol{A}$。
\end{myTheorem}

\begin{proof}
对任意向量 $\boldsymbol{a}$，
$$
\boldsymbol{a}\cdot\boldsymbol{A}
= \boldsymbol{a}\cdot\boldsymbol{U}\cos\theta - \boldsymbol{a}\cdot(\boldsymbol{U}\times\boldsymbol{\lambda})\sin\theta + \boldsymbol{a}\cdot\boldsymbol{\lambda}\boldsymbol{\lambda}(1-\cos\theta).
$$
由单位并矢性质 $\boldsymbol{a}\cdot\boldsymbol{U} = \boldsymbol{a}$，并矢叉积定义 $\boldsymbol{a}\cdot(\boldsymbol{U}\times\boldsymbol{\lambda}) = \boldsymbol{a}\times\boldsymbol{\lambda}$ 以及 $\boldsymbol{a}\cdot\boldsymbol{\lambda}\boldsymbol{\lambda} = \boldsymbol{\lambda}(\boldsymbol{\lambda}\cdot\boldsymbol{a})$，即得
$$
\boldsymbol{a}\cdot\boldsymbol{A} = \boldsymbol{a}\cos\theta - (\boldsymbol{a}\times\boldsymbol{\lambda})\sin\theta + \boldsymbol{\lambda}(\boldsymbol{\lambda}\cdot\boldsymbol{a})(1-\cos\theta) = \boldsymbol{b},
$$
与式~\eqref{eq:并矢:旋转公式} 一致。
\end{proof}

由并矢的矩阵表示可知，旋转并矢 $\boldsymbol{A}$ 在某组基下的矩阵即该旋转的方向余弦矩阵，二者是同一几何对象在不同表示体系下的体现。此外，刚体动力学中的转动惯量张量即为对称并矢，其矩阵表示就是惯量矩阵，这将在后续章节中详细讨论。
