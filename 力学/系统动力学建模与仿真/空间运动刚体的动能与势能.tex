\section{空间运动体元件的动能}

\paragraph{空间运动体元件动能的微元求和形式}

空间运动体元件的动能可用微元求和的形式表示为
\begin{equation}
T = \sum_{k=1}^{N} \frac{1}{2} m_k \dot{\boldsymbol{r}}_{Ok}^{\mathrm{T}} \dot{\boldsymbol{r}}_{Ok},
\end{equation}
式中
\begin{itemize}
\item $k$ 表示刚体上微元的标号；
\item $N$ 表示微元的总数；
\item $m_k$ 表示微元 $k$ 的质量；
\item $\boldsymbol{r}_{Ok}$ 表示微元 $k$ 在全局惯性坐标系 $Oxyz$ 中的坐标列阵；
\item $\dot{\boldsymbol{r}}_{Ok}$ 表示 $\boldsymbol{r}_{Ok}$ 对时间的一阶导数，即在 $Oxyz$ 中的绝对速度列阵。
\end{itemize}

为便于将体元件的动能表示为系统广义坐标 $\boldsymbol{q}$ 和广义速度 $\dot{\boldsymbol{q}}$ 的形式，为体元件引入一个连体坐标系 $R\xi\eta\zeta$，此时 $\boldsymbol{r}_{Ok}$ 可表示为
\begin{equation}
\boldsymbol{r}_{Ok} = \boldsymbol{r}_{OR} + \boldsymbol{A}_{OR} \boldsymbol{l}_{Rk},
\end{equation}
式中
\begin{itemize}
\item $\boldsymbol{r}_{OR}$ 表示 $R\xi\eta\zeta$ 连体系原点在全局惯性坐标系中的坐标列阵；
\item $\boldsymbol{A}_{OR}$ 表示从 $R\xi\eta\zeta$ 连体系到全局惯性坐标系的坐标变换矩阵；
\item $\boldsymbol{l}_{Rk}$ 表示微元 $k$ 在 $R\xi\eta\zeta$ 连体系中的坐标列阵。
\end{itemize}

上式对时间的一阶导数可表示为
\begin{equation}
\dot{\boldsymbol{r}}_{Ok} = \dot{\boldsymbol{r}}_{OR} + \boldsymbol{A}_{OR} \tilde{\boldsymbol{\omega}}_{OR|R} \boldsymbol{l}_{Rk},
\end{equation}
式中 $\boldsymbol{\omega}_{OR|R}$ 表示连体坐标系 $R\xi\eta\zeta$ 相对于全局惯性坐标系 $Oxyz$ 的绝对角速度在 $R\xi\eta\zeta$ 系中的投影列阵（推导详见第\ref{chap:kinematics}章\ref{sec:角速度矩阵}节）。

引入连体坐标系 $R\xi\eta\zeta$ 后，上式中的速度列阵可表示为
\begin{equation}
\dot{\boldsymbol{r}}_{Ok} = \dot{\boldsymbol{r}}_{OR} + \boldsymbol{A}_{OR} \tilde{\boldsymbol{\omega}}_{OR|R} \boldsymbol{l}_{Rk}.
\end{equation}

上式可进一步改写为
\begin{align}
\dot{\boldsymbol{r}}_{Ok} &= \dot{\boldsymbol{r}}_{OR} + \boldsymbol{A}_{OR} \tilde{\boldsymbol{\omega}}_{OR|R} \boldsymbol{l}_{Rk} \notag \\
&= \boldsymbol{A}_{OR} \boldsymbol{A}_{OR}^{\mathrm{T}} \dot{\boldsymbol{r}}_{OR} + \boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \boldsymbol{\omega}_{OR|R} \notag \\
&= \begin{bmatrix} \boldsymbol{A}_{OR} & \boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix} \begin{bmatrix} \boldsymbol{A}_{OR}^{\mathrm{T}} \dot{\boldsymbol{r}}_{OR} \\ \boldsymbol{\omega}_{OR|R} \end{bmatrix} \notag \\
&=: \boldsymbol{C}_{Rk} \boldsymbol{v}_R,
\end{align}
式中
\begin{itemize}
\item 第一行到第二行利用了旋转矩阵的正交性 $\boldsymbol{A}_{OR} \boldsymbol{A}_{OR}^{\mathrm{T}} = \boldsymbol{I}_3$，以及叉乘矩阵的性质 $\tilde{\boldsymbol{\omega}}_{OR|R} \boldsymbol{l}_{Rk} = -\tilde{\boldsymbol{l}}_{Rk} \boldsymbol{\omega}_{OR|R} = \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \boldsymbol{\omega}_{OR|R}$（其中 $\tilde{\boldsymbol{l}}_{Rk}$ 为由向量 $\boldsymbol{l}_{Rk}$ 构造的叉乘矩阵，满足反对称性 $\tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} = -\tilde{\boldsymbol{l}}_{Rk}$）；
\item 第二行到第三行利用了矩阵乘法的分块形式 $\boldsymbol{A}\boldsymbol{x} + \boldsymbol{B}\boldsymbol{y} = \begin{bmatrix} \boldsymbol{A} & \boldsymbol{B} \end{bmatrix} \begin{bmatrix} \boldsymbol{x} \\ \boldsymbol{y} \end{bmatrix}$；
\item $\boldsymbol{C}_{Rk} = \begin{bmatrix} \boldsymbol{A}_{OR} & \boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix}$ 为 $3 \times 6$ 矩阵；
\item $\boldsymbol{v}_R = \begin{bmatrix} \boldsymbol{A}_{OR}^{\mathrm{T}} \dot{\boldsymbol{r}}_{OR} \\ \boldsymbol{\omega}_{OR|R} \end{bmatrix}$ 为连体系 $R\xi\eta\zeta$ 原点 $R$ 的广义速度列阵。
\end{itemize}

\paragraph{用连体坐标系运动学量表示的体元件的动能}

空间运动体元件的动能
\begin{align}
T &= \sum_{k=1}^{N} \frac{1}{2} m_k \dot{\boldsymbol{r}}_{Ok}^{\mathrm{T}} \dot{\boldsymbol{r}}_{Ok} \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k (\boldsymbol{C}_{Rk} \boldsymbol{v}_R)^{\mathrm{T}} (\boldsymbol{C}_{Rk} \boldsymbol{v}_R) \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k \boldsymbol{v}_R^{\mathrm{T}} \boldsymbol{C}_{Rk}^{\mathrm{T}} \boldsymbol{C}_{Rk} \boldsymbol{v}_R \notag \\
&= \frac{1}{2} \boldsymbol{v}_R^{\mathrm{T}} \left( \sum_{k=1}^{N} m_k \boldsymbol{C}_{Rk}^{\mathrm{T}} \boldsymbol{C}_{Rk} \right) \boldsymbol{v}_R \notag \\
&=: \frac{1}{2} \boldsymbol{v}_R^{\mathrm{T}} \boldsymbol{M}_R \boldsymbol{v}_R,
\end{align}

\begin{definition}[体元件的质量矩阵]
空间运动体元件的质量矩阵定义为
\begin{equation}
\boldsymbol{M}_R = \sum_{k=1}^{N} m_k \begin{bmatrix} \boldsymbol{I}_3 & \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \\ \tilde{\boldsymbol{l}}_{Rk} & \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix}.
\end{equation}
\end{definition}

\begin{proof}
由 $\boldsymbol{C}_{Rk} = \begin{bmatrix} \boldsymbol{A}_{OR} & \boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix}$，可得
\begin{align}
\boldsymbol{M}_R &= \sum_{k=1}^{N} m_k \boldsymbol{C}_{Rk}^{\mathrm{T}} \boldsymbol{C}_{Rk} \notag \\
&= \sum_{k=1}^{N} m_k \begin{bmatrix} \boldsymbol{A}_{OR} \\ \boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix}^{\mathrm{T}} \begin{bmatrix} \boldsymbol{A}_{OR} \\ \boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix} \notag \\
&= \sum_{k=1}^{N} m_k \begin{bmatrix} \boldsymbol{A}_{OR}^{\mathrm{T}} \\ \tilde{\boldsymbol{l}}_{Rk} \boldsymbol{A}_{OR}^{\mathrm{T}} \end{bmatrix} \begin{bmatrix} \boldsymbol{A}_{OR} & \boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix} \notag \\
&= \sum_{k=1}^{N} m_k \begin{bmatrix} \boldsymbol{A}_{OR}^{\mathrm{T}}\boldsymbol{A}_{OR} & \boldsymbol{A}_{OR}^{\mathrm{T}}\boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \\ \tilde{\boldsymbol{l}}_{Rk} \boldsymbol{A}_{OR}^{\mathrm{T}}\boldsymbol{A}_{OR} & \tilde{\boldsymbol{l}}_{Rk} \boldsymbol{A}_{OR}^{\mathrm{T}}\boldsymbol{A}_{OR} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix}.
\end{align}
利用旋转矩阵的正交性 $\boldsymbol{A}_{OR}^{\mathrm{T}}\boldsymbol{A}_{OR} = \boldsymbol{I}_3$，即得
\begin{equation}
\boldsymbol{M}_R = \sum_{k=1}^{N} m_k \begin{bmatrix} \boldsymbol{I}_3 & \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \\ \tilde{\boldsymbol{l}}_{Rk} & \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix}.
\end{equation}
\end{proof}

由此可知，此处定义的空间运动体元件的质量矩阵 $\boldsymbol{M}_R$ 为定常（时不变）矩阵。

\begin{definition}[体元件时不变质量矩阵]
空间运动体元件时不变质量矩阵可表示为
\begin{equation}
\boldsymbol{M}_R = \begin{bmatrix} m \boldsymbol{I}_3 & m \tilde{\boldsymbol{l}}_{RC}^{\mathrm{T}} \\ m \tilde{\boldsymbol{l}}_{RC} & \boldsymbol{J}_{R|R} \end{bmatrix},
\end{equation}
式中
\begin{itemize}
\item $m = \sum_{k=1}^{N} m_k$ 表示体元件的质量；
\item $\boldsymbol{l}_{RC} = \frac{1}{m}\sum_{k=1}^{N} m_k \boldsymbol{l}_{Rk}$ 表示体元件质心在连体坐标系 $R\xi\eta\zeta$ 中的坐标列阵；
\item $\boldsymbol{J}_{R|R} = \sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}}$ 表示体元件关于连体系原点 $R$ 的惯性张量矩阵。
\end{itemize}
\end{definition}

\begin{proof}
由定义1中的质量矩阵表达式，展开求和得
\begin{align}
\boldsymbol{M}_R &= \sum_{k=1}^{N} m_k \begin{bmatrix} \boldsymbol{I}_3 & \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \\ \tilde{\boldsymbol{l}}_{Rk} & \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix} \notag \\
&= \begin{bmatrix} \boldsymbol{I}_3 \sum_{k=1}^{N} m_k & \sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \\ \sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} & \sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} \end{bmatrix}.
\end{align}
利用叉乘矩阵的线性性质 $\sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} = \widetilde{\left(\sum_{k=1}^{N} m_k \boldsymbol{l}_{Rk}\right)} = \left[\sum_{k=1}^{N} m_k \boldsymbol{l}_{Rk}\right]_{\times}$，以及质心定义 $\sum_{k=1}^{N} m_k \boldsymbol{l}_{Rk} = m \boldsymbol{l}_{RC}$，可得
\begin{align}
\sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} &= m \tilde{\boldsymbol{l}}_{RC}, \\
\sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} &= m \tilde{\boldsymbol{l}}_{RC}^{\mathrm{T}}.
\end{align}
代入即得
\begin{equation}
\boldsymbol{M}_R = \begin{bmatrix} m \boldsymbol{I}_3 & m \tilde{\boldsymbol{l}}_{RC}^{\mathrm{T}} \\ m \tilde{\boldsymbol{l}}_{RC} & \boldsymbol{J}_{R|R} \end{bmatrix}.
\end{equation}
\end{proof}

\begin{definition}[惯性张量矩阵的分量表达式]
体元件质量矩阵中 $\boldsymbol{J}_{R|R}$ 的分量表达式为
\begin{equation}
\boldsymbol{J}_{R|R} = \begin{bmatrix} \sum_{k=1}^{N} m_k (\eta_{Rk}^2 + \zeta_{Rk}^2) & -\sum_{k=1}^{N} m_k \xi_{Rk} \eta_{Rk} & -\sum_{k=1}^{N} m_k \xi_{Rk} \zeta_{Rk} \\ -\sum_{k=1}^{N} m_k \eta_{Rk} \xi_{Rk} & \sum_{k=1}^{N} m_k (\xi_{Rk}^2 + \zeta_{Rk}^2) & -\sum_{k=1}^{N} m_k \eta_{Rk} \zeta_{Rk} \\ -\sum_{k=1}^{N} m_k \zeta_{Rk} \xi_{Rk} & -\sum_{k=1}^{N} m_k \zeta_{Rk} \eta_{Rk} & \sum_{k=1}^{N} m_k (\xi_{Rk}^2 + \eta_{Rk}^2) \end{bmatrix}.
\end{equation}
\end{definition}

\begin{proof}
由 $\boldsymbol{J}_{R|R} = \sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}}$，利用叉乘矩阵的性质 $\tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} = l_{Rk}^2 \boldsymbol{I}_3 - \boldsymbol{l}_{Rk} \boldsymbol{l}_{Rk}^{\mathrm{T}}$，可得
\begin{align}
\boldsymbol{J}_{R|R} &= \sum_{k=1}^{N} m_k \left( l_{Rk}^2 \boldsymbol{I}_3 - \boldsymbol{l}_{Rk} \boldsymbol{l}_{Rk}^{\mathrm{T}} \right) \notag \\
&= \sum_{k=1}^{N} m_k \begin{bmatrix} \eta_{Rk}^2 + \zeta_{Rk}^2 & -\xi_{Rk} \eta_{Rk} & -\xi_{Rk} \zeta_{Rk} \\ -\eta_{Rk} \xi_{Rk} & \xi_{Rk}^2 + \zeta_{Rk}^2 & -\eta_{Rk} \zeta_{Rk} \\ -\zeta_{Rk} \xi_{Rk} & -\zeta_{Rk} \eta_{Rk} & \xi_{Rk}^2 + \eta_{Rk}^2 \end{bmatrix}.
\end{align}
展开即得上述分量表达式。
\end{proof}

\begin{definition}[惯性积矩阵]
体元件惯性积矩阵定义为
\begin{equation}
\boldsymbol{I}_{R|R} = \begin{bmatrix} \sum_{k=1}^{N} m_k (\eta_{Rk}^2 + \zeta_{Rk}^2) & \sum_{k=1}^{N} m_k \xi_{Rk} \eta_{Rk} & \sum_{k=1}^{N} m_k \xi_{Rk} \zeta_{Rk} \\ \sum_{k=1}^{N} m_k \eta_{Rk} \xi_{Rk} & \sum_{k=1}^{N} m_k (\xi_{Rk}^2 + \zeta_{Rk}^2) & \sum_{k=1}^{N} m_k \eta_{Rk} \zeta_{Rk} \\ \sum_{k=1}^{N} m_k \zeta_{Rk} \xi_{Rk} & \sum_{k=1}^{N} m_k \zeta_{Rk} \eta_{Rk} & \sum_{k=1}^{N} m_k (\xi_{Rk}^2 + \eta_{Rk}^2) \end{bmatrix}.
\end{equation}
\end{definition}

\section{空间运动体元件的重力势能}

空间运动体元件的重力势能可用微元求和的形式表示为
\begin{align}
V_{\text{gravity}} &= -\sum_{k=1}^{N} m_k \boldsymbol{g}^{\mathrm{T}} \boldsymbol{r}_{Ok} \notag \\
&= -\boldsymbol{g}^{\mathrm{T}} \sum_{k=1}^{N} m_k \boldsymbol{r}_{Ok} \notag \\
&= -\boldsymbol{g}^{\mathrm{T}} \sum_{k=1}^{N} m_k (\boldsymbol{r}_{OR} + \boldsymbol{A}_{OR} \boldsymbol{l}_{Rk}) \notag \\
&= -\boldsymbol{g}^{\mathrm{T}} (m \boldsymbol{r}_{OR} + m \boldsymbol{A}_{OR} \boldsymbol{l}_{RC}) \notag \\
&= -m \boldsymbol{g}^{\mathrm{T}} (\boldsymbol{r}_{OR} + \boldsymbol{A}_{OR} \boldsymbol{l}_{RC}).
\end{align}

\section{空间运动体元件的转动惯量张量}

绕连体坐标系 $R\xi\eta\zeta$ 原点 $R$ 作定点转动的体元件的动能可表示为
\begin{align}
T &= \sum_{k=1}^{N} \frac{1}{2} m_k (\vec{\omega}_{OR} \times \vec{r}_{Rk}) \cdot (\vec{\omega}_{OR} \times \vec{r}_{Rk}) \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k [\vec{r}_{Rk} \times (\vec{\omega}_{OR} \times \vec{r}_{Rk})] \cdot \vec{\omega}_{OR} \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k [(\vec{r}_{Rk} \cdot \vec{r}_{Rk}) \vec{\omega}_{OR} - (\vec{r}_{Rk} \cdot \vec{\omega}_{OR}) \vec{r}_{Rk}] \cdot \vec{\omega}_{OR} \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k [\vec{\omega}_{OR} (\vec{r}_{Rk} \cdot \vec{r}_{Rk}) - (\vec{\omega}_{OR} \cdot \vec{r}_{Rk}) \vec{r}_{Rk}] \cdot \vec{\omega}_{OR} \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k \left[ \vec{\omega}_{OR} \cdot \vec{\boldsymbol{I}} (\vec{r}_{Rk} \cdot \vec{r}_{Rk}) - \vec{\omega}_{OR} \cdot \vec{r}_{Rk} \vec{r}_{Rk} \right] \cdot \vec{\omega}_{OR} \notag \\
&= \vec{\omega}_{OR} \cdot \frac{1}{2} \sum_{k=1}^{N} m_k \left[ \vec{\boldsymbol{I}} (\vec{r}_{Rk} \cdot \vec{r}_{Rk}) - \vec{r}_{Rk} \vec{r}_{Rk} \right] \cdot \vec{\omega}_{OR} \notag \\
&=: \frac{1}{2} \vec{\omega}_{OR} \cdot \vec{\boldsymbol{J}}_R \cdot \vec{\omega}_{OR},
\end{align}

\begin{definition}[转动惯量张量]
体元件相对于点 $R$ 的转动惯量张量 $\vec{\boldsymbol{J}}_R$ 定义为
\begin{equation}
\vec{\boldsymbol{J}}_R = \sum_{k=1}^{N} m_k \left[ \vec{\boldsymbol{I}} (\vec{r}_{Rk} \cdot \vec{r}_{Rk}) - \vec{r}_{Rk} \vec{r}_{Rk} \right].
\end{equation}
其在惯性系 $Oxyz$ 和连体系 $R\xi\eta\zeta$ 中的投影矩阵分别为
\begin{align}
\boldsymbol{J}_{R|O} &= \sum_{k=1}^{N} m_k \left[ \boldsymbol{I}_3 (\boldsymbol{r}_{Rk}^{\mathrm{T}} \boldsymbol{r}_{Rk}) - \boldsymbol{r}_{Rk} \boldsymbol{r}_{Rk}^{\mathrm{T}} \right], \\
\boldsymbol{J}_{R|R} &= \sum_{k=1}^{N} m_k \left[ \boldsymbol{I}_3 (\boldsymbol{l}_{Rk}^{\mathrm{T}} \boldsymbol{l}_{Rk}) - \boldsymbol{l}_{Rk} \boldsymbol{l}_{Rk}^{\mathrm{T}} \right] = \sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}},
\end{align}
且满足坐标变换关系 $\boldsymbol{J}_{R|O} = \boldsymbol{A}_{OR} \boldsymbol{J}_{R|R} \boldsymbol{A}_{OR}^{\mathrm{T}}$。
\end{definition}

\begin{proof}
绕连体坐标系 $R\xi\eta\zeta$ 原点 $R$ 作定点转动的体元件的动能为
\begin{align}
T &= \sum_{k=1}^{N} \frac{1}{2} m_k (\vec{\omega}_{OR} \times \vec{r}_{Rk}) \cdot (\vec{\omega}_{OR} \times \vec{r}_{Rk}) \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k [\vec{r}_{Rk} \times (\vec{\omega}_{OR} \times \vec{r}_{Rk})] \cdot \vec{\omega}_{OR} \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k [(\vec{r}_{Rk} \cdot \vec{r}_{Rk}) \vec{\omega}_{OR} - (\vec{r}_{Rk} \cdot \vec{\omega}_{OR}) \vec{r}_{Rk}] \cdot \vec{\omega}_{OR} \notag \\
&= \sum_{k=1}^{N} \frac{1}{2} m_k [\vec{\omega}_{OR} (\vec{r}_{Rk} \cdot \vec{r}_{Rk}) - (\vec{\omega}_{OR} \cdot \vec{r}_{Rk}) \vec{r}_{Rk}] \cdot \vec{\omega}_{OR} \notag \\
&= \vec{\omega}_{OR} \cdot \frac{1}{2} \sum_{k=1}^{N} m_k \left[ \vec{\boldsymbol{I}} (\vec{r}_{Rk} \cdot \vec{r}_{Rk}) - \vec{r}_{Rk} \vec{r}_{Rk} \right] \cdot \vec{\omega}_{OR} \notag \\
&=: \frac{1}{2} \vec{\omega}_{OR} \cdot \vec{\boldsymbol{J}}_R \cdot \vec{\omega}_{OR}.
\end{align}
由此得到转动惯量张量的定义。

在惯性系 $Oxyz$ 中，张量的投影矩阵为
\begin{align}
\vec{\boldsymbol{J}}_R &= \boldsymbol{e}_O^{\mathrm{T}} \sum_{k=1}^{N} m_k \left[ \boldsymbol{I}_3 (\boldsymbol{r}_{Rk}^{\mathrm{T}} \boldsymbol{r}_{Rk}) - \boldsymbol{r}_{Rk} \boldsymbol{r}_{Rk}^{\mathrm{T}} \right] \boldsymbol{e}_O \notag \\
&=: \boldsymbol{e}_O^{\mathrm{T}} \boldsymbol{J}_{R|O} \boldsymbol{e}_O.
\end{align}

在连体系 $R\xi\eta\zeta$ 中，张量的投影矩阵为
\begin{align}
\vec{\boldsymbol{J}}_R &= \boldsymbol{e}_R^{\mathrm{T}} \sum_{k=1}^{N} m_k \left[ \boldsymbol{I}_3 (\boldsymbol{l}_{Rk}^{\mathrm{T}} \boldsymbol{l}_{Rk}) - \boldsymbol{l}_{Rk} \boldsymbol{l}_{Rk}^{\mathrm{T}} \right] \boldsymbol{e}_R \notag \\
&=: \boldsymbol{e}_R^{\mathrm{T}} \boldsymbol{J}_{R|R} \boldsymbol{e}_R.
\end{align}

利用坐标变换 $\boldsymbol{r}_{Rk} = \boldsymbol{A}_{OR} \boldsymbol{l}_{Rk}$，可得
\begin{align}
\boldsymbol{J}_{R|O} &= \sum_{k=1}^{N} m_k \left[ \boldsymbol{I}_3 (\boldsymbol{l}_{Rk}^{\mathrm{T}} \boldsymbol{A}_{OR}^{\mathrm{T}} \boldsymbol{A}_{OR} \boldsymbol{l}_{Rk}) - \boldsymbol{A}_{OR} \boldsymbol{l}_{Rk} \boldsymbol{l}_{Rk}^{\mathrm{T}} \boldsymbol{A}_{OR}^{\mathrm{T}} \right] \notag \\
&= \boldsymbol{A}_{OR} \left( \sum_{k=1}^{N} m_k \left[ \boldsymbol{I}_3 (\boldsymbol{l}_{Rk}^{\mathrm{T}} \boldsymbol{l}_{Rk}) - \boldsymbol{l}_{Rk} \boldsymbol{l}_{Rk}^{\mathrm{T}} \right] \right) \boldsymbol{A}_{OR}^{\mathrm{T}} \notag \\
&= \boldsymbol{A}_{OR} \boldsymbol{J}_{R|R} \boldsymbol{A}_{OR}^{\mathrm{T}}.
\end{align}
\end{proof}

\begin{theorem}[转动惯量张量的平行移轴定理]
体元件相对于点 $R$ 的转动惯量张量可借助质心点 $C$ 作如下分解
\begin{equation}
\vec{\boldsymbol{J}}_R = m \left[ \vec{\boldsymbol{I}} (\vec{r}_{RC} \cdot \vec{r}_{RC}) - (\vec{r}_{RC} \vec{r}_{RC}) \right] + \vec{\boldsymbol{J}}_C,
\end{equation}
式中 $\vec{\boldsymbol{J}}_C$ 为体元件相对于质心 $C$ 的转动惯量张量。
\end{theorem}

\begin{proof}
由转动惯量张量的定义
\begin{align}
\vec{\boldsymbol{J}}_R &= \sum_{k=1}^{N} m_k \left[ \vec{\boldsymbol{I}} (\vec{r}_{Rk} \cdot \vec{r}_{Rk}) - \vec{r}_{Rk} \vec{r}_{Rk} \right].
\end{align}

利用矢量分解 $\vec{r}_{Rk} = \vec{r}_{RC} + \vec{r}_{Ck}$，代入得
\begin{align}
\vec{\boldsymbol{J}}_R &= \sum_{k=1}^{N} m_k \vec{\boldsymbol{I}} [(\vec{r}_{RC} + \vec{r}_{Ck}) \cdot (\vec{r}_{RC} + \vec{r}_{Ck})] \notag \\
&\quad - \sum_{k=1}^{N} m_k [(\vec{r}_{RC} + \vec{r}_{Ck})(\vec{r}_{RC} + \vec{r}_{Ck})].
\end{align}

展开得
\begin{align}
\vec{\boldsymbol{J}}_R &= \sum_{k=1}^{N} m_k \vec{\boldsymbol{I}} (\vec{r}_{RC} \cdot \vec{r}_{RC}) + 2\sum_{k=1}^{N} m_k \vec{\boldsymbol{I}} (\vec{r}_{RC} \cdot \vec{r}_{Ck}) + \sum_{k=1}^{N} m_k \vec{\boldsymbol{I}} (\vec{r}_{Ck} \cdot \vec{r}_{Ck}) \notag \\
&\quad - \sum_{k=1}^{N} m_k (\vec{r}_{RC} \vec{r}_{RC}) - \sum_{k=1}^{N} m_k (\vec{r}_{RC} \vec{r}_{Ck}) - \sum_{k=1}^{N} m_k (\vec{r}_{Ck} \vec{r}_{RC}) - \sum_{k=1}^{N} m_k (\vec{r}_{Ck} \vec{r}_{Ck}).
\end{align}

由于质心的定义满足 $\sum_{k=1}^{N} m_k \vec{r}_{Ck} = \boldsymbol{0}$，交叉项全部消去，得
\begin{align}
\vec{\boldsymbol{J}}_R &= m \vec{\boldsymbol{I}} (\vec{r}_{RC} \cdot \vec{r}_{RC}) + \sum_{k=1}^{N} m_k \vec{\boldsymbol{I}} (\vec{r}_{Ck} \cdot \vec{r}_{Ck}) \notag \\
&\quad - m (\vec{r}_{RC} \vec{r}_{RC}) - \sum_{k=1}^{N} m_k (\vec{r}_{Ck} \vec{r}_{Ck}) \notag \\
&= m \left[ \vec{\boldsymbol{I}} (\vec{r}_{RC} \cdot \vec{r}_{RC}) - (\vec{r}_{RC} \vec{r}_{RC}) \right] + \sum_{k=1}^{N} m_k \left[ \vec{\boldsymbol{I}} (\vec{r}_{Ck} \cdot \vec{r}_{Ck}) - \vec{r}_{Ck} \vec{r}_{Ck} \right].
\end{align}

最后一项即为相对于质心 $C$ 的转动惯量张量 $\vec{\boldsymbol{J}}_C$，故
\begin{equation}
\vec{\boldsymbol{J}}_R = m \left[ \vec{\boldsymbol{I}} (\vec{r}_{RC} \cdot \vec{r}_{RC}) - (\vec{r}_{RC} \vec{r}_{RC}) \right] + \vec{\boldsymbol{J}}_C.
\end{equation}
\end{proof}

\begin{theorem}[惯性主轴系的存在性]
对于体元件相对于任意点 $R$ 的转动惯量张量，总存在一个坐标系 $R'$（称为惯性主轴系），使得转动惯量张量在该系中的投影矩阵 $\boldsymbol{J}_{R|R'}$ 为对角矩阵。
\end{theorem}

\begin{proof}
体元件相对于点 $R$ 的转动惯量张量在 $R\xi\eta\zeta$ 系中的投影矩阵为
\begin{align}
\boldsymbol{J}_{R|R} &= \sum_{k=1}^{N} m_k \tilde{\boldsymbol{l}}_{Rk} \tilde{\boldsymbol{l}}_{Rk}^{\mathrm{T}} = \sum_{k=1}^{N} m_k \left( l_{Rk}^2 \boldsymbol{I}_3 - \boldsymbol{l}_{Rk} \boldsymbol{l}_{Rk}^{\mathrm{T}} \right) \notag \\
&= \begin{bmatrix} \sum_{k=1}^{N} m_k (\eta_{Rk}^2 + \zeta_{Rk}^2) & -\sum_{k=1}^{N} m_k \xi_{Rk} \eta_{Rk} & -\sum_{k=1}^{N} m_k \xi_{Rk} \zeta_{Rk} \\ -\sum_{k=1}^{N} m_k \eta_{Rk} \xi_{Rk} & \sum_{k=1}^{N} m_k (\xi_{Rk}^2 + \zeta_{Rk}^2) & -\sum_{k=1}^{N} m_k \eta_{Rk} \zeta_{Rk} \\ -\sum_{k=1}^{N} m_k \zeta_{Rk} \xi_{Rk} & -\sum_{k=1}^{N} m_k \zeta_{Rk} \eta_{Rk} & \sum_{k=1}^{N} m_k (\xi_{Rk}^2 + \eta_{Rk}^2) \end{bmatrix} \notag \\
&=: \begin{bmatrix} J_{\xi\xi} & J_{\xi\eta} & J_{\xi\zeta} \\ J_{\eta\xi} & J_{\eta\eta} & J_{\eta\zeta} \\ J_{\zeta\xi} & J_{\zeta\eta} & J_{\zeta\zeta} \end{bmatrix}_{R|R}.
\end{align}

一般情况下 $\boldsymbol{J}_{R|R}$ 为非对角矩阵。对于 $3 \times 3$ 的实对称矩阵 $\boldsymbol{J}_{R|R}$，由矩阵特征值相关理论可知，一定存在三个线性独立且模为 1 的 $3 \times 1$ 的实数列阵 $\boldsymbol{\varphi}_i$，使得
\begin{equation}
\boldsymbol{J}_{R|R} \boldsymbol{\varphi}_i = \lambda_i \boldsymbol{\varphi}_i, \tag{I}
\end{equation}
式中实数 $\lambda_i$ 称为矩阵 $\boldsymbol{J}_{R|R}$ 的特征值，特征向量 $\boldsymbol{\varphi}_i$ 满足两两正交，
\begin{equation}
\boldsymbol{\varphi}_i^{\mathrm{T}} \boldsymbol{\varphi}_j = \delta_{ij}, \quad \boldsymbol{\varphi}_3 = \boldsymbol{\varphi}_1 \times \boldsymbol{\varphi}_2.
\end{equation}

由式 (I) 可得
\begin{equation}
\boldsymbol{J}_{R|R} \begin{bmatrix} \boldsymbol{\varphi}_1 & \boldsymbol{\varphi}_2 & \boldsymbol{\varphi}_3 \end{bmatrix} = \begin{bmatrix} \boldsymbol{\varphi}_1 & \boldsymbol{\varphi}_2 & \boldsymbol{\varphi}_3 \end{bmatrix} \begin{bmatrix} \lambda_1 & 0 & 0 \\ 0 & \lambda_2 & 0 \\ 0 & 0 & \lambda_3 \end{bmatrix}.
\end{equation}

上式可简记为
\begin{equation}
\boldsymbol{J}_{R|R} \boldsymbol{\Phi} = \boldsymbol{\Phi} \mathrm{diag}(\boldsymbol{\lambda}),
\end{equation}
式中
\begin{equation}
\boldsymbol{\Phi} = \begin{bmatrix} \boldsymbol{\varphi}_1 & \boldsymbol{\varphi}_2 & \boldsymbol{\varphi}_3 \end{bmatrix}, \quad \boldsymbol{\Phi}^{-1} = \boldsymbol{\Phi}^{\mathrm{T}}, \quad \boldsymbol{\lambda} = \begin{bmatrix} \lambda_1 \\ \lambda_2 \\ \lambda_3 \end{bmatrix}.
\end{equation}

由
\begin{equation}
\boldsymbol{J}_{R|R} \boldsymbol{\Phi} = \boldsymbol{\Phi} \mathrm{diag}(\boldsymbol{\lambda}), \quad \boldsymbol{\Phi}^{-1} = \boldsymbol{\Phi}^{\mathrm{T}}
\end{equation}

可得
\begin{equation}
\mathrm{diag}(\boldsymbol{\lambda}) = \boldsymbol{\Phi}^{\mathrm{T}} \boldsymbol{J}_{R|R} \boldsymbol{\Phi}.
\end{equation}

上式可改写为
\begin{equation}
\boldsymbol{J}_{R|R'} = \boldsymbol{A}_{RR'}^{\mathrm{T}} \boldsymbol{J}_{R|R} \boldsymbol{A}_{RR'},
\end{equation}
式中 $\boldsymbol{J}_{R|R'}$ 为对角矩阵，$\boldsymbol{A}_{RR'}$ 为从 $R'$ 系到 $R$ 系的坐标变换矩阵，$R'$ 系称为转动惯量张量 $\vec{\boldsymbol{J}}_R$ 的惯性主轴系。
\end{proof}
